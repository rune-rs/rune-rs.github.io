8/// A syntax tree.
9///
10/// A tree is constructed through a [Builder][crate::Builder] or by modifying an
11/// existing tree through a [ChangeSet][crate::edit::ChangeSet].
12///
13/// # Type parameters and bounds
14///
15/// The two type parameters of the tree determine the following properties:
16/// * `T` is the data stored in the tree.
17/// * `F` is the [Flavor], declared with [`flavor!`][crate::flavor!]. Its
18///   [`Index`][Flavor::Index] determines the numerical bounds of spans, and if
19///   set to [Empty][crate::Empty] the tree does not store any spans. Its
20///   [`Width`][Flavor::Width] determines the bounds of pointers in the tree,
21///   which decides how many nodes can be stored in it. It also selects the
22///   [`Storage`][Flavor::Storage] used and its error type.
23///
24/// To use the default values, use the [Builder::new][crate::Builder::new]
25/// constructor.
26/// # Layout
27///
28/// Tokens are stored in a flat array in source order, and interior nodes (the
29/// ones opened with [`Builder::open`][crate::Builder::open] or wrapped with
30/// [`Builder::close_at`][crate::Builder::close_at]) in a second array. Node and
31/// token ids are indexes into their own array, so they are not dense: token ids
32/// skip the internal entries which carry the gaps explicit spans leave between
33/// tokens.
34pub struct Tree<T, F>
35where
36    T: Copy,
37    F: Flavor,
38{
39    /// Interior nodes.
40    nodes: F::Storage<NodeLinks<T, F::Index, F::Pointer>>,
41    /// Tokens and gap entries, in source order.
42    tokens: F::Storage<TokenLinks<T, F::Index, F::Pointer>>,
43    /// The span of the whole tree.
44    span: Span<F::Index>,
45    /// The first interior node at the root.
46    first: Option<F::Pointer>,
47    /// The last interior node at the root.
48    last: Option<F::Pointer>,
49    /// The number of gap entries in `tokens`.
50    gaps: usize,
51}
52
53impl<T, F> Tree<T, F>
54where
55    T: Copy,

313    pub(crate) fn push_token(
314        &mut self,
315        links: TokenLinks<T, F::Index, F::Pointer>,
316    ) -> Result<(), F::Error> {
317        if links.is_gap() {
318            self.gaps = self.gaps.saturating_add(1);

333    /// The start of the token entry at `index`, which is the end of the entry
334    /// before it.
335    pub(crate) fn token_start(&self, index: usize) -> F::Index {
336        match index.checked_sub(1).and_then(|n| self.tokens.get(n)) {
337            Some(prev) => prev.end,

347    /// The span of an element.
348    pub(crate) fn span_of(&self, id: Id<F::Pointer>) -> Option<Span<F::Index>> {
349        match id {
350            Id::Node(id) => Some(self.node_links(id)?.span),
351            Id::Token(id) => {
352                let index = id.get();
353                let links = self.tokens.get(index)?;
354                Some(Span::new(self.token_start(index), links.end))
355            }
356        }
357    }

359    /// The element at a level which follows the position before token entry
360    /// `tok`, where `node` is the next interior node at the level, if any.
361    ///
362    /// This is the merge step between the token array and a chain of interior
363    /// nodes: an interior node comes first if its tokens start at or before
364    /// `tok`.
365    pub(crate) fn next_at(
366        &self,
367        parent: Option<F::Pointer>,
368        tok: usize,
369        node: Option<F::Pointer>,
370    ) -> Option<Id<F::Pointer>> {
371        if let Some(id) = node {
372            if self.node_links(id)?.tok_start.get() <= tok {
373                return Some(Id::Node(id));
374            }
375        }
376
377        let mut index = tok;
378        let mut links = self.tokens.get(index)?;
379
380        // A gap entry is always directly followed by its token.
381        if links.is_gap() {
382            index = index.checked_add(1)?;
383            links = self.tokens.get(index)?;
384        }
385
386        if links.parent() != parent {
387            return None;
388        }
389
390        Some(Id::Token(F::Pointer::new(index)?))
391    }

393    /// The element at a level which precedes the position before token entry
394    /// `tok`, where `node` is the previous interior node at the level, if any.
395    pub(crate) fn prev_at(
396        &self,
397        parent: Option<F::Pointer>,
398        tok: usize,
399        node: Option<F::Pointer>,
400    ) -> Option<Id<F::Pointer>> {
401        let mut tok = tok;
402
403        // A position right after a gap entry is the same as the one before it,
404        // since nothing is ever placed between a gap and its token.
405        if let Some(index) = tok.checked_sub(1) {
406            if self.tokens.get(index).is_some_and(|links| links.is_gap()) {
407                tok = index;
408            }
409        }
410
411        if let Some(id) = node {
412            if self.node_links(id)?.tok_end.get() >= tok {
413                return Some(Id::Node(id));
414            }
415        }
416
417        let index = tok.checked_sub(1)?;
418        let links = self.tokens.get(index)?;
419
420        if links.is_gap() || links.parent() != parent {
421            return None;
422        }
423
424        Some(Id::Token(F::Pointer::new(index)?))
425    }

427    /// The position at a level after `id`, where `node` is the next interior
428    /// node at the level if `id` is a token. This is what [`Tree::next_at`]
429    /// takes to find the element which follows `id`.
430    pub(crate) fn after(
431        &self,
432        id: Id<F::Pointer>,
433        node: Option<F::Pointer>,
434    ) -> Option<(usize, Option<F::Pointer>)> {
435        match id {
436            Id::Node(n) => {
437                let links = self.node_links(n)?;
438                Some((links.tok_end.get(), links.next))
439            }
440            Id::Token(t) => Some((t.get().checked_add(1)?, node)),
441        }
442    }

444    /// The position at a level before `id`, where `node` is the previous
445    /// interior node at the level if `id` is a token. This is what
446    /// [`Tree::prev_at`] takes to find the element which precedes `id`.
447    pub(crate) fn before(
448        &self,
449        id: Id<F::Pointer>,
450        node: Option<F::Pointer>,
451    ) -> Option<(usize, Option<F::Pointer>)> {
452        match id {
453            Id::Node(n) => {
454                let links = self.node_links(n)?;
455                Some((links.tok_start.get(), links.prev))
456            }
457            Id::Token(t) => Some((t.get(), node)),
458        }
459    }

469    /// The interior nodes at a level immediately before and after token entry
470    /// `index`.
471    ///
472    /// This walks the chain of interior nodes at the level, so it is linear in
473    /// their number.
474    pub(crate) fn nodes_around(
475        &self,
476        parent: Option<F::Pointer>,
477        index: usize,
478    ) -> (Option<F::Pointer>, Option<F::Pointer>) {
479        let mut prev = None;
480        let mut cur = self.level_first(parent);
481
482        while let Some(id) = cur {
483            let Some(links) = self.node_links(id) else {
484                break;
485            };
486
487            if links.tok_start.get() > index {
488                break;
489            }
490
491            prev = Some(id);
492            cur = links.next;
493        }
494
495        (prev, cur)
496    }

587    /// Query for the smallest node which contains the given [Range].
588    ///
589    /// A node contains the range when it starts at or before `range.start`,
590    /// ends at or after `range.end`, and `range.start` lies before its end.
591    /// Tokens are nodes too, so a range within a single token returns that
592    /// token, however deeply it is nested.
593    ///
594    /// * Returns `None` if `range.start > range.end`, or if no node contains
595    ///   the range, such as one which spans two nodes at the root of the tree.
596    /// * An empty range on the boundary between two tokens belongs to the token
597    ///   which starts there. An empty range at the end of the tree is contained
598    ///   by no node and returns `None`.
599    /// * Only tokens with a non-empty span and their ancestors are considered,
600    ///   so empty nodes and empty tokens are never returned.
601    ///
602    /// Tokens are looked up by binary search, which assumes they were added
603    /// to the tree in non-decreasing order of their spans.
604    ///
605    /// # Examples
606    ///
607    /// ```
608    /// let tree = syntree::tree! {
609    ///     "root" => {
610    ///         "child1" => {
611    ///             ("token1", 3)
612    ///         },
613    ///         "child2" => {
614    ///             "nested1" => {
615    ///                 ("token2", 4),
616    ///             },
617    ///             ("token3", 1),
618    ///         },
619    ///         "child3" => {
620    ///             ("token4", 5)
621    ///         }
622    ///     },
623    ///     "root2" => {}
624    /// };
625    ///
626    /// let node = tree.node_with_range(0..0).ok_or("missing 0")?;
627    /// assert_eq!(node.value(), "token1");
628    ///
629    /// let node = tree.node_with_range(0..3).ok_or("missing 0..3")?;
630    /// assert_eq!(node.value(), "token1");
631    ///
632    /// let node = tree.node_with_range(3..3).ok_or("missing 3")?;
633    /// assert_eq!(node.value(), "token2");
634    ///
635    /// let node = tree.node_with_range(3..7).ok_or("missing 3..7")?;
636    /// assert_eq!(node.value(), "token2");
637    /// assert_eq!(node.parent().map(|n| n.value()), Some("nested1"));
638    ///
639    /// let node = tree.node_with_range(7..8).ok_or("missing 7..8")?;
640    /// assert_eq!(node.value(), "token3");
641    ///
642    /// let node = tree.node_with_range(3..8).ok_or("missing 3..8")?;
643    /// assert_eq!(node.value(), "child2");
644    ///
645    /// let node = tree.node_with_range(8..13).ok_or("missing 8..13")?;
646    /// assert_eq!(node.value(), "token4");
647    ///
648    /// let node = tree.node_with_range(2..4).ok_or("missing 2..4")?;
649    /// assert_eq!(node.value(), "root");
650    ///
651    /// // The end of the tree, where the empty "root2" sits.
652    /// assert!(tree.node_with_range(13..13).is_none());
653    /// assert!(tree.node_with_range(0..14).is_none());
654    /// assert!(tree.node_with_range(4..2).is_none());
655    /// # Ok::<_, Box<dyn core::error::Error>>(())
656    /// ```
657    ///
658    /// Range queries work as expected with checkpoints:
659    ///
660    /// ```
661    /// let mut tree = syntree::Builder::new();
662    ///
663    /// let c = tree.checkpoint()?;
664    /// tree.open("child")?;
665    /// tree.token("lit", 3)?;
666    /// tree.close()?;
667    /// tree.close_at(c, "root")?;
668    /// tree.token("sibling", 3)?;
669    ///
670    /// let tree = tree.build()?;
671    ///
672    /// let lit = tree.node_with_range(0..3).ok_or("missing at 0..3")?;
673    /// assert_eq!(lit.value(), "lit");
674    /// assert_eq!(lit.parent().map(|n| n.value()), Some("child"));
675    ///
676    /// assert!(tree.node_with_range(2..4).is_none());
677    /// # Ok::<_, Box<dyn core::error::Error>>(())
678    /// ```
679    #[must_use]
680    pub fn node_with_range(&self, span: Range<usize>) -> Option<Element<'_, T, F>> {
681        let start = F::Index::from_usize(span.start)?;
682        let end = F::Index::from_usize(span.end)?;
683        self.node_with_span_internal(start, end)
684    }
685
686    /// Query for the smallest node which contains the given [Span].
687    ///
688    /// This behaves like [Tree::node_with_range]: tokens are returned when
689    /// they contain the span, and `None` is returned if `span.start >
690    /// span.end`, if no node contains the span, or for an empty span at the
691    /// end of the tree. Empty nodes and empty tokens are never returned.
692    ///
693    /// Tokens are looked up by binary search, which assumes they were added
694    /// to the tree in non-decreasing order of their spans.
695    ///
696    /// # Examples
697    ///
698    /// ```
699    /// use syntree::Span;
700    ///
701    /// let tree = syntree::tree! {
702    ///     "root" => {
703    ///         "child1" => {
704    ///             ("token1", 3)
705    ///         },
706    ///         "child2" => {
707    ///             "nested1" => {
708    ///                 ("token2", 4),
709    ///             },
710    ///             ("token3", 1),
711    ///         },
712    ///         "child3" => {
713    ///             ("token4", 5)
714    ///         }
715    ///     },
716    ///     "root2" => {}
717    /// };
718    ///
719    /// let node = tree.node_with_span(Span::point(0)).ok_or("missing 0")?;
720    /// assert_eq!(node.value(), "token1");
721    ///
722    /// let node = tree.node_with_span(Span::new(0, 3)).ok_or("missing 0..3")?;
723    /// assert_eq!(node.value(), "token1");
724    ///
725    /// let node = tree.node_with_span(Span::point(3)).ok_or("missing 3")?;
726    /// assert_eq!(node.value(), "token2");
727    ///
728    /// let node = tree.node_with_span(Span::new(3, 7)).ok_or("missing 3..7")?;
729    /// assert_eq!(node.value(), "token2");
730    ///
731    /// let node = tree.node_with_span(Span::new(7, 8)).ok_or("missing 7..8")?;
732    /// assert_eq!(node.value(), "token3");
733    ///
734    /// let node = tree.node_with_span(Span::new(3, 8)).ok_or("missing 3..8")?;
735    /// assert_eq!(node.value(), "child2");
736    ///
737    /// let node = tree.node_with_span(Span::new(8, 13)).ok_or("missing 8..13")?;
738    /// assert_eq!(node.value(), "token4");
739    ///
740    /// let node = tree.node_with_span(Span::new(2, 4)).ok_or("missing 2..4")?;
741    /// assert_eq!(node.value(), "root");
742    ///
743    /// assert!(tree.node_with_span(Span::point(13)).is_none());
744    /// assert!(tree.node_with_span(Span::new(4, 2)).is_none());
745    /// # Ok::<_, Box<dyn core::error::Error>>(())
746    /// ```
747    ///
748    /// Range queries work as expected with checkpoints:
749    ///
750    /// ```
751    /// use syntree::{Span, Builder};
752    ///
753    /// let mut tree = Builder::new();
754    ///
755    /// let c = tree.checkpoint()?;
756    ///
757    /// tree.open("child1")?;
758    /// tree.token("lit1", 3)?;
759    /// tree.close()?;
760    ///
761    /// tree.open("child2")?;
762    /// tree.token("lit2", 2)?;
763    /// tree.close()?;
764    ///
765    /// tree.close_at(c, "root")?;
766    ///
767    /// let tree = tree.build()?;
768    ///
769    /// let node = tree.node_with_span(Span::point(0)).ok_or("missing at point 0")?;
770    /// assert_eq!(node.value(), "lit1");
771    ///
772    /// let node = tree.node_with_span(Span::new(0, 3)).ok_or("missing at 0..3")?;
773    /// assert_eq!(node.value(), "lit1");
774    /// assert_eq!(node.parent().map(|n| n.value()), Some("child1"));
775    ///
776    /// let node = tree.node_with_span(Span::point(3)).ok_or("missing at point 3")?;
777    /// assert_eq!(node.value(), "lit2");
778    ///
779    /// let node = tree.node_with_span(Span::new(3, 5)).ok_or("missing at 3..5")?;
780    /// assert_eq!(node.value(), "lit2");
781    ///
782    /// let node = tree.node_with_span(Span::new(2, 5)).ok_or("missing at 2..5")?;
783    /// assert_eq!(node.value(), "root");
784    /// # Ok::<_, Box<dyn core::error::Error>>(())
785    /// ```
786    #[must_use]
787    pub fn node_with_span(&self, span: Span<F::Index>) -> Option<Element<'_, T, F>> {
788        self.node_with_span_internal(span.start, span.end)
789    }
790
791    fn node_with_span_internal(&self, start: F::Index, end: F::Index) -> Option<Element<'_, T, F>> {
792        if start > end {
793            return None;
794        }
795
796        // Index of the first entry which ends after `start`. Empty tokens
797        // never qualify, since the entry before them ends where they do.
798        let n = self.tokens.partition_point(|t| t.end <= start);
799
800        let mut next = n;
801
802        // `start` falls in a gap left by an explicit span, so the token after
803        // it starts after `start`.
804        let in_gap = match self.tokens.get(n) {
805            Some(links) if links.is_gap() => {
806                next = n.checked_add(1)?;
807                true
808            }
809            Some(..) => false,
810            None => true,
811        };
812
813        let found = F::Pointer::new(next)
814            .and_then(|id| self.element(Id::Token(id)))
815            .and_then(|e| smallest_containing(e, start, end));
816
817        if !in_gap {
818            return found;
819        }
820
821        // A node spanning the gap can instead contain the closest preceding
822        // token with a non-empty span.
823        let mut index = n;
824
825        let preceding = loop {
826            let Some(i) = index.checked_sub(1) else {
827                break None;
828            };
829
830            index = i;
831
832            let Some(links) = self.tokens.get(i) else {
833                break None;
834            };
835
836            if links.is_gap() || links.end <= self.token_start(i) {
837                continue;
838            }
839
840            break F::Pointer::new(i)
841                .and_then(|id| self.element(Id::Token(id)))
842                .and_then(|e| smallest_containing(e, start, end));
843        };
844
845        match (found, preceding) {
846            (Some(a), Some(b))
847                if a.span().start <= b.span().start && b.span().end <= a.span().end =>
848            {
849                Some(b)
850            }
851            (None, b) => b,
852            (a, _) => a,
853        }
854    }
855}
856
857/// Climb from `element` to the first element which contains `start..end`,
858/// where `start` must lie strictly before the end of the element.
859fn smallest_containing<'a, T, F>(
860    element: Element<'a, T, F>,
861    start: F::Index,
862    end: F::Index,
863) -> Option<Element<'a, T, F>>
864where
865    T: Copy,
866    F: Flavor,
867{
868    let mut element = element;
869
870    loop {
871        let span = element.span();
872
873        if span.start <= start && start < span.end && end <= span.end {
874            return Some(element);
875        }
876
877        element = NodeOrToken::Node(element.parent()?);
878    }
879}

920        where

939            .walk()

943    }
