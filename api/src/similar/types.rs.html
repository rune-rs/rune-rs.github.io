15/// An enum representing a diffing algorithm.
16#[derive(Clone, Copy, Hash, PartialEq, Eq, PartialOrd, Ord, Debug)]
17#[cfg_attr(
18    feature = "serde",
19    derive(serde::Serialize, serde::Deserialize),
20    serde(rename_all = "snake_case")
21)]
22#[non_exhaustive]
23pub enum Algorithm {
24    /// Picks the practical, heuristic Myers algorithm from
25    /// [`crate::algorithms::myers`].
26    ///
27    /// This is the default. It can trade a minimal edit script for bounded work
28    /// on difficult inputs. Use [`Algorithm::RawMyers`] when a
29    /// shortest-edit-script search without these heuristics is required.
30    Myers,
31    /// Picks the raw Myers algorithm from [`crate::algorithms::myers`].
32    ///
33    /// Unlike [`Algorithm::Myers`], this does not use heuristics that can change
34    /// the resulting edit script. Without a deadline it computes a shortest
35    /// edit script, but pathological inputs can take a very long time.
36    RawMyers,
37    /// Picks the patience algorithm from [`crate::algorithms::patience`]
38    Patience,
39    /// Picks the classic LCS table algorithm from [`crate::algorithms::lcs`]
40    Lcs,
41    /// Picks the Hunt-style algorithm from [`crate::algorithms::hunt`]
42    Hunt,
43    /// Picks the Histogram algorithm from [`crate::algorithms::histogram`]
44    Histogram,
45}
46
47impl Default for Algorithm {
48    /// Returns the default algorithm ([`Algorithm::Myers`]).
49    fn default() -> Algorithm {
50        Algorithm::Myers
51    }
52}

54/// The tag of a change.
55#[derive(Debug, PartialEq, Eq, Hash, Clone, Copy, Ord, PartialOrd)]
56#[cfg_attr(
57    feature = "serde",
58    derive(serde::Serialize, serde::Deserialize),
59    serde(rename_all = "snake_case")
60)]
61pub enum ChangeTag {
62    /// The change indicates equality (not a change)
63    Equal,
64    /// The change indicates deleted text.
65    Delete,
66    /// The change indicates inserted text.
67    Insert,
68}

70impl fmt::Display for ChangeTag {

72        write!(
73            f,
74            "{}",
75            match &self {

77                ChangeTag::Delete => '-',
78                ChangeTag::Insert => '+',
79            }
80        )
81    }
82}

84/// Represents the expanded [`DiffOp`] change.
85///
86/// This type is returned from [`DiffOp::iter_changes`] and
87/// [`TextDiff::iter_changes`](crate::text::TextDiff::iter_changes).
88///
89/// It exists so that it's more convenient to work with textual differences as
90/// the underlying [`DiffOp`] encodes a group of changes.
91///
92/// This type has additional methods that are only available for types
93/// implementing [`DiffableStr`](crate::text::DiffableStr).
94#[derive(Debug, PartialEq, Eq, Hash, Clone, Copy, Ord, PartialOrd)]
95#[cfg_attr(feature = "serde", derive(serde::Serialize))]
96pub struct Change<T> {
97    pub(crate) tag: ChangeTag,
98    pub(crate) old_index: Option<usize>,
99    pub(crate) new_index: Option<usize>,
100    pub(crate) value: T,
101}
102
103/// These methods are available for all change types.
104impl<T: Clone> Change<T> {
105    /// Returns the change tag.
106    pub fn tag(&self) -> ChangeTag {
107        self.tag
108    }
109
110    /// Returns the old index if available.
111    pub fn old_index(&self) -> Option<usize> {
112        self.old_index
113    }
114
115    /// Returns the new index if available.
116    pub fn new_index(&self) -> Option<usize> {
117        self.new_index
118    }
119
120    /// Returns the underlying changed value.
121    ///
122    /// Depending on the type of the underlying [`crate::text::DiffableStr`]
123    /// this value is more or less useful.  If you always want to have a utf-8
124    /// string it's best to use the [`Change::as_str`] and
125    /// [`Change::to_string_lossy`] methods.
126    pub fn value(&self) -> T {
127        self.value.clone()
128    }
129
130    /// Returns the underlying changed value as reference.
131    pub fn value_ref(&self) -> &T {
132        &self.value
133    }
134
135    /// Returns the underlying changed value as mutable reference.
136    pub fn value_mut(&mut self) -> &mut T {
137        &mut self.value
138    }
139}

141/// Utility enum to capture a diff operation.
142///
143/// This is used by [`Capture`](crate::algorithms::Capture).
144#[derive(Debug, PartialEq, Eq, Hash, Clone, Copy)]
145#[cfg_attr(
146    feature = "serde",
147    derive(serde::Serialize, serde::Deserialize),
148    serde(rename_all = "snake_case", tag = "op")
149)]
150pub enum DiffOp {
151    /// A segment is equal (see [`DiffHook::equal`])
152    Equal {
153        /// The starting index in the old sequence.
154        old_index: usize,
155        /// The starting index in the new sequence.
156        new_index: usize,
157        /// The length of the segment.
158        len: usize,
159    },
160    /// A segment was deleted (see [`DiffHook::delete`])
161    Delete {
162        /// The starting index in the old sequence.
163        old_index: usize,
164        /// The length of the old segment.
165        old_len: usize,
166        /// The starting index in the new sequence.
167        new_index: usize,
168    },
169    /// A segment was inserted (see [`DiffHook::insert`])
170    Insert {
171        /// The starting index in the old sequence.
172        old_index: usize,
173        /// The starting index in the new sequence.
174        new_index: usize,
175        /// The length of the new segment.
176        new_len: usize,
177    },
178    /// A segment was replaced (see [`DiffHook::replace`])
179    Replace {
180        /// The starting index in the old sequence.
181        old_index: usize,
182        /// The length of the old segment.
183        old_len: usize,
184        /// The starting index in the new sequence.
185        new_index: usize,
186        /// The length of the new segment.
187        new_len: usize,
188    },
189}

191/// The tag of a diff operation.
192#[derive(Debug, PartialEq, Eq, Hash, Clone, Copy, Ord, PartialOrd)]
193#[cfg_attr(
194    feature = "serde",
195    derive(serde::Serialize, serde::Deserialize),
196    serde(rename_all = "snake_case")
197)]
198pub enum DiffTag {
199    /// The diff op encodes an equal segment.
200    Equal,
201    /// The diff op encodes a deleted segment.
202    Delete,
203    /// The diff op encodes an inserted segment.
204    Insert,
205    /// The diff op encodes a replaced segment.
206    Replace,
207}
208
209impl DiffOp {
210    /// Returns the tag of the operation.
211    pub fn tag(self) -> DiffTag {
212        self.as_tag_tuple().0
213    }
214
215    /// Returns the old range.
216    pub fn old_range(&self) -> Range<usize> {
217        self.as_tag_tuple().1
218    }
219
220    /// Returns the new range.
221    pub fn new_range(&self) -> Range<usize> {
222        self.as_tag_tuple().2
223    }
224
225    /// Transform the op into a tuple of diff tag and ranges.
226    ///
227    /// This is useful when operating on slices.  The returned format is
228    /// `(tag, i1..i2, j1..j2)`:
229    ///
230    /// * `Replace`: `a[i1..i2]` should be replaced by `b[j1..j2]`
231    /// * `Delete`: `a[i1..i2]` should be deleted (`j1 == j2` in this case).
232    /// * `Insert`: `b[j1..j2]` should be inserted at `a[i1..i2]` (`i1 == i2` in this case).
233    /// * `Equal`: `a[i1..i2]` is equal to `b[j1..j2]`.
234    pub fn as_tag_tuple(&self) -> (DiffTag, Range<usize>, Range<usize>) {
235        match *self {
236            DiffOp::Equal {
237                old_index,
238                new_index,
239                len,
240            } => (
241                DiffTag::Equal,
242                old_index..old_index + len,
243                new_index..new_index + len,
244            ),
245            DiffOp::Delete {
246                old_index,
247                new_index,
248                old_len,
249            } => (
250                DiffTag::Delete,
251                old_index..old_index + old_len,
252                new_index..new_index,
253            ),
254            DiffOp::Insert {
255                old_index,
256                new_index,
257                new_len,
258            } => (
259                DiffTag::Insert,
260                old_index..old_index,
261                new_index..new_index + new_len,
262            ),
263            DiffOp::Replace {
264                old_index,
265                old_len,
266                new_index,
267                new_len,
268            } => (
269                DiffTag::Replace,
270                old_index..old_index + old_len,
271                new_index..new_index + new_len,
272            ),
273        }
274    }
275
276    /// Apply this operation to a diff hook.
277    pub fn apply_to_hook<D: DiffHook>(&self, d: &mut D) -> Result<(), D::Error> {
278        match *self {
279            DiffOp::Equal {
280                old_index,
281                new_index,
282                len,
283            } => d.equal(old_index, new_index, len),
284            DiffOp::Delete {
285                old_index,
286                old_len,
287                new_index,
288            } => d.delete(old_index, old_len, new_index),
289            DiffOp::Insert {
290                old_index,
291                new_index,
292                new_len,
293            } => d.insert(old_index, new_index, new_len),
294            DiffOp::Replace {
295                old_index,
296                old_len,
297                new_index,
298                new_len,
299            } => d.replace(old_index, old_len, new_index, new_len),
300        }
301    }

303    /// Iterates over all changes encoded in the diff op against old and new
304    /// sequences.
305    ///
306    /// `old` and `new` are two indexable objects like the types you pass to
307    /// the diffing algorithm functions.
308    ///
309    /// ```rust
310    /// use similar::{ChangeTag, Algorithm};
311    /// use similar::capture_diff_slices;
312    /// let old = vec!["foo", "bar", "baz"];
313    /// let new = vec!["foo", "bar", "blah"];
314    /// let ops = capture_diff_slices(Algorithm::Myers, &old, &new);
315    /// let changes: Vec<_> = ops
316    ///     .iter()
317    ///     .flat_map(|x| x.iter_changes(&old, &new))
318    ///     .map(|x| (x.tag(), x.value()))
319    ///     .collect();
320    /// assert_eq!(changes, vec![
321    ///     (ChangeTag::Equal, "foo"),
322    ///     (ChangeTag::Equal, "bar"),
323    ///     (ChangeTag::Delete, "baz"),
324    ///     (ChangeTag::Insert, "blah"),
325    /// ]);
326    /// ```
327    pub fn iter_changes<'lookup, Old, New, T>(
328        &self,
329        old: &'lookup Old,
330        new: &'lookup New,
331    ) -> ChangesIter<'lookup, Old, New, T>
332    where
333        Old: Index<usize, Output = T> + ?Sized,
334        New: Index<usize, Output = T> + ?Sized,
335    {
336        ChangesIter::new(old, new, *self)
337    }

339    /// Given a diffop yields the changes it encodes against the given slices.
340    ///
341    /// This is similar to [`DiffOp::iter_changes`] but instead of yielding the
342    /// individual changes it yields consequitive changed slices.
343    ///
344    /// This will only ever yield a single tuple or two tuples in case a
345    /// [`DiffOp::Replace`] operation is passed.
346    ///
347    /// ```rust
348    /// use similar::{ChangeTag, Algorithm};
349    /// use similar::capture_diff_slices;
350    /// let old = vec!["foo", "bar", "baz"];
351    /// let new = vec!["foo", "bar", "blah"];
352    /// let ops = capture_diff_slices(Algorithm::Myers, &old, &new);
353    /// let changes: Vec<_> = ops.iter().flat_map(|x| x.iter_slices(&old, &new)).collect();
354    /// assert_eq!(changes, vec![
355    ///     (ChangeTag::Equal, &["foo", "bar"][..]),
356    ///     (ChangeTag::Delete, &["baz"][..]),
357    ///     (ChangeTag::Insert, &["blah"][..]),
358    /// ]);
359    /// ```
360    ///
361    /// Due to lifetime restrictions it's currently impossible for the
362    /// returned slices to outlive the lookup.
363    pub fn iter_slices<'lookup, Old, New, T>(
364        &self,
365        old: &'lookup Old,
366        new: &'lookup New,
367    ) -> impl Iterator<Item = (ChangeTag, &'lookup T)> + use<'lookup, Old, New, T>
368    where
369        T: 'lookup + ?Sized,
370        Old: Index<Range<usize>, Output = T> + ?Sized,
371        New: Index<Range<usize>, Output = T> + ?Sized,
372    {
373        match *self {
374            DiffOp::Equal { old_index, len, .. } => {
375                Some((ChangeTag::Equal, &old[old_index..old_index + len]))
376                    .into_iter()
377                    .chain(None)
378            }
379            DiffOp::Insert {
380                new_index, new_len, ..
381            } => Some((ChangeTag::Insert, &new[new_index..new_index + new_len]))
382                .into_iter()
383                .chain(None),
384            DiffOp::Delete {
385                old_index, old_len, ..
386            } => Some((ChangeTag::Delete, &old[old_index..old_index + old_len]))
387                .into_iter()
388                .chain(None),
389            DiffOp::Replace {
390                old_index,
391                old_len,
392                new_index,
393                new_len,
394            } => Some((ChangeTag::Delete, &old[old_index..old_index + old_len]))
395                .into_iter()
396                .chain(Some((
397                    ChangeTag::Insert,
398                    &new[new_index..new_index + new_len],
399                ))),
400        }
401    }

403    pub(crate) fn is_empty(&self) -> bool {
404        let (_, old, new) = self.as_tag_tuple();
405        is_empty_range(&old) && is_empty_range(&new)
406    }

432    fn adjust(&mut self, adjust_offset: (usize, bool), adjust_len: (usize, bool)) {
433        #[inline(always)]
434        fn modify(val: &mut usize, adj: (usize, bool)) {
435            if adj.1 {
436                *val -= adj.0;
437            } else {
438                *val += adj.0;
439            }
440        }
441
442        match self {
443            DiffOp::Equal {
444                old_index,
445                new_index,
446                len,
447            } => {
448                modify(old_index, adjust_offset);
449                modify(new_index, adjust_offset);
450                modify(len, adjust_len);
451            }
452            DiffOp::Delete {
453                old_index,
454                old_len,
455                new_index,
456            } => {
457                modify(old_index, adjust_offset);
458                modify(old_len, adjust_len);
459                modify(new_index, adjust_offset);
460            }
461            DiffOp::Insert {
462                old_index,
463                new_index,
464                new_len,
465            } => {
466                modify(old_index, adjust_offset);
467                modify(new_index, adjust_offset);
468                modify(new_len, adjust_len);
469            }
470            DiffOp::Replace {
471                old_index,
472                old_len,
473                new_index,
474                new_len,
475            } => {
476                modify(old_index, adjust_offset);
477                modify(old_len, adjust_len);
478                modify(new_index, adjust_offset);
479                modify(new_len, adjust_len);
480            }
481        }
482    }

491    /// The text interface can produce changes over [`DiffableStr`] implementing
492    /// values.  As those are generic interfaces for different types of strings
493    /// utility methods to make working with standard rust strings more enjoyable.
494    impl<'s, T: DiffableStr + ?Sized> Change<&'s T> {
495        /// Returns the value as string if it is utf-8.
496        pub fn as_str(&self) -> Option<&'s str> {
497            T::as_str(self.value)
498        }
499
500        /// Returns the value (lossy) decoded as utf-8 string.
501        pub fn to_string_lossy(&self) -> Cow<'s, str> {
502            T::to_string_lossy(self.value)
503        }
504
505        /// Returns `true` if this change does not end in a newline and must be
506        /// followed up by one if line based diffs are used.
507        ///
508        /// The [`std::fmt::Display`] implementation of [`Change`] will automatically
509        /// insert a newline after the value if this is true.
510        pub fn missing_newline(&self) -> bool {
511            !T::ends_with_newline(self.value)
512        }
513    }
514
515    impl<T: DiffableStr + ?Sized> fmt::Display for Change<&T> {
516        fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
517            write!(
518                f,
519                "{}{}",
520                self.to_string_lossy(),
521                if self.missing_newline() { "\n" } else { "" }
522            )
523        }
524    }
