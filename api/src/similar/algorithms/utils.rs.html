10/// A small bucket that stores its first value inline.
11///
12/// Most hash buckets in diff inputs contain a single value.  Using a `Vec` for
13/// every bucket adds one heap allocation per distinct input item, so the common
14/// one-value case lives directly in the map instead.
15pub(crate) struct HashBucket<T> {
16    first: T,
17    rest: Vec<T>,
18}
19
20impl<T> HashBucket<T> {
21    pub(crate) fn new(first: T) -> Self {
22        Self {
23            first,
24            rest: Vec::new(),
25        }
26    }
27
28    pub(crate) fn push(&mut self, value: T) {
29        self.rest.push(value);
30    }
31
32    pub(crate) fn len(&self) -> usize {
33        self.rest.len() + 1
34    }
35
36    pub(crate) fn iter(&self) -> impl DoubleEndedIterator<Item = &T> {
37        core::iter::once(&self.first).chain(self.rest.iter())
38    }
39
40    pub(crate) fn iter_mut(&mut self) -> impl Iterator<Item = &mut T> {
41        core::iter::once(&mut self.first).chain(self.rest.iter_mut())
42    }
43
44    pub(crate) fn into_iter(self) -> impl Iterator<Item = T> {
45        core::iter::once(self.first).chain(self.rest)
46    }
47}

56/// Multiplies two words and folds the 128-bit product back into 64 bits.
57///
58/// Both halves of the product depend on all input bits, so folding them
59/// together produces a result whose low and high bits are well mixed.  This
60/// is only suitable for mixing keys that are already hashes or dense ids;
61/// content hashing uses SipHash below.
62#[cfg(any(feature = "std", feature = "hashbrown", test))]
63#[inline(always)]
64pub(crate) fn folded_multiply(a: u64, b: u64) -> u64 {
65    let full = (a as u128).wrapping_mul(b as u128);
66    (full as u64) ^ ((full >> 64) as u64)
67}

74/// Returns the SipHash key used for content hashing.
75///
76/// Diff inputs may be untrusted, and the hash buckets in [`IdentifyDistinct`]
77/// and friends degrade to linear scans on collisions.  A per-process random
78/// key makes it impossible to craft colliding inputs ahead of time.  Without
79/// `std` there is no entropy source, so a fixed key is used; SipHash has no
80/// known practical collision families even with a public key, which is
81/// still far better than the trivially invertible multiplicative hashes.
82#[inline]
83fn content_hash_keys() -> (u64, u64) {
84    #[cfg(feature = "std")]
85    {
86        use std::hash::{BuildHasher, RandomState};
87        use std::sync::OnceLock;
88
89        static KEYS: OnceLock<(u64, u64)> = OnceLock::new();
90        *KEYS.get_or_init(|| {
91            let state = RandomState::new();
92            (state.hash_one(0u64), state.hash_one(1u64))
93        })
94    }
95    #[cfg(not(feature = "std"))]
96    {
97        (0x243F_6A88_85A3_08D3, 0x1319_8A2E_0370_7344)
98    }
99}

101/// SipHash-1-3 for deduplicating diff input items.
102///
103/// Diff inputs are hashed once per item and only compared by hash within a
104/// single invocation, but the inputs may be untrusted and the buckets keyed
105/// by these hashes degrade to linear scans on collisions.  Ad-hoc
106/// multiplicative mixers are cheap but admit collision families that can be
107/// solved for algebraically; SipHash is the vetted choice for exactly this
108/// situation (it is what `std`'s `HashMap` uses) and is still several times
109/// faster than the previous byte-at-a-time FNV loop because it consumes
110/// eight bytes per round.
111#[derive(Clone, Copy)]
112pub(crate) struct ContentHasher {
113    v0: u64,
114    v1: u64,
115    v2: u64,
116    v3: u64,
117    /// Unprocessed trailing bytes, little endian.
118    tail: u64,
119    /// Number of valid bytes in `tail`.
120    ntail: usize,
121    length: usize,
122}
123
124impl ContentHasher {
125    #[inline]
126    fn new(k0: u64, k1: u64) -> Self {
127        ContentHasher {
128            v0: k0 ^ 0x736f_6d65_7073_6575,
129            v1: k1 ^ 0x646f_7261_6e64_6f6d,
130            v2: k0 ^ 0x6c79_6765_6e65_7261,
131            v3: k1 ^ 0x7465_6462_7974_6573,
132            tail: 0,
133            ntail: 0,
134            length: 0,
135        }
136    }
137
138    #[inline(always)]
139    fn sip_round(&mut self) {
140        self.v0 = self.v0.wrapping_add(self.v1);
141        self.v1 = self.v1.rotate_left(13);
142        self.v1 ^= self.v0;
143        self.v0 = self.v0.rotate_left(32);
144        self.v2 = self.v2.wrapping_add(self.v3);
145        self.v3 = self.v3.rotate_left(16);
146        self.v3 ^= self.v2;
147        self.v0 = self.v0.wrapping_add(self.v3);
148        self.v3 = self.v3.rotate_left(21);
149        self.v3 ^= self.v0;
150        self.v2 = self.v2.wrapping_add(self.v1);
151        self.v1 = self.v1.rotate_left(17);
152        self.v1 ^= self.v2;
153        self.v2 = self.v2.rotate_left(32);
154    }
155
156    #[inline(always)]
157    fn compress(&mut self, word: u64) {
158        self.v3 ^= word;
159        self.sip_round();
160        self.v0 ^= word;
161    }
162}

164impl Default for ContentHasher {
165    #[inline]
166    fn default() -> Self {
167        let (k0, k1) = content_hash_keys();
168        ContentHasher::new(k0, k1)
169    }
170}

172impl Hasher for ContentHasher {
173    #[inline]
174    fn finish(&self) -> u64 {
175        let mut state = *self;
176        let last = ((self.length as u64 & 0xff) << 56) | self.tail;
177        state.compress(last);
178        state.v2 ^= 0xff;
179        state.sip_round();
180        state.sip_round();
181        state.sip_round();
182        state.v0 ^ state.v1 ^ state.v2 ^ state.v3
183    }
184
185    #[inline]
186    fn write(&mut self, bytes: &[u8]) {
187        self.length = self.length.wrapping_add(bytes.len());
188        let len = bytes.len();
189
190        // The common case is a single write of a whole line or word into a
191        // fresh hasher.  Keep that path free of per-byte loops: full words
192        // are consumed directly and the remainder is taken from an
193        // overlapping read of the last word.
194        if self.ntail == 0 {
195            let mut pos = 0;
196            while pos + 8 <= len {
197                self.compress(read_u64(&bytes[pos..]));
198                pos += 8;
199            }
200            let remainder = len - pos;
201            if remainder != 0 {
202                self.tail = if len >= 8 {
203                    read_u64(&bytes[len - 8..]) >> (8 * (8 - remainder))
204                } else {
205                    let mut tail = 0u64;
206                    for (index, &byte) in bytes.iter().enumerate() {
207                        tail |= (byte as u64) << (8 * index);
208                    }
209                    tail
210                };
211                self.ntail = remainder;
212            }
213            return;
214        }
215
216        // Complete a partially filled word first.
217        let mut rest = bytes;
218        let take = (8 - self.ntail).min(rest.len());
219        for &byte in &rest[..take] {
220            self.tail |= (byte as u64) << (8 * self.ntail);
221            self.ntail += 1;
222        }
223        rest = &rest[take..];
224        if self.ntail < 8 {
225            return;
226        }
227        self.compress(self.tail);
228        self.tail = 0;
229        self.ntail = 0;
230
231        let mut chunks = rest.chunks_exact(8);
232        for chunk in &mut chunks {
233            self.compress(u64::from_le_bytes(chunk.try_into().unwrap()));
234        }
235
236        let remainder = chunks.remainder();
237        let mut tail = 0u64;
238        for (index, &byte) in remainder.iter().enumerate() {
239            tail |= (byte as u64) << (8 * index);
240        }
241        self.tail = tail;
242        self.ntail = remainder.len();
243    }
244
245    #[inline]
246    fn write_u8(&mut self, i: u8) {
247        // `str` hashing appends a single terminator byte after the content.
248        self.length = self.length.wrapping_add(1);
249        self.tail |= (i as u64) << (8 * self.ntail);
250        self.ntail += 1;
251        if self.ntail == 8 {
252            self.compress(self.tail);
253            self.tail = 0;
254            self.ntail = 0;
255        }
256    }
257
258    #[inline]
259    fn write_u16(&mut self, i: u16) {
260        self.write(&i.to_le_bytes());
261    }
262
263    #[inline]
264    fn write_u32(&mut self, i: u32) {
265        if self.ntail == 0 {
266            self.length = self.length.wrapping_add(4);
267            self.tail = i as u64;
268            self.ntail = 4;
269        } else {
270            self.write(&i.to_le_bytes());
271        }
272    }
273
274    #[inline]
275    fn write_u64(&mut self, i: u64) {
276        if self.ntail == 0 {
277            self.length = self.length.wrapping_add(8);
278            self.compress(i);
279        } else {
280            self.write(&i.to_le_bytes());
281        }
282    }
283
284    #[inline]
285    fn write_u128(&mut self, i: u128) {
286        self.write(&i.to_le_bytes());
287    }
288
289    #[inline]
290    fn write_usize(&mut self, i: usize) {
291        self.write(&i.to_le_bytes());
292    }
293}

295/// Hashes a value into a `u64` that is stable for the duration of a diff.
296///
297/// The result is consistent within a process (and with `std`, randomized
298/// per process); it is only used to bucket equal items within a single diff
299/// operation and never influences the produced diff.
300#[inline(always)]
301pub(crate) fn stable_hash<T: Hash + ?Sized>(value: &T) -> u64 {
302    let mut hasher = ContentHasher::default();
303    value.hash(&mut hasher);
304    hasher.finish()
305}

375/// Returns only unique items in the sequence as vector.
376///
377/// Each item is wrapped in a [`UniqueItem`] so that both the value and the
378/// index can be extracted.
379pub fn unique<Idx>(lookup: &Idx, range: Range<usize>) -> Vec<UniqueItem<'_, Idx>>
380where
381    Idx: Index<usize> + ?Sized,
382    Idx::Output: Hash + Eq,
383{
384    // We key buckets by hash to stay compatible with both HashMap and
385    // BTreeMap backends without requiring `Idx::Output: Ord`.
386    let mut by_hash =
387        int_key_map_with_capacity::<u64, HashBucket<(usize, Option<usize>)>>(range.len());
388    for index in range {
389        let hash = stable_hash(&lookup[index]);
390        let Some(bucket) = by_hash.get_mut(&hash) else {
391            by_hash.insert(hash, HashBucket::new((index, Some(index))));
392            continue;
393        };
394
395        let mut found = false;
396        for (representative, unique_index) in bucket.iter_mut() {
397            if lookup[index] == lookup[*representative] {
398                if unique_index.is_some() {
399                    *unique_index = None;
400                }
401                found = true;
402                break;
403            }
404        }
405
406        if !found {
407            bucket.push((index, Some(index)));
408        }
409    }
410
411    let mut rv = by_hash
412        .into_values()
413        .flat_map(|bucket| bucket.into_iter())
414        .filter_map(|(_, unique_index)| unique_index)
415        .map(|index| UniqueItem { lookup, index })
416        .collect::<Vec<_>>();
417    rv.sort_by_key(|a| a.original_index());
418    rv
419}

421/// Given two lookups and ranges calculates the length of the common prefix.
422pub fn common_prefix_len<Old, New>(
423    old: &Old,
424    old_range: Range<usize>,
425    new: &New,
426    new_range: Range<usize>,
427) -> usize
428where
429    Old: Index<usize> + ?Sized,
430    New: Index<usize> + ?Sized,
431    New::Output: PartialEq<Old::Output>,
432{
433    if is_empty_range(&old_range) || is_empty_range(&new_range) {
434        return 0;
435    }
436    new_range
437        .zip(old_range)
438        .take_while(
439            #[inline(always)]
440            |x| new[x.0] == old[x.1],
441        )
442        .count()
443}

445/// Given two lookups and ranges calculates the length of common suffix.
446pub fn common_suffix_len<Old, New>(
447    old: &Old,
448    old_range: Range<usize>,
449    new: &New,
450    new_range: Range<usize>,
451) -> usize
452where
453    Old: Index<usize> + ?Sized,
454    New: Index<usize> + ?Sized,
455    New::Output: PartialEq<Old::Output>,
456{
457    if is_empty_range(&old_range) || is_empty_range(&new_range) {
458        return 0;
459    }
460    new_range
461        .rev()
462        .zip(old_range.rev())
463        .take_while(
464            #[inline(always)]
465            |x| new[x.0] == old[x.1],
466        )
467        .count()
468}

484/// A utility struct to convert distinct items to unique integers.
485///
486/// This can be helpful on larger inputs to speed up the comparisons
487/// performed by doing a first pass where the data set gets reduced
488/// to (small) integers. For lazily computed or projected values, see
489/// [`CachedLookup`](super::CachedLookup).
490///
491/// The idea is that instead of passing two sequences to a diffling algorithm
492/// you first pass it via [`IdentifyDistinct`]:
493///
494/// ```rust
495/// use similar::capture_diff;
496/// use similar::algorithms::{Algorithm, IdentifyDistinct};
497///
498/// let old = &["foo", "bar", "baz"][..];
499/// let new = &["foo", "blah", "baz"][..];
500/// let h = IdentifyDistinct::<u32>::new(old, 0..old.len(), new, 0..new.len());
501/// let ops = capture_diff(
502///     Algorithm::Myers,
503///     h.old_lookup(),
504///     h.old_range(),
505///     h.new_lookup(),
506///     h.new_range(),
507/// );
508/// ```
509///
510/// The indexes are the same as with the passed source ranges.
511pub struct IdentifyDistinct<Int> {
512    old: OffsetLookup<Int>,
513    new: OffsetLookup<Int>,
514    old_distinct: usize,
515}
516
517impl<Int> IdentifyDistinct<Int>
518where
519    Int: Add<Output = Int> + From<u8> + Default + Copy,
520{
521    /// Creates an int hasher for two sequences.
522    pub fn new<Old, New>(
523        old: &Old,
524        old_range: Range<usize>,
525        new: &New,
526        new_range: Range<usize>,
527    ) -> Self
528    where
529        Old: Index<usize> + ?Sized,
530        Old::Output: Eq + Hash,
531        New: Index<usize> + ?Sized,
532        New::Output: Eq + Hash + PartialEq<Old::Output>,
533    {
534        Self::build(old, old_range, new, new_range, false)
535    }
536
537    /// Like [`new`](Self::new), but every value that only occurs in `new`
538    /// shares the single identifier [`old_distinct_count`](Self::old_distinct_count).
539    ///
540    /// Algorithms that only ever compare old items against new items (Myers,
541    /// Hunt, Histogram) never need to tell two new-only values apart, and
542    /// skipping them keeps the map proportional to the old side.  This is
543    /// not suitable for Patience, which needs uniqueness within `new`.
544    pub(crate) fn new_matching_old<Old, New>(
545        old: &Old,
546        old_range: Range<usize>,
547        new: &New,
548        new_range: Range<usize>,
549    ) -> Self
550    where
551        Old: Index<usize> + ?Sized,
552        Old::Output: Eq + Hash,
553        New: Index<usize> + ?Sized,
554        New::Output: Eq + Hash + PartialEq<Old::Output>,
555    {
556        Self::build(old, old_range, new, new_range, true)
557    }
558
559    fn build<Old, New>(
560        old: &Old,
561        old_range: Range<usize>,
562        new: &New,
563        new_range: Range<usize>,
564        collapse_new_only: bool,
565    ) -> Self
566    where
567        Old: Index<usize> + ?Sized,
568        Old::Output: Eq + Hash,
569        New: Index<usize> + ?Sized,
570        New::Output: Eq + Hash + PartialEq<Old::Output>,
571    {
572        #[derive(Clone, Copy)]
573        enum Representative {
574            Old(usize),
575            New(usize),
576        }
577
578        let mut map =
579            int_key_map_with_capacity::<u64, HashBucket<(Representative, Int)>>(old_range.len());
580        let mut old_seq = Vec::with_capacity(old_range.len());
581        let mut new_seq = Vec::with_capacity(new_range.len());
582        let mut next_id = Int::default();
583        let mut old_distinct = 0usize;
584        let step = Int::from(1);
585        let old_start = old_range.start;
586        let new_start = new_range.start;
587
588        for idx in old_range {
589            let hash = stable_hash(&old[idx]);
590            let id = if let Some(bucket) = map.get_mut(&hash) {
591                if let Some((_, id)) = bucket.iter().find(
592                    |(rep, _)| matches!(rep, Representative::Old(rep_idx) if old[idx] == old[*rep_idx]),
593                ) {
594                    *id
595                } else {
596                    let id = next_id;
597                    next_id = next_id + step;
598                    old_distinct += 1;
599                    bucket.push((Representative::Old(idx), id));
600                    id
601                }
602            } else {
603                let id = next_id;
604                next_id = next_id + step;
605                old_distinct += 1;
606                map.insert(hash, HashBucket::new((Representative::Old(idx), id)));
607                id
608            };
609            old_seq.push(id);
610        }
611
612        let new_only_id = next_id;
613        if collapse_new_only {
614            next_id = next_id + step;
615        }
616
617        for idx in new_range {
618            let hash = stable_hash(&new[idx]);
619            let found = map.get_mut(&hash).map(|bucket| {
620                let existing = bucket
621                    .iter()
622                    .find(|(rep, _)| match rep {
623                        Representative::Old(rep_idx) => new[idx] == old[*rep_idx],
624                        Representative::New(rep_idx) => new[idx] == new[*rep_idx],
625                    })
626                    .map(|(_, id)| *id);
627                (bucket, existing)
628            });
629            let id = match found {
630                Some((_, Some(id))) => id,
631                _ if collapse_new_only => new_only_id,
632                Some((bucket, None)) => {
633                    let id = next_id;
634                    next_id = next_id + step;
635                    bucket.push((Representative::New(idx), id));
636                    id
637                }
638                None => {
639                    let id = next_id;
640                    next_id = next_id + step;
641                    map.insert(hash, HashBucket::new((Representative::New(idx), id)));
642                    id
643                }
644            };
645            new_seq.push(id);
646        }
647
648        IdentifyDistinct {
649            old: OffsetLookup {
650                offset: old_start,
651                vec: old_seq,
652            },
653            new: OffsetLookup {
654                offset: new_start,
655                vec: new_seq,
656            },
657            old_distinct,
658        }
659    }
660
661    /// Returns the number of distinct values that occur in the old lookup.
662    ///
663    /// Old values are numbered first, so exactly the identifiers below this
664    /// count occur in the old lookup; values that only occur in the new
665    /// lookup have larger identifiers.
666    pub(crate) fn old_distinct_count(&self) -> usize {
667        self.old_distinct
668    }
669
670    /// Returns a lookup for the old side.
671    pub fn old_lookup(&self) -> &impl Index<usize, Output = Int> {
672        &self.old
673    }
674
675    /// Returns a lookup for the new side.
676    pub fn new_lookup(&self) -> &impl Index<usize, Output = Int> {
677        &self.new
678    }
679
680    /// Convenience method to get back the old range.
681    pub fn old_range(&self) -> Range<usize> {
682        self.old.offset..self.old.offset + self.old.vec.len()
683    }
684
685    /// Convenience method to get back the new range.
686    pub fn new_range(&self) -> Range<usize> {
687        self.new.offset..self.new.offset + self.new.vec.len()
688    }
689}

696    let mut seed = 0x9E37_79B9u32;
697    for _ in 0..80 {
698        seed ^= seed << 13;
699        seed ^= seed >> 17;
700        seed ^= seed << 5;
701        data.push(seed as u8);
702    }
703
704    for len in 0..=64 {
705        let bytes = &data[..len];
706        let mut whole = ContentHasher::new(0, 0);
707        whole.write(bytes);
708        let expected = whole.finish();
709
710        // `DefaultHasher::new()` is SipHash-1-3 with a zero key.
711        #[cfg(feature = "std")]
712        {
713            let mut reference = std::hash::DefaultHasher::new();
714            reference.write(bytes);
715            assert_eq!(expected, reference.finish(), "len {len}");
716        }
717
718        // Splitting the input across writes, including through the
719        // integer shortcuts, must not change the result.
720        for split in 0..=len {
721            let mut pieces = ContentHasher::new(0, 0);
722            pieces.write(&bytes[..split]);
723            pieces.write(&bytes[split..]);
724            assert_eq!(pieces.finish(), expected, "len {len} split {split}");
725
726            let mut bytewise = ContentHasher::new(0, 0);
727            bytewise.write(&bytes[..split]);
728            for &byte in &bytes[split..] {
729                bytewise.write_u8(byte);
730            }
731            assert_eq!(bytewise.finish(), expected, "len {len} bytewise {split}");
732        }
733        if len >= 4 {
734            let mut prefixed = ContentHasher::new(0, 0);
735            prefixed.write_u32(u32::from_le_bytes(bytes[..4].try_into().unwrap()));
736            prefixed.write(&bytes[4..]);
737            assert_eq!(prefixed.finish(), expected, "len {len} u32 prefix");
738        }
739        if len >= 8 {
740            let mut prefixed = ContentHasher::new(0, 0);
741            prefixed.write_u64(read_u64(bytes));
742            prefixed.write(&bytes[8..]);
743            assert_eq!(prefixed.finish(), expected, "len {len} u64 prefix");
744        }
745    }
746}

748#[test]
749fn test_stable_hash_quality() {
750    use alloc::format;
751    use alloc::string::String;
752
753    // Typical diff inputs: dense integer ids and many short, similar strings.
754    // A weak hash would collide or cluster on these, which would make the
755    // hash buckets degenerate into linear scans.
756    let mut seen = crate::types::IntKeyMap::<u64, ()>::default();
757    for value in 0..100_000u32 {
758        assert!(seen.insert(stable_hash(&value), ()).is_none());
759    }
760
761    let mut seen = crate::types::IntKeyMap::<u64, ()>::default();
762    let mut ones_per_bit = [0usize; 64];
763    let mut samples = 0usize;
764    for index in 0..50_000u32 {
765        for text in [
766            format!("line {index}\n"),
767            format!("    return {index};\n"),
768            format!("{index}"),
769            String::from_utf8(alloc::vec![b'x'; (index % 40) as usize + 1]).unwrap() + "\n",
770        ] {
771            let hash = stable_hash(text.as_str());
772            // Repeated padding lines are equal across iterations; only
773            // distinct strings must produce distinct hashes.
774            if text.starts_with('x') && index >= 40 {
775                continue;
776            }
777            assert!(seen.insert(hash, ()).is_none(), "collision for {text:?}");
778            for (bit, count) in ones_per_bit.iter_mut().enumerate() {
779                *count += ((hash >> bit) & 1) as usize;
780            }
781            samples += 1;
782        }
783    }
784    // Every output bit should be set roughly half of the time; hashbrown
785    // relies on the top bits as tags and the low bits as bucket index.
786    for (bit, count) in ones_per_bit.iter().enumerate() {
787        let ratio = *count as f64 / samples as f64;
788        assert!(
789            (0.45..0.55).contains(&ratio),
790            "bit {bit} is set in {ratio} of hashes"
791        );
792    }
793}

945/// Regression test for the multiplicative mixer this crate used before
946/// switching to SipHash.
947///
948/// That mixer computed `folded_multiply((a ^ state) | HIGH, (b ^ K1) | HIGH)`
949/// plus linear feed-forward terms.  Choosing `b = K1` turns the second
950/// multiplicand into `1 << 63`, which makes the multiplication a shift and
951/// the whole block affine over GF(2).  Two such blocks give 128 bits of
952/// freedom for 64 bits of state, so colliding inputs can be produced with
953/// plain linear algebra.  The test rebuilds that family (checking that it
954/// does collide under the old mixer) and asserts the current hash keeps
955/// the values apart.
956#[test]
957fn test_stable_hash_resists_affine_block_family() {
958    use alloc::vec::Vec;
959
960    const SEED: u64 = 0x243F_6A88_85A3_08D3;
961    const K0: u64 = 0x9E37_79B9_7F4A_7C15;
962    const K1: u64 = 0xD6E8_FEB8_6659_FD93;
963    const K2: u64 = 0xA076_1D64_78BD_642F;
964    const HIGH: u64 = 1 << 63;
965
966    fn weak_mix(state: u64, a: u64, b: u64) -> u64 {
967        let x = (a ^ state) | HIGH;
968        let y = (b ^ K1) | HIGH;
969        folded_multiply(x, y) ^ a.rotate_left(29) ^ b.rotate_left(47) ^ state.rotate_left(11)
970    }
971
972    fn weak_write_u64(state: u64, i: u64) -> u64 {
973        folded_multiply((state ^ i) | HIGH, K1) ^ i.rotate_left(29) ^ state.rotate_left(11)
974    }
975
976    // State after hashing a `[u8; 32]` up to (excluding) the final empty
977    // tail mix and finalization: length prefix, length mixing, two blocks.
978    fn weak_two_blocks(a1: u64, a2: u64) -> u64 {
979        let state = weak_write_u64(SEED, 32) ^ 32u64.wrapping_mul(K0);
980        let state = weak_mix(state, a1, K1);
981        weak_mix(state, a2, K1)
982    }
983
984    fn weak_hash(a1: u64, a2: u64) -> u64 {
985        folded_multiply(weak_mix(weak_two_blocks(a1, a2), 0, 0), K2)
986    }
987
988    // `weak_two_blocks` is affine over GF(2) in the 128 input bits.  Extract
989    // the linear part column by column and compute its kernel with a xor
990    // basis; every kernel vector added to an input leaves the state as is.
991    let split = |v: u128| (v as u64, (v >> 64) as u64);
992    let constant = weak_two_blocks(0, 0);
993    let mut basis: [Option<(u64, u128)>; 64] = [None; 64];
994    let mut kernel: Vec<u128> = Vec::new();
995    for column in 0..128 {
996        let input = 1u128 << column;
997        let (a1, a2) = split(input);
998        let mut vector = weak_two_blocks(a1, a2) ^ constant;
999        let mut combination = input;
1000        for bit in (0..64).rev() {
1001            if (vector >> bit) & 1 == 0 {
1002                continue;
1003            }
1004            match basis[bit] {
1005                Some((basis_vector, basis_combination)) => {
1006                    vector ^= basis_vector;
1007                    combination ^= basis_combination;
1008                }
1009                None => {
1010                    basis[bit] = Some((vector, combination));
1011                    break;
1012                }
1013            }
1014        }
1015        if vector == 0 {
1016            kernel.push(combination);
1017        }
1018    }
1019    assert!(kernel.len() >= 12, "kernel dimension {}", kernel.len());
1020
1021    let n = 4_000usize;
1022    let inputs: Vec<Vec<u8>> = (1..=n)
1023        .map(|index| {
1024            let mut vector = 0u128;
1025            for (bit, kernel_vector) in kernel.iter().take(12).enumerate() {
1026                if (index >> bit) & 1 == 1 {
1027                    vector ^= kernel_vector;
1028                }
1029            }
1030            let (a1, a2) = split(vector);
1031            let mut bytes = Vec::with_capacity(32);
1032            bytes.extend_from_slice(&a1.to_le_bytes());
1033            bytes.extend_from_slice(&K1.to_le_bytes());
1034            bytes.extend_from_slice(&a2.to_le_bytes());
1035            bytes.extend_from_slice(&K1.to_le_bytes());
1036            bytes
1037        })
1038        .collect();
1039
1040    // Sanity check the construction: the family collides under the mixer
1041    // it was built against.
1042    let reference = weak_hash(0, 0);
1043    for input in &inputs {
1044        let a1 = u64::from_le_bytes(input[..8].try_into().unwrap());
1045        let a2 = u64::from_le_bytes(input[16..24].try_into().unwrap());
1046        assert_eq!(weak_hash(a1, a2), reference);
1047    }
1048
1049    assert_no_cheap_collisions(&inputs, "affine two-block family");
1050}
