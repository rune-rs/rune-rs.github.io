1//! Hunt-style diff algorithm.
2//!
3//! This implementation follows the classic Hunt family approach by building
4//! match lists for the new sequence and then tracking k-candidates with a
5//! threshold array and backpointers.
6//!
7//! In practice this is close to the Hunt–Szymanski variant (descending match
8//! list processing).
9//!
10//! * time: `O((N + R) log L)` where `R` is the number of matching pairs
11//! * space: `O(N + M + R)`
12//!
13//! # Heuristics
14//!
15//! See [`crate::algorithms`] for shared heuristics and the
16//! `diff_deadline_raw` API.
17

45#[derive(Clone, Copy)]
46struct Candidate {
47    old_index: usize,
48    new_index: usize,
49    // `usize::MAX` is the end-of-chain sentinel.  Avoiding `Option<usize>`
50    // keeps each candidate to three machine words instead of four.
51    prev: usize,
52}

54/// Hunt-style diff algorithm.
55///
56/// Diff `old`, between indices `old_range` and `new` between indices `new_range`.
57pub fn diff<Old, New, D>(
58    d: &mut D,
59    old: &Old,
60    old_range: Range<usize>,
61    new: &New,
62    new_range: Range<usize>,
63) -> Result<(), D::Error>
64where
65    Old: Index<usize> + ?Sized,
66    New: Index<usize> + ?Sized,
67    D: DiffHook,
68    Old::Output: Hash + Eq,
69    New::Output: PartialEq<Old::Output> + Hash + Eq,
70{
71    diff_deadline(d, old, old_range, new, new_range, None)
72}

74/// Hunt-style diff algorithm with deadline.
75///
76/// Diff `old`, between indices `old_range` and `new` between indices `new_range`.
77///
78/// This diff is done with an optional deadline that defines the maximal
79/// execution time permitted before it bails and falls back to Myers.
80pub fn diff_deadline<Old, New, D>(
81    d: &mut D,
82    old: &Old,
83    old_range: Range<usize>,
84    new: &New,
85    new_range: Range<usize>,
86    deadline: Option<Instant>,
87) -> Result<(), D::Error>
88where
89    Old: Index<usize> + ?Sized,
90    New: Index<usize> + ?Sized,
91    D: DiffHook,
92    Old::Output: Hash + Eq,
93    New::Output: PartialEq<Old::Output> + Hash + Eq,
94{
95    diff_deadline_impl(
96        d,
97        old,
98        old_range,
99        new,
100        new_range,
101        deadline,
102        HEURISTIC_OPTIONS,
103    )
104}

109    old: &Old,
110    old_range: Range<usize>,
111    new: &New,
112    new_range: Range<usize>,
113    deadline: Option<Instant>,

117    New: Index<usize> + ?Sized,
118    D: DiffHook,
119    Old::Output: Hash + Eq,

125fn diff_deadline_impl<Old, New, D>(
126    d: &mut D,
127    old: &Old,
128    old_range: Range<usize>,
129    new: &New,
130    new_range: Range<usize>,
131    deadline: Option<Instant>,
132    options: DiffOptions,
133) -> Result<(), D::Error>
134where
135    Old: Index<usize> + ?Sized,
136    New: Index<usize> + ?Sized,
137    D: DiffHook,
138    Old::Output: Hash + Eq,
139    New::Output: PartialEq<Old::Output> + Hash + Eq,
140{
141    if options.run_preflight
142        && preflight::maybe_emit_replace_fast_path(
143            d,
144            old,
145            old_range.clone(),
146            new,
147            new_range.clone(),
148            deadline,
149        )?
150    {
151        return Ok(());
152    }
153
154    // Trim before building the shared integer domain. Identical inputs then
155    // need no allocation, and sparse edits only remap their changed middle.
156    let prefix = common_prefix_len(old, old_range.clone(), new, new_range.clone());
157    let old_after_prefix = old_range.start + prefix..old_range.end;
158    let new_after_prefix = new_range.start + prefix..new_range.end;
159    let suffix = common_suffix_len(old, old_after_prefix.clone(), new, new_after_prefix.clone());
160    let old_middle = old_after_prefix.start..old_after_prefix.end - suffix;
161    let new_middle = new_after_prefix.start..new_after_prefix.end - suffix;
162
163    if prefix > 0 {
164        d.equal(old_range.start, new_range.start, prefix)?;
165    }
166
167    if !old_middle.is_empty() || !new_middle.is_empty() {
168        // Build a shared integer domain first so we can use a compact key type
169        // for match lists while supporting differing old/new output types.
170        let h = IdentifyDistinct::<usize>::new(old, old_middle, new, new_middle);
171        let mut no_finish_d = NoFinishHook::new(&mut *d);
172        diff_deadline_int(
173            &mut no_finish_d,
174            h.old_lookup(),
175            h.old_range(),
176            h.new_lookup(),
177            h.new_range(),
178            deadline,
179            options.use_raw_myers,
180        )?;
181    }
182
183    if suffix > 0 {
184        d.equal(old_range.end - suffix, new_range.end - suffix, suffix)?;
185    }
186
187    d.finish()
188}

190fn diff_deadline_int<Old, New, D>(
191    d: &mut D,
192    old: &Old,
193    old_range: Range<usize>,
194    new: &New,
195    new_range: Range<usize>,
196    deadline: Option<Instant>,
197    use_raw_myers: bool,
198) -> Result<(), D::Error>
199where
200    Old: Index<usize, Output = usize> + ?Sized,
201    New: Index<usize, Output = usize> + ?Sized,
202    D: DiffHook,
203{
204    if is_empty_range(&old_range) && is_empty_range(&new_range) {
205        d.finish()?;
206        return Ok(());
207    } else if is_empty_range(&new_range) {
208        d.delete(old_range.start, old_range.len(), new_range.start)?;
209        d.finish()?;
210        return Ok(());
211    } else if is_empty_range(&old_range) {
212        d.insert(old_range.start, new_range.start, new_range.len())?;
213        d.finish()?;
214        return Ok(());
215    }
216
217    let common_prefix_len = common_prefix_len(old, old_range.clone(), new, new_range.clone());
218    let common_suffix_len = common_suffix_len(
219        old,
220        old_range.start + common_prefix_len..old_range.end,
221        new,
222        new_range.start + common_prefix_len..new_range.end,
223    );
224
225    if common_prefix_len == old_range.len() && (old_range.len() == new_range.len()) {
226        d.equal(old_range.start, new_range.start, old_range.len())?;
227        d.finish()?;
228        return Ok(());
229    }
230
231    let old_mid_range =
232        (old_range.start + common_prefix_len)..(old_range.end.saturating_sub(common_suffix_len));
233    let new_mid_range =
234        (new_range.start + common_prefix_len)..(new_range.end.saturating_sub(common_suffix_len));
235
236    if common_prefix_len > 0 {
237        d.equal(old_range.start, new_range.start, common_prefix_len)?;
238    }
239
240    if is_empty_range(&old_mid_range) {
241        if !is_empty_range(&new_mid_range) {
242            d.insert(
243                old_mid_range.start,
244                new_mid_range.start,
245                new_mid_range.len(),
246            )?;
247        }
248    } else if is_empty_range(&new_mid_range) {
249        d.delete(
250            old_mid_range.start,
251            old_mid_range.len(),
252            new_mid_range.start,
253        )?;
254    } else if let Some(match_list) = build_match_list(new, new_mid_range.clone(), deadline) {
255        if let Some(anchors) = hunt_anchors(old, old_mid_range.clone(), &match_list, deadline) {
256            emit_anchored_script(
257                d,
258                old_mid_range.start,
259                old_mid_range.end,
260                new_mid_range.start,
261                new_mid_range.end,
262                &anchors,
263            )?;
264        } else {
265            let mut no_finish_d = NoFinishHook::new(&mut *d);
266            if use_raw_myers {
267                myers::diff_deadline_raw(
268                    &mut no_finish_d,
269                    old,
270                    old_mid_range,
271                    new,
272                    new_mid_range,
273                    deadline,
274                )?;
275            } else {
276                myers::diff_deadline(
277                    &mut no_finish_d,
278                    old,
279                    old_mid_range,
280                    new,
281                    new_mid_range,
282                    deadline,
283                )?;
284            }
285        }
286    } else {
287        let mut no_finish_d = NoFinishHook::new(&mut *d);
288        if use_raw_myers {
289            myers::diff_deadline_raw(
290                &mut no_finish_d,
291                old,
292                old_mid_range,
293                new,
294                new_mid_range,
295                deadline,
296            )?;
297        } else {
298            myers::diff_deadline(
299                &mut no_finish_d,
300                old,
301                old_mid_range,
302                new,
303                new_mid_range,
304                deadline,
305            )?;
306        }
307    }
308
309    if common_suffix_len > 0 {
310        d.equal(
311            old_range.end - common_suffix_len,
312            new_range.end - common_suffix_len,
313            common_suffix_len,
314        )?;
315    }
316
317    d.finish()
318}

320fn build_match_list<New>(
321    new: &New,
322    new_range: Range<usize>,
323    deadline: Option<Instant>,
324) -> Option<MapType<usize, HashBucket<usize>>>
325where
326    New: Index<usize, Output = usize> + ?Sized,
327{
328    let mut rv = MapType::<usize, HashBucket<usize>>::new();
329    for new_index in new_range {
330        if deadline_exceeded(deadline) {
331            return None;
332        }
333        if let Some(positions) = rv.get_mut(&new[new_index]) {
334            positions.push(new_index);
335        } else {
336            rv.insert(new[new_index], HashBucket::new(new_index));
337        }
338    }
339    Some(rv)
340}

342fn lower_bound(slice: &[usize], value: usize) -> usize {
343    let mut lo = 0;
344    let mut hi = slice.len();
345    while lo < hi {
346        let mid = lo + (hi - lo) / 2;
347        if slice[mid] < value {
348            lo = mid + 1;
349        } else {
350            hi = mid;
351        }
352    }
353    lo
354}

356fn hunt_anchors<Old>(
357    old: &Old,
358    old_range: Range<usize>,
359    match_list: &MapType<usize, HashBucket<usize>>,
360    deadline: Option<Instant>,
361) -> Option<Vec<(usize, usize)>>
362where
363    Old: Index<usize, Output = usize> + ?Sized,
364{
365    const MAX_MATCH_PAIRS_PER_INPUT_ITEM: usize = 64;
366    const MAX_MATCH_PAIRS: usize = 1_000_000;
367
368    let max_match_pairs = old_range
369        .len()
370        .saturating_add(match_list.values().map(HashBucket::len).sum())
371        .saturating_mul(MAX_MATCH_PAIRS_PER_INPUT_ITEM)
372        .min(MAX_MATCH_PAIRS);
373    let mut match_pairs = 0usize;
374    for old_index in old_range.clone() {
375        if deadline_exceeded(deadline) {
376            return None;
377        }
378        if let Some(new_indexes) = match_list.get(&old[old_index]) {
379            match_pairs = match_pairs.saturating_add(new_indexes.len());
380            if match_pairs > max_match_pairs {
381                // Highly repetitive inputs make R (and the candidate chain)
382                // quadratic. Myers handles these cases with bounded linear
383                // memory and is usually faster, so use it as a safety valve
384                // before allocating a partial candidate chain.
385                return None;
386            }
387        }
388    }
389
390    let mut thresh = Vec::new();
391    let mut links = Vec::new();
392    let mut candidates = Vec::new();
393
394    for old_index in old_range {
395        if deadline_exceeded(deadline) {
396            return None;
397        }
398
399        if let Some(new_indexes) = match_list.get(&old[old_index]) {
400            for &new_index in new_indexes.iter().rev() {
401                let k = lower_bound(&thresh, new_index);
402
403                if k == thresh.len() || new_index < thresh[k] {
404                    let prev = if k > 0 { links[k - 1] } else { usize::MAX };
405                    let candidate_index = candidates.len();
406                    candidates.push(Candidate {
407                        old_index,
408                        new_index,
409                        prev,
410                    });
411
412                    if k == thresh.len() {
413                        thresh.push(new_index);
414                        links.push(candidate_index);
415                    } else {
416                        thresh[k] = new_index;
417                        links[k] = candidate_index;
418                    }
419                }
420            }
421        }
422    }
423
424    let mut anchors = Vec::with_capacity(links.len());
425    let mut ptr = if let Some(ptr) = links.last().copied() {
426        ptr
427    } else {
428        return Some(anchors);
429    };
430
431    loop {
432        let c = candidates[ptr];
433        anchors.push((c.old_index, c.new_index));
434        if c.prev == usize::MAX {
435            break;
436        }
437        ptr = c.prev;
438    }
439
440    anchors.reverse();
441    Some(anchors)
442}

444fn emit_anchored_script<D>(
445    d: &mut D,
446    old_start: usize,
447    old_end: usize,
448    new_start: usize,
449    new_end: usize,
450    anchors: &[(usize, usize)],
451) -> Result<(), D::Error>
452where
453    D: DiffHook,
454{
455    let mut old_index = old_start;
456    let mut new_index = new_start;
457    let mut anchor_ptr = 0;
458
459    while anchor_ptr < anchors.len() {
460        let run_old_start = anchors[anchor_ptr].0;
461        let run_new_start = anchors[anchor_ptr].1;
462
463        if old_index < run_old_start {
464            d.delete(old_index, run_old_start - old_index, new_index)?;
465            old_index = run_old_start;
466        }
467
468        if new_index < run_new_start {
469            d.insert(old_index, new_index, run_new_start - new_index)?;
470        }
471
472        let mut run_len = 1;
473        anchor_ptr += 1;
474        while anchor_ptr < anchors.len()
475            && anchors[anchor_ptr].0 == run_old_start + run_len
476            && anchors[anchor_ptr].1 == run_new_start + run_len
477        {
478            run_len += 1;
479            anchor_ptr += 1;
480        }
481
482        d.equal(run_old_start, run_new_start, run_len)?;
483        old_index = run_old_start + run_len;
484        new_index = run_new_start + run_len;
485    }
486
487    if old_index < old_end {
488        d.delete(old_index, old_end - old_index, new_index)?;
489        old_index = old_end;
490    }
491
492    if new_index < new_end {
493        d.insert(old_index, new_index, new_end - new_index)?;
494    }
495
496    Ok(())
497}

600#[test]
601fn test_lcs_length_matches_classic_lcs() {
602    use crate::{Algorithm, DiffOp, capture_diff_slices};
603
604    fn equal_len(ops: &[DiffOp]) -> usize {
605        ops.iter()
606            .map(|op| match *op {
607                DiffOp::Equal { len, .. } => len,
608                _ => 0,
609            })
610            .sum()
611    }
612
613    let cases = [
614        (vec![1, 2, 3, 4, 5], vec![1, 2, 9, 4, 5]),
615        (vec![1, 2, 1, 2, 1, 2], vec![2, 1, 2, 1, 2, 1]),
616        (vec![0, 1, 4, 5, 8, 9], vec![0, 1, 3, 4, 5]),
617        (vec![5, 1, 2, 3, 4, 5, 6], vec![1, 2, 7, 3, 4, 8, 6]),
618    ];
619
620    for (a, b) in cases {
621        let hunt = capture_diff_slices(Algorithm::Hunt, &a, &b);
622        let lcs = capture_diff_slices(Algorithm::Lcs, &a, &b);
623        assert_eq!(equal_len(&hunt), equal_len(&lcs));
624    }
625}

637#[test]
638fn test_finish_called() {
639    struct HasRunFinish(bool);
640
641    impl DiffHook for HasRunFinish {
642        type Error = ();
643        fn finish(&mut self) -> Result<(), Self::Error> {
644            self.0 = true;
645            Ok(())
646        }
647    }
648
649    let mut d = HasRunFinish(false);
650    let slice = &[1, 2];
651    let slice2 = &[1, 2, 3];
652    diff(&mut d, slice, 0..slice.len(), slice2, 0..slice2.len()).unwrap();
653    assert!(d.0);
654
655    let mut d = HasRunFinish(false);
656    let slice = &[1, 2];
657    diff(&mut d, slice, 0..slice.len(), slice, 0..slice.len()).unwrap();
658    assert!(d.0);
659
660    let mut d = HasRunFinish(false);
661    let slice: &[u8] = &[];
662    diff(&mut d, slice, 0..slice.len(), slice, 0..slice.len()).unwrap();
663    assert!(d.0);
664}
