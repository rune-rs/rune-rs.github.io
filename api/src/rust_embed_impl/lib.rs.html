19fn embedded(
20  ident: &syn::Ident, relative_folder_path: Option<&str>, absolute_folder_path: Option<String>, prefix: Option<&str>, includes: &[String], excludes: &[String],
21  metadata_only: bool, crate_path: &syn::Path, compression: &str,
22) -> syn::Result<TokenStream2> {
23  extern crate rust_embed_utils;
24
25  let mut match_values = BTreeMap::new();
26  let mut list_values = Vec::<String>::new();
27
28  let includes: Vec<&str> = includes.iter().map(AsRef::as_ref).collect();
29  let excludes: Vec<&str> = excludes.iter().map(AsRef::as_ref).collect();
30  let matcher = PathMatcher::new(&includes, &excludes);
31  let entries = absolute_folder_path.clone()
32    .map(|v| rust_embed_utils::get_files(v, matcher))
33    .into_iter()
34    .flatten();
35  for rust_embed_utils::FileEntry { rel_path, full_canonical_path } in entries {
36    match_values.insert(
37      rel_path.clone(),
38      embed_file(relative_folder_path, ident, &rel_path, &full_canonical_path, metadata_only, crate_path, compression)?,
39    );
40
41    list_values.push(if let Some(prefix) = prefix {
42      format!("{}{}", prefix, rel_path)
43    } else {
44      rel_path
45    });
46  }
47
48  let array_len = list_values.len();
49
50  // If debug-embed is on, unconditionally include the code below. Otherwise,
51  // make it conditional on cfg(not(debug_assertions)).
52  let not_debug_attr = if cfg!(feature = "debug-embed") {
53    quote! {}
54  } else {
55    quote! { #[cfg(not(debug_assertions))]}
56  };
57
58  let handle_prefix = if let Some(prefix) = prefix {
59    quote! {
60      let file_path = file_path.strip_prefix(#prefix)?;
61    }
62  } else {
63    TokenStream2::new()
64  };
65  let match_values = match_values.into_iter().map(|(path, bytes)| {
66    quote! {
67        (#path, #bytes),
68    }
69  });
70  let use_compression = cfg!(feature = "compression") && !metadata_only;
71  let (entry_type, return_type) = if use_compression {
72    (
73      quote! { fn() -> #crate_path::EmbeddedCompressedFile },
74      quote! { #crate_path::EmbeddedCompressedFile },
75    )
76  } else {
77    (quote! { #crate_path::EmbeddedFile }, quote! { #crate_path::EmbeddedFile })
78  };
79  let get_value = if use_compression {
80    quote! {|idx| (ENTRIES[idx].1)()}
81  } else {
82    quote! {|idx| ENTRIES[idx].1.clone()}
83  };
84  let impls = if cfg!(feature = "compression") {
85    if metadata_only {
86      quote! {
87          pub fn compressed(_file_path: &str) -> ::std::option::Option<#crate_path::EmbeddedCompressedFile> {
88            ::std::option::Option::None
89          }
90          pub fn get(file_path: &str) -> ::std::option::Option<#crate_path::EmbeddedFile> {
91            #ident::__file(file_path)
92          }
93      }
94    } else {
95      quote! {
96          pub fn compressed(file_path: &str) -> ::std::option::Option<#crate_path::EmbeddedCompressedFile> {
97            #ident::__file(file_path)
98          }
99          pub fn get(file_path: &str) -> ::std::option::Option<#crate_path::EmbeddedFile> {
100            #ident::__file(file_path).map(|f| #crate_path::EmbeddedFile {
101                data: f.data.decoded().into(),
102                metadata: f.metadata,
103            })
104          }
105      }
106    }
107  } else {
108    quote! {
109        /// Get an embedded file and its metadata.
110        pub fn get(file_path: &str) -> ::std::option::Option<#crate_path::EmbeddedFile> {
111          #ident::__file(file_path)
112        }
113    }
114  };
115  let compressed_trait_impl = if cfg!(feature = "compression") {
116    quote! {
117      fn compressed(file_path: &str) -> ::std::option::Option<#crate_path::EmbeddedCompressedFile> {
118        #ident::compressed(file_path)
119      }
120    }
121  } else {
122    TokenStream2::new()
123  };
124  Ok(quote! {
125      #not_debug_attr
126      impl #ident {
127          fn __file(file_path: &str) -> ::std::option::Option<#return_type> {
128              #handle_prefix
129              let key = file_path.replace("\\", "/");
130              static ENTRIES: &'static [(&'static str, #entry_type)] = &[
131                  #(#match_values)*];
132              let position = ENTRIES.binary_search_by_key(&key.as_str(), |entry| entry.0);
133              position.ok().map(#get_value)
134          }
135
136          #impls
137
138          fn names() -> ::std::slice::Iter<'static, &'static str> {
139              const ITEMS: [&str; #array_len] = [#(#list_values),*];
140              ITEMS.iter()
141          }
142
143          /// Iterates over the file paths in the folder.
144          pub fn iter() -> impl ::std::iter::Iterator<Item = ::std::borrow::Cow<'static, str>> + 'static {
145              Self::names().map(|x| ::std::borrow::Cow::from(*x))
146          }
147      }
148
149      #not_debug_attr
150      impl #crate_path::RustEmbed for #ident {
151        #compressed_trait_impl
152        fn get(file_path: &str) -> ::std::option::Option<#crate_path::EmbeddedFile> {
153          #ident::get(file_path)
154        }
155        fn iter() -> #crate_path::Filenames {
156          #crate_path::Filenames::Embedded(#ident::names())
157        }
158      }
159  })
160}

162fn dynamic(
163  ident: &syn::Ident, folder_path: Option<String>, prefix: Option<&str>, includes: &[String], excludes: &[String], metadata_only: bool, crate_path: &syn::Path,
164) -> TokenStream2 {
165  let dynamic_compressed = if cfg!(feature = "compression") {
166    quote! {
167      fn compressed(_file_path: &str) -> ::std::option::Option<#crate_path::EmbeddedCompressedFile> {
168        ::std::option::Option::None
169      }
170    }
171  } else {
172    TokenStream2::new()
173  };
174  let (handle_prefix, map_iter) = if let ::std::option::Option::Some(prefix) = prefix {
175    (
176      quote! { let file_path = file_path.strip_prefix(#prefix)?; },
177      quote! { ::std::borrow::Cow::Owned(format!("{}{}", #prefix, e.rel_path)) },
178    )
179  } else {
180    (TokenStream2::new(), quote! { ::std::borrow::Cow::from(e.rel_path) })
181  };
182
183  let declare_includes = quote! {
184    const INCLUDES: &[&str] = &[#(#includes),*];
185  };
186
187  let declare_excludes = quote! {
188    const EXCLUDES: &[&str] = &[#(#excludes),*];
189  };
190
191  // In metadata_only mode, we still need to read file contents to generate the
192  // file hash, but then we drop the file data.
193  let strip_contents = metadata_only.then_some(quote! {
194      .map(|mut file| { file.data = ::std::default::Default::default(); file })
195  });
196
197  let methods = if let Some(ref folder_path) = folder_path {
198    quote! {
199          /// Get an embedded file and its metadata.
200          pub fn get(file_path: &str) -> ::std::option::Option<#crate_path::EmbeddedFile> {
201              #handle_prefix
202
203              let rel_file_path = file_path.replace("\\", "/");
204              let file_path = ::std::path::Path::new(#folder_path).join(&rel_file_path);
205
206              // Make sure the path requested does not escape the folder path.
207              // The folder path is canonicalized here at runtime rather than in the macro,
208              // because the two can differ when the binary runs in another environment
209              // than the one it was compiled in (e.g. cross-compiled and run under Wine,
210              // where `/home/..` canonicalizes to `\\?\Z:\home\..`).
211              let canonical_file_path = file_path.canonicalize().ok()?;
212              let canonical_folder_path = ::std::path::Path::new(#folder_path).canonicalize().ok()?;
213              if !canonical_file_path.starts_with(&canonical_folder_path) {
214                  // Tried to request a path that is not in the embedded folder
215
216                  // TODO: Currently it allows "path_traversal_attack" for the symlink files
217                  // For it to be working properly we need to get absolute path first
218                  // and check that instead if it starts with `canonical_folder_path`
219                  // https://doc.rust-lang.org/std/path/fn.absolute.html (currently nightly)
220                  // Should be allowed only if it was a symlink
221                  let metadata = ::std::fs::symlink_metadata(&file_path).ok()?;
222                  if !metadata.is_symlink() {
223                    return ::std::option::Option::None;
224                  }
225              }
226              let path_matcher = Self::matcher();
227              if path_matcher.is_path_included(&rel_file_path) {
228                #crate_path::utils::read_file_from_fs(&canonical_file_path).ok() #strip_contents
229              } else {
230                ::std::option::Option::None
231              }
232          }
233
234          /// Iterates over the file paths in the folder.
235          pub fn iter() -> impl ::std::iter::Iterator<Item = ::std::borrow::Cow<'static, str>> + 'static {
236              use ::std::path::Path;
237
238              #crate_path::utils::get_files(::std::string::String::from(#folder_path), Self::matcher())
239                  .map(|e| #map_iter)
240          }
241    }
242  } else {
243    quote! {
244          /// Get an embedded file and its metadata.
245          pub fn get(file_path: &str) -> ::std::option::Option<#crate_path::EmbeddedFile> {
246              // No folder_path defined so nothing can be returned.
247              None
248          }
249
250          /// Iterates over the file paths in the folder.
251          pub fn iter() -> impl ::std::iter::Iterator<Item = ::std::borrow::Cow<'static, str>> + 'static {
252              None.into_iter()
253          }
254    }
255  };
256
257  quote! {
258      #[cfg(debug_assertions)]
259      impl #ident {
260
261
262        fn matcher() -> #crate_path::utils::PathMatcher {
263            #declare_includes
264            #declare_excludes
265            static PATH_MATCHER: ::std::sync::OnceLock<#crate_path::utils::PathMatcher> = ::std::sync::OnceLock::new();
266            PATH_MATCHER.get_or_init(|| #crate_path::utils::PathMatcher::new(INCLUDES, EXCLUDES)).clone()
267        }
268
269          #methods
270      }
271
272      #[cfg(debug_assertions)]
273      impl #crate_path::RustEmbed for #ident {
274        #dynamic_compressed
275        fn get(file_path: &str) -> ::std::option::Option<#crate_path::EmbeddedFile> {
276          #ident::get(file_path)
277        }
278        fn iter() -> #crate_path::Filenames {
279          // the return type of iter() is unnamable, so we have to box it
280          #crate_path::Filenames::Dynamic(::std::boxed::Box::new(#ident::iter()))
281        }
282      }
283  }
284}

286fn generate_assets(
287  ident: &syn::Ident, relative_folder_path: Option<&str>, absolute_folder_path: Option<String>, prefix: Option<String>, includes: Vec<String>, excludes: Vec<String>,
288  metadata_only: bool, crate_path: &syn::Path, compression: &str,
289) -> syn::Result<TokenStream2> {
290  let embedded_impl = embedded(
291    ident,
292    relative_folder_path,
293    absolute_folder_path.clone(),
294    prefix.as_deref(),
295    &includes,
296    &excludes,
297    metadata_only,
298    crate_path,
299    compression,
300  );
301  if cfg!(feature = "debug-embed") {
302    return embedded_impl;
303  }
304  let embedded_impl = embedded_impl?;
305  let dynamic_impl = dynamic(ident, absolute_folder_path, prefix.as_deref(), &includes, &excludes, metadata_only, crate_path);
306
307  Ok(quote! {
308      #embedded_impl
309      #dynamic_impl
310  })
311}

313fn embed_file(
314  folder_path: Option<&str>, ident: &syn::Ident, rel_path: &str, full_canonical_path: &str, metadata_only: bool, crate_path: &syn::Path, compression: &str,
315) -> syn::Result<TokenStream2> {
316  let file = rust_embed_utils::read_file_from_fs(Path::new(full_canonical_path)).expect("File should be readable");
317  let hash = file.metadata.sha256_hash();
318  let (last_modified, created) = if cfg!(feature = "deterministic-timestamps") {
319    (quote! { ::std::option::Option::Some(0u64) }, quote! { ::std::option::Option::Some(0u64) })
320  } else {
321    let last_modified = match file.metadata.last_modified() {
322      Some(last_modified) => quote! { ::std::option::Option::Some(#last_modified) },
323      None => quote! { ::std::option::Option::None },
324    };
325    let created = match file.metadata.created() {
326      Some(created) => quote! { ::std::option::Option::Some(#created) },
327      None => quote! { ::std::option::Option::None },
328    };
329    (last_modified, created)
330  };
331  let embedding_code = if metadata_only {
332    quote! {
333        const BYTES: &'static [u8] = &[];
334    }
335  } else if cfg!(feature = "compression") {
336    let folder_path = folder_path.ok_or(syn::Error::new(ident.span(), "`folder` must be provided under `compression` feature."))?;
337    let full_relative_path = PathBuf::from_iter([folder_path, rel_path]);
338    let full_relative_path = full_relative_path.to_string_lossy();
339    let compression_ident = format_ident!("{}", compression);
340    quote! {
341      #crate_path::flate!(static BYTES: IFlate from #full_relative_path with #compression_ident);
342    }
343  } else {
344    quote! {
345      const BYTES: &'static [u8] = include_bytes!(#full_canonical_path);
346    }
347  };
348  Ok(if cfg!(feature = "compression") && !metadata_only {
349    quote! { || {
350        #embedding_code
351
352        #crate_path::EmbeddedCompressedFile {
353            data: &*BYTES,
354            metadata: #crate_path::utils::__rust_embed_metadata!([#(#hash),*], #last_modified, #created, #crate_path::__mimetype_of!(#full_canonical_path))
355        }
356    } }
357  } else {
358    quote! {
359      {
360        #embedding_code
361
362        #crate_path::EmbeddedFile {
363            data: ::std::borrow::Cow::Borrowed(BYTES),
364            metadata: #crate_path::utils::__rust_embed_metadata!([#(#hash),*], #last_modified, #created, #crate_path::__mimetype_of!(#full_canonical_path))
365        }
366      }
367    }
368  })
369}

371/// Find all pairs of the `name = "value"` attribute from the derive input
372fn find_attribute_values(ast: &syn::DeriveInput, attr_name: &str) -> Vec<String> {
373  ast
374    .attrs
375    .iter()
376    .filter(|value| value.path().is_ident(attr_name))
377    .filter_map(|attr| match &attr.meta {
378      Meta::NameValue(MetaNameValue {
379        value: Expr::Lit(ExprLit { lit: Lit::Str(val), .. }),
380        ..
381      }) => Some(val.value()),
382      _ => None,
383    })
384    .collect()
385}
386
387fn find_bool_attribute(ast: &syn::DeriveInput, attr_name: &str) -> Option<bool> {
388  ast
389    .attrs
390    .iter()
391    .find(|value| value.path().is_ident(attr_name))
392    .and_then(|attr| match &attr.meta {
393      Meta::NameValue(MetaNameValue {
394        value: Expr::Lit(ExprLit { lit: Lit::Bool(val), .. }),
395        ..
396      }) => Some(val.value()),
397      _ => None,
398    })
399}

401fn impl_rust_embed(ast: &syn::DeriveInput) -> syn::Result<TokenStream2> {
402  match ast.data {
403    Data::Struct(ref data) => match data.fields {
404      Fields::Unit => {}
405      _ => return Err(syn::Error::new_spanned(ast, "RustEmbed can only be derived for unit structs")),
406    },
407    _ => return Err(syn::Error::new_spanned(ast, "RustEmbed can only be derived for unit structs")),
408  };
409
410  let crate_path: syn::Path = find_attribute_values(ast, "crate_path")
411    .last()
412    .map_or_else(|| syn::parse_str("rust_embed").unwrap(), |v| syn::parse_str(v).unwrap());
413
414  let mut folder_paths = find_attribute_values(ast, "folder");
415  if folder_paths.len() != 1 {
416    return Err(syn::Error::new_spanned(
417      ast,
418      "#[derive(RustEmbed)] must contain one attribute like this #[folder = \"examples/public/\"]",
419    ));
420  }
421  let folder_path = folder_paths.remove(0);
422
423  const COMPRESSION_ALGOS: &[&str] = &["deflate", "zstd"];
424  let mut compression = find_attribute_values(ast, "compression");
425  if cfg!(feature = "compression") && (compression.len() > 1 || !compression.first().is_none_or(|v| COMPRESSION_ALGOS.contains(&v.as_str()))) {
426    return Err(syn::Error::new_spanned(
427      ast,
428      format!(
429        "#[derive(RustEmbed)] must contain one attribute like this #[compression = \"deflate\"]\navailable compression algorithm are {COMPRESSION_ALGOS:?}"
430      ),
431    ));
432  }
433  let compression = compression.pop().unwrap_or_else(|| "deflate".to_owned());
434
435  let prefix = find_attribute_values(ast, "prefix").into_iter().next();
436  let includes = find_attribute_values(ast, "include");
437  let excludes = find_attribute_values(ast, "exclude");
438  let metadata_only = find_bool_attribute(ast, "metadata_only").unwrap_or(false);
439  let allow_missing = find_bool_attribute(ast, "allow_missing").unwrap_or(false);
440
441  #[cfg(not(feature = "include-exclude"))]
442  if !includes.is_empty() || !excludes.is_empty() {
443    return Err(syn::Error::new_spanned(
444      ast,
445      "Please turn on the `include-exclude` feature to use the `include` and `exclude` attributes",
446    ));
447  }
448
449  #[cfg(feature = "interpolate-folder-path")]
450  let folder_path = match shellexpand::full(&folder_path) {
451    Ok(v) => Ok(Some(v.to_string())),
452    Err(v) if v.cause == std::env::VarError::NotPresent && allow_missing => Ok(None),
453    Err(v) => Err(syn::Error::new_spanned(ast, v.to_string())),
454  }?;
455  #[cfg(not(feature = "interpolate-folder-path"))]
456  let folder_path = Some(folder_path);
457
458  // Base relative paths on the Cargo.toml location
459  let (relative_path, absolute_folder_path) = if let Some(folder_path) = folder_path {
460    let (relative_path, absolute_folder_path) = if Path::new(&folder_path).is_relative() {
461      let absolute_path = Path::new(&env::var("CARGO_MANIFEST_DIR").unwrap())
462        .join(&folder_path)
463        .to_str()
464        .unwrap()
465        .to_owned();
466      (Some(folder_path.clone()), absolute_path)
467    } else {
468      if cfg!(feature = "compression") {
469        let cargo_manifest_dir = env::var("CARGO_MANIFEST_DIR").unwrap();
470        match Path::new(&folder_path).strip_prefix(&cargo_manifest_dir) {
471          Ok(rel) => {
472            let rel_str = rel.to_str().unwrap().to_owned();
473            (Some(rel_str), folder_path)
474          }
475          Err(_) => return Err(syn::Error::new_spanned(ast, "`folder` must be a relative path under `compression` feature.")),
476        }
477      } else {
478        (None, folder_path)
479      }
480    };
481
482    if !Path::new(&absolute_folder_path).exists() && !allow_missing {
483      let mut message = format!(
484        "#[derive(RustEmbed)] folder '{}' does not exist. cwd: '{}'",
485        absolute_folder_path,
486        std::env::current_dir().unwrap().to_str().unwrap()
487      );
488
489      // Add a message about the interpolate-folder-path feature if the path may
490      // include a variable
491      if absolute_folder_path.contains('$') && cfg!(not(feature = "interpolate-folder-path")) {
492        message += "\nA variable has been detected. RustEmbed can expand variables \
493                    when the `interpolate-folder-path` feature is enabled.";
494      }
495
496      return Err(syn::Error::new_spanned(ast, message));
497    };
498    (relative_path, Some(absolute_folder_path))
499  } else {
500    (None, None)
501  };
502
503  generate_assets(
504    &ast.ident,
505    relative_path.as_deref(),
506    absolute_folder_path,
507    prefix,
508    includes,
509    excludes,
510    metadata_only,
511    &crate_path,
512    &compression,
513  )
514}

516#[proc_macro_derive(RustEmbed, attributes(folder, prefix, include, exclude, allow_missing, metadata_only, crate_path, compression))]
517pub fn derive_input_object(input: TokenStream) -> TokenStream {
518  let ast = parse_macro_input!(input as DeriveInput);
519  match impl_rust_embed(&ast) {
520    Ok(ok) => ok.into(),
521    Err(e) => e.to_compile_error().into(),
522  }
523}

525/// Expands to a string literal containing the mimetype of the file at the given
526/// path, guessed from its extension.
527///
528/// This lives in the proc-macro crate (which depends on `mime_guess`
529/// unconditionally) rather than being computed inside the derive's host-side
530/// logic: the derive must emit tokens that do not depend on the *host*'s
531/// `mime-guess` feature, and this proc macro is invoked only from the
532/// target-side `__rust_embed_metadata!` macro. When the target has `mime-guess`
533/// disabled, the invocation generated by the derive is discarded by that macro
534/// without ever being expanded.
535#[doc(hidden)]
536#[proc_macro]
537pub fn __mimetype_of(input: TokenStream) -> TokenStream {
538  let path = parse_macro_input!(input as syn::LitStr).value();
539  let mimetype = mime_guess::from_path(&path).first_or_octet_stream().to_string();
540  quote! { #mimetype }.into()
541}
