9macro_rules! try_browser {
10    ( $options: expr, $name:expr, $( $arg:expr ),+ ) => {
11        for_matching_path($name, |pb| {
12            let mut cmd = Command::new(pb);
13            $(
14                cmd.arg($arg);
15            )+
16            run_command(&mut cmd, !is_text_browser(&pb), $options)
17        })
18    }
19}

21/// Deal with opening of browsers on Linux and *BSD - currently supports only the default browser
22///
23/// The mechanism of opening the default browser is as follows:
24/// 1. Attempt to use $BROWSER env var if available
25/// 2. Attempt to use xdg-open
26/// 3. Attempt to use window manager specific commands, like gnome-open, kde-open etc. incl. WSL
27/// 4. Fallback to x-www-browser
28pub(super) fn open_browser_internal(
29    browser: Browser,
30    target: &TargetType,
31    options: &BrowserOptions,
32) -> Result<()> {
33    match browser {
34        Browser::Default => open_browser_default(target, options),
35        _ => Err(Error::new(
36            ErrorKind::NotFound,
37            "only default browser supported",
38        )),
39    }
40}

42/// Open the default browser.
43///
44/// [BrowserOptions::dry_run] is handled inside [run_command], as all execution paths eventually
45/// rely on it to execute.
46fn open_browser_default(target: &TargetType, options: &BrowserOptions) -> Result<()> {
47    let url: &str = target;
48
49    // we first try with the $BROWSER env
50    try_with_browser_env(url, options)
51        // allow for haiku's open specifically
52        .or_else(|_| try_haiku(options, url))
53        // then we try with xdg configuration
54        .or_else(|_| try_xdg(options, url))
55        // else do desktop specific stuff
56        .or_else(|r| match guess_desktop_env() {
57            "kde" => try_browser!(options, "kde-open", url)
58                .or_else(|_| try_browser!(options, "kde-open5", url))
59                .or_else(|_| try_browser!(options, "kfmclient", "newTab", url)),
60
61            "gnome" => try_browser!(options, "gio", "open", url)
62                .or_else(|_| try_browser!(options, "gvfs-open", url))
63                .or_else(|_| try_browser!(options, "gnome-open", url)),
64
65            "mate" => try_browser!(options, "gio", "open", url)
66                .or_else(|_| try_browser!(options, "gvfs-open", url))
67                .or_else(|_| try_browser!(options, "mate-open", url)),
68
69            "xfce" => try_browser!(options, "exo-open", url)
70                .or_else(|_| try_browser!(options, "gio", "open", url))
71                .or_else(|_| try_browser!(options, "gvfs-open", url)),
72
73            "wsl" => try_wsl(options, target),
74
75            "flatpak" => try_flatpak(options, target),
76
77            _ => Err(r),
78        })
79        // at the end, we'll try x-www-browser and return the result as is
80        .or_else(|_| try_browser!(options, "x-www-browser", url))
81        // if all above failed, map error to not found
82        .map_err(|_| {
83            Error::new(
84                ErrorKind::NotFound,
85                "No valid browsers detected. You can specify one in BROWSER environment variable",
86            )
87        })
88        // and convert a successful result into a ()
89        .map(|_| ())
90}

92fn try_with_browser_env(url: &str, options: &BrowserOptions) -> Result<()> {
93    // $BROWSER can contain ':' delimited options, each representing a potential browser command line
94    for browser_env in std::env::var("BROWSER")
95        .unwrap_or_else(|_| String::from(""))
96        .split(':')
97    {
98        if !browser_env.is_empty() {
99            let cmdarr = browser_env_cmd_arr(browser_env, url);
100            let browser_cmd = &cmdarr[0];
101            let env_exit = for_matching_path(browser_cmd, |pb| {
102                let mut cmd = Command::new(pb);
103                for arg in cmdarr.iter().skip(1) {
104                    cmd.arg(arg);
105                }
106                run_command(&mut cmd, !is_text_browser(pb), options)
107            });
108            if env_exit.is_ok() {
109                return Ok(());
110            }
111        }
112    }
113    Err(Error::new(
114        ErrorKind::NotFound,
115        "No valid browser configured in BROWSER environment variable",
116    ))
117}

119/// Given a part in BROWSER env variable, return a Vec of parsed strings, such
120/// that:
121/// - `%s` is replaced with the provided `url`.
122/// - `%c` is replaced with `:`.
123/// - `%%` is replaced with `%`.
124///
125/// If none of the options had a `%s` in them, `url` is appended as the last option, so that
126/// it's always passed to the browser exactly once. See issue #120
127fn browser_env_cmd_arr(browser_env: &str, url: &str) -> Vec<String> {
128    let mut url_set = false;
129    let mut arr: Vec<String> = browser_env
130        .split_ascii_whitespace()
131        .map(|opt| {
132            let (val, has_url) = expand_field_codes(opt, url);
133            url_set |= has_url;
134            val
135        })
136        .collect();
137    if !url_set {
138        arr.push(url.to_string());
139    }
140    arr
141}

143/// Expand the field codes inside a single BROWSER option `opt`, returning the expanded option,
144/// and whether it had a `%s` in it.
145///
146/// We expand in a single left to right pass, instead of one [str::replace] per field code, so
147/// that each `%` is consumed by exactly one expansion. Doing it otherwise would make an option
148/// like `--url=%s%%` lose either its `%s` or its `%%` expansion, and would also make the text
149/// coming in from `url` liable to be scanned for field codes.
150fn expand_field_codes(opt: &str, url: &str) -> (String, bool) {
151    let mut val = String::with_capacity(opt.len());
152    let mut url_set = false;
153    let mut chars = opt.chars();
154    while let Some(ch) = chars.next() {
155        if ch != '%' {
156            val.push(ch);
157            continue;
158        }
159        match chars.next() {
160            Some('s') => {
161                val.push_str(url);
162                url_set = true;
163            }
164            Some('c') => val.push(':'),
165            Some('%') => val.push('%'),
166            // we leave unknown field codes untouched
167            Some(other) => {
168                val.push('%');
169                val.push(other);
170            }
171            None => val.push('%'),
172        }
173    }
174    (val, url_set)
175}

177/// Check if we are inside WSL on Windows, and interoperability with Windows tools is
178/// enabled.
179fn is_wsl() -> bool {
180    // we should check in procfs only on linux, as for non-linux it will likely be
181    // a disk hit, which we should avoid.
182    if cfg!(target_os = "linux") {
183        // we check if interop with windows tools is allowed, as if it isn't, we won't
184        // be able to invoke windows commands anyways.
185        // See: https://learn.microsoft.com/en-us/windows/wsl/filesystems#disable-interoperability
186        if let Ok(s) = std::fs::read_to_string("/proc/sys/fs/binfmt_misc/WSLInterop") {
187            s.contains("enabled")
188        } else {
189            false
190        }
191    } else {
192        // we short-circuit and return false on non-linux
193        false
194    }
195}

205/// Detect the desktop environment
206fn guess_desktop_env() -> &'static str {
207    let unknown = "unknown";
208    let xcd: String = std::env::var("XDG_CURRENT_DESKTOP")
209        .unwrap_or_else(|_| unknown.into())
210        .to_ascii_lowercase();
211    let dsession: String = std::env::var("DESKTOP_SESSION")
212        .unwrap_or_else(|_| unknown.into())
213        .to_ascii_lowercase();
214
215    if is_flatpak() {
216        "flatpak"
217    } else if xcd.contains("gnome") || xcd.contains("cinnamon") || dsession.contains("gnome") {
218        // GNOME and its derivatives
219        "gnome"
220    } else if xcd.contains("kde")
221        || std::env::var("KDE_FULL_SESSION").is_ok()
222        || std::env::var("KDE_SESSION_VERSION").is_ok()
223    {
224        // KDE: https://userbase.kde.org/KDE_System_Administration/Environment_Variables#Automatically_Set_Variables
225        "kde"
226    } else if xcd.contains("mate") || dsession.contains("mate") {
227        // We'll treat MATE as distinct from GNOME due to mate-open
228        "mate"
229    } else if xcd.contains("xfce") || dsession.contains("xfce") {
230        // XFCE
231        "xfce"
232    } else if is_wsl() {
233        // WSL
234        "wsl"
235    } else {
236        // All others
237        unknown
238    }
239}

241/// Open browser in WSL environments
242fn try_wsl(options: &BrowserOptions, target: &TargetType) -> Result<()> {
243    match target.0.scheme() {
244        "http" | "https" => {
245            let url: &str = target;
246            try_browser!(
247                options,
248                "cmd.exe",
249                "/c",
250                "start",
251                url.replace('^', "^^").replace('&', "^&")
252            )
253            .or_else(|_| {
254                try_browser!(
255                    options,
256                    "powershell.exe",
257                    "Start",
258                    url.replace('&', "\"&\"")
259                )
260            })
261            .or_else(|_| try_browser!(options, "wsl-open", url))
262        }
263        #[cfg(all(
264            target_os = "linux",
265            not(feature = "hardened"),
266            not(feature = "disable-wsl")
267        ))]
268        "file" => {
269            // we'll need to detect the default browser and then invoke it
270            // with wsl translated path
271            let wc = wsl::get_wsl_win_config()?;
272            let mut cmd = if wc.powershell_path.is_some() {
273                wsl::get_wsl_windows_browser_ps(&wc, target)
274            } else {
275                wsl::get_wsl_windows_browser_cmd(&wc, target)
276            }?;
277            run_command(&mut cmd, true, options)
278        }
279        _ => Err(Error::new(ErrorKind::NotFound, "invalid browser")),
280    }
281}

283/// Open browser in Flatpak environments
284fn try_flatpak(options: &BrowserOptions, target: &TargetType) -> Result<()> {
285    match target.0.scheme() {
286        "http" | "https" => {
287            let url: &str = target;
288            // we assume xdg-open to be present, given that it's a part of standard
289            // runtime & SDK of flatpak
290            try_browser!(options, "xdg-open", url)
291        }
292        // we support only http urls under Flatpak to adhere to the defined
293        // Consistent Behaviour, as effectively DBUS is used interally, and
294        // there doesn't seem to be a way for us to determine actual browser
295        _ => Err(Error::new(ErrorKind::NotFound, "only http urls supported")),
296    }
297}

309/// Dig into XDG settings (if xdg is available) to force it to open the browser, instead of
310/// the default application
311fn try_xdg(options: &BrowserOptions, url: &str) -> Result<()> {
312    // run: xdg-settings get default-web-browser
313    let browser_name_os = for_matching_path("xdg-settings", |pb| {
314        Command::new(pb)
315            .args(["get", "default-web-browser"])
316            .stdin(Stdio::null())
317            .stderr(Stdio::null())
318            .output()
319    })
320    .map_err(|_| Error::new(ErrorKind::NotFound, "unable to determine xdg browser"))?
321    .stdout;
322
323    // convert browser name to a utf-8 string and trim off the trailing newline
324    let browser_name = String::from_utf8(browser_name_os)
325        .map_err(|_| Error::new(ErrorKind::NotFound, "invalid default browser name"))?
326        .trim()
327        .to_owned();
328    if browser_name.is_empty() {
329        return Err(Error::new(ErrorKind::NotFound, "no default xdg browser"));
330    }
331    trace!("found xdg browser: {browser_name:?}");
332
333    // search for the config file corresponding to this browser name
334    let mut config_found = false;
335    let app_suffix = "applications";
336    for xdg_dir in get_xdg_dirs().iter_mut() {
337        let mut config_path = xdg_dir.join(app_suffix).join(&browser_name);
338        trace!("checking for xdg config at {config_path:?}");
339        let mut metadata = config_path.metadata();
340        if metadata.is_err() && browser_name.contains('-') {
341            // as per the spec, we need to replace '-' with /
342            let child_path = browser_name.replace('-', "/");
343            config_path = xdg_dir.join(app_suffix).join(child_path);
344            metadata = config_path.metadata();
345        }
346        if metadata.is_ok() {
347            // we've found the config file, so we try running using that
348            config_found = true;
349            match open_using_xdg_config(&config_path, options, url) {
350                Ok(x) => return Ok(x), // return if successful
351                Err(err) => {
352                    // if we got an error other than NotFound, then we short
353                    // circuit, and do not try any more options, else we
354                    // continue to try more
355                    if err.kind() != ErrorKind::NotFound {
356                        return Err(err);
357                    }
358                }
359            }
360        }
361    }
362
363    if config_found {
364        Err(Error::other("xdg-open failed"))
365    } else {
366        Err(Error::new(ErrorKind::NotFound, "no valid xdg config found"))
367    }
368}

370/// Opens `url` using xdg configuration found in `config_path`
371///
372/// See https://specifications.freedesktop.org/desktop-entry-spec/latest for details
373fn open_using_xdg_config(config_path: &PathBuf, options: &BrowserOptions, url: &str) -> Result<()> {
374    let file = std::fs::File::open(config_path)?;
375    let mut in_desktop_entry = false;
376    let mut hidden = false;
377    let mut cmdline: Option<String> = None;
378    let mut requires_terminal = false;
379
380    // we capture important keys under the [Desktop Entry] section, as defined under:
381    // https://specifications.freedesktop.org/desktop-entry-spec/latest/ar01s06.html
382    for line in BufReader::new(file).lines().map_while(Result::ok) {
383        if line == "[Desktop Entry]" {
384            in_desktop_entry = true;
385        } else if line.starts_with('[') {
386            in_desktop_entry = false;
387        } else if in_desktop_entry && !line.starts_with('#') {
388            if let Some(idx) = line.find('=') {
389                let key = &line[..idx];
390                let value = &line[idx + 1..];
391                match key {
392                    "Exec" => cmdline = Some(value.to_owned()),
393                    "Hidden" => hidden = value == "true",
394                    "Terminal" => requires_terminal = value == "true",
395                    _ => (), // ignore
396                }
397            }
398        }
399    }
400
401    if hidden {
402        // we ignore this config if it was marked hidden/deleted
403        return Err(Error::new(ErrorKind::NotFound, "xdg config is hidden"));
404    }
405
406    if let Some(cmdline) = cmdline {
407        // we have a valid configuration
408        let cmdarr: Vec<&str> = cmdline.split_ascii_whitespace().collect();
409        let browser_cmd = cmdarr[0];
410        for_matching_path(browser_cmd, |pb| {
411            let mut cmd = Command::new(pb);
412            let mut url_added = false;
413            for arg in cmdarr.iter().skip(1) {
414                match *arg {
415                    "%u" | "%U" | "%f" | "%F" => {
416                        url_added = true;
417                        cmd.arg(url)
418                    }
419                    _ => cmd.arg(arg),
420                };
421            }
422            if !url_added {
423                // append the url as an argument only if it was not already set
424                cmd.arg(url);
425            }
426            run_command(&mut cmd, !requires_terminal, options)
427        })
428    } else {
429        // we don't have a valid config
430        Err(Error::new(ErrorKind::NotFound, "not a valid xdg config"))
431    }
432}

434/// Get the list of directories in which the desktop file needs to be searched
435fn get_xdg_dirs() -> Vec<PathBuf> {
436    let mut xdg_dirs: Vec<PathBuf> = Vec::new();
437
438    #[allow(deprecated)]
439    let data_home = std::env::var("XDG_DATA_HOME")
440        .ok()
441        .map(PathBuf::from)
442        .filter(|path| path.is_absolute())
443        .or_else(|| std::env::home_dir().map(|path| path.join(".local/share")));
444    if let Some(data_home) = data_home {
445        xdg_dirs.push(data_home);
446    }
447
448    if let Ok(data_dirs) = std::env::var("XDG_DATA_DIRS") {
449        for d in data_dirs.split(':') {
450            xdg_dirs.push(PathBuf::from(d));
451        }
452    } else {
453        xdg_dirs.push(PathBuf::from("/usr/local/share"));
454        xdg_dirs.push(PathBuf::from("/usr/share"));
455    }
456
457    xdg_dirs
458}

460/// Returns true if specified command refers to a known list of text browsers
461fn is_text_browser(pb: &Path) -> bool {
462    for browser in TEXT_BROWSERS.iter() {
463        if pb.ends_with(browser) {
464            return true;
465        }
466    }
467    false
468}

470fn for_matching_path<F, T>(name: &str, op: F) -> Result<T>
471where
472    F: FnOnce(&PathBuf) -> Result<T>,
473{
474    let err = Err(Error::new(ErrorKind::NotFound, "command not found"));
475
476    // if the name already includes path separator, we should not try to do a PATH search on it
477    // as it's likely an absolutely or relative name, so we treat it as such.
478    if name.contains(MAIN_SEPARATOR) {
479        let pb = std::path::PathBuf::from(name);
480        if let Ok(metadata) = pb.metadata() {
481            if metadata.is_file() && metadata.permissions().mode() & 0o111 != 0 {
482                return op(&pb);
483            }
484        } else {
485            return err;
486        }
487    } else {
488        // search for this name inside PATH
489        if let Ok(path) = std::env::var("PATH") {
490            for entry in path.split(':') {
491                let mut pb = std::path::PathBuf::from(entry);
492                pb.push(name);
493                if let Ok(metadata) = pb.metadata() {
494                    if metadata.is_file() && metadata.permissions().mode() & 0o111 != 0 {
495                        return op(&pb);
496                    }
497                }
498            }
499        }
500    }
501    // return the not found err, if we didn't find anything above
502    err
503}

642            let mut cmd_path: Option<PathBuf> = None;
643            let mut powershell_path: Option<PathBuf> = None;
644            for path in std::env::split_paths(&path_env) {
645                let path_s = path.to_string_lossy().to_ascii_lowercase();
646                let path_s = path_s.trim_end_matches('/');
647                if path_s.ends_with("/windows/system32") {
648                    root = Some(std::fs::canonicalize(path.join("../.."))?);
649                    cmd_path = Some(path.join("cmd.exe"));
650                    break;
651                }
652            }
653            if let Some(ref root) = root {
654                for path in std::env::split_paths(&path_env) {
655                    if path.starts_with(root) {
656                        let pb = path.join("powershell.exe");
657                        if pb.is_file() {
658                            powershell_path = Some(pb);
659                        }
660                    }
661                }
662            }
663            if let Some(root) = root {
664                let cmd_path = cmd_path.unwrap_or_else(|| (root).join("windows/system32/cmd.exe"));
665                Ok(WindowsConfig {
666                    root,
667                    cmd_path,
668                    powershell_path,
669                })
670            } else {
671                Err(err_fn())
672            }
673        } else {
674            Err(err_fn())
675        }
676    }

678    /// Try to get default browser command from powershell.exe
679    pub(super) fn get_wsl_windows_browser_ps(
680        wc: &WindowsConfig,
681        url: &TargetType,
682    ) -> Result<Command> {
683        let err_fn = || Error::new(ErrorKind::NotFound, "powershell.exe error");
684        let ps_exe = wc.powershell_path.as_ref().ok_or_else(err_fn)?;
685        let mut cmd = Command::new(ps_exe);
686        cmd.arg("-NoLogo")
687            .arg("-NoProfile")
688            .arg("-NonInteractive")
689            .arg("-Command")
690            .arg("-")
691            .stdin(Stdio::piped())
692            .stdout(Stdio::piped())
693            .stderr(Stdio::null());
694        log::debug!("running command: ${cmd:?}");
695        let mut child = cmd.spawn()?;
696
697        let mut stdin = child.stdin.take().ok_or_else(err_fn)?;
698        std::io::Write::write_all(&mut stdin, WSL_PS_SCRIPT.as_bytes())?;
699        drop(stdin); // flush to stdin, and close
700        let output_u8 = child.wait_with_output()?;
701        let output = String::from_utf8_lossy(&output_u8.stdout);
702        let output = output.trim();
703        if output.is_empty() {
704            Err(err_fn())
705        } else {
706            parse_wsl_cmdline(wc, output, url)
707        }
708    }

727        let output = output.trim();
728        if output.is_empty() {
729            Err(err_fn())
730        } else {
731            parse_wsl_cmdline(wc, output, url)
732        }

735    /// Given the configured command line `cmdline` in registry, and the given `url`,
736    /// return the appropriate `Command` to invoke
737    fn parse_wsl_cmdline(wc: &WindowsConfig, cmdline: &str, url: &TargetType) -> Result<Command> {
738        let mut tokens: Vec<String> = Vec::new();
739        let filepath = wsl_get_filepath_from_url(wc, url)?;
740        let fp = &filepath;
741        for_each_token(cmdline, |token: &str| {
742            if matches!(token, "%0" | "%1") {
743                tokens.push(fp.to_owned());
744            } else {
745                tokens.push(token.to_string());
746            }
747        });
748        if tokens.is_empty() {
749            Err(Error::new(ErrorKind::NotFound, "invalid command"))
750        } else {
751            let progpath = wsl_path_win2lin(wc, &tokens[0])?;
752            let mut cmd = Command::new(progpath);
753            if tokens.len() > 1 {
754                cmd.args(&tokens[1..]);
755            }
756            Ok(cmd)
757        }
758    }

760    fn wsl_get_filepath_from_url(wc: &WindowsConfig, target: &TargetType) -> Result<String> {
761        let url = &target.0;
762        if url.scheme() == "file" {
763            if url.host().is_none() {
764                let path = url
765                    .to_file_path()
766                    .map_err(|_| Error::new(ErrorKind::NotFound, "invalid path"))?;
767                wsl_path_lin2win(wc, path)
768            } else {
769                Ok(format!("\\\\wsl${}", url.path().replace('/', "\\")))
770            }
771        } else {
772            Ok(url.as_str().to_string())
773        }
774    }

776    /// Converts a windows path to linux `PathBuf`
777    fn wsl_path_win2lin(wc: &WindowsConfig, path: &str) -> Result<PathBuf> {
778        let err_fn = || Error::new(ErrorKind::NotFound, "invalid windows path");
779        if path.len() > 3 {
780            let pfx = &path[..3];
781            if matches!(pfx, "C:\\" | "c:\\") {
782                let win_path = path[3..].replace('\\', "/");
783                Ok(wc.root.join(win_path))
784            } else {
785                Err(err_fn())
786            }
787        } else {
788            Err(err_fn())
789        }
790    }

792    /// Converts a linux path to windows. We using `String` instead of `OsString` as
793    /// return type because the `OsString` will be different b/w Windows & Linux.
794    fn wsl_path_lin2win(wc: &WindowsConfig, path: impl AsRef<Path>) -> Result<String> {
795        let path = path.as_ref();
796        if let Ok(path) = path.strip_prefix(&wc.root) {
797            // windows can access this path directly
798            Ok(format!("C:\\{}", path.as_os_str().to_string_lossy()).replace('/', "\\"))
799        } else {
800            // windows needs to access it via network
801            let wsl_hostname = get_wsl_distro_name(wc)?;
802            Ok(format!(
803                "\\\\wsl$\\{wsl_hostname}{}",
804                path.as_os_str().to_string_lossy()
805            )
806            .replace('/', "\\"))
807        }
808    }

810    /// Gets the WSL distro name
811    fn get_wsl_distro_name(wc: &WindowsConfig) -> Result<String> {
812        let err_fn = || Error::other("unable to determine wsl distro name");
813
814        // mostly we should be able to get it from the WSL_DISTRO_NAME env var
815        if let Ok(wsl_hostname) = std::env::var("WSL_DISTRO_NAME") {
816            Ok(wsl_hostname)
817        } else {
818            // but if not (e.g. if we were running as root), we can invoke
819            // powershell.exe to determine pwd and from there infer the distro name
820            let psexe = wc.powershell_path.as_ref().ok_or_else(err_fn)?;
821            let mut cmd = Command::new(psexe);
822            cmd.arg("-NoLogo")
823                .arg("-NoProfile")
824                .arg("-NonInteractive")
825                .arg("-Command")
826                .arg("$loc = Get-Location\nWrite-Output $loc.Path")
827                .current_dir("/")
828                .stdin(Stdio::null())
829                .stderr(Stdio::null());
830            log::debug!("running command: ${cmd:?}");
831            let output_u8 = cmd.output()?.stdout;
832            let output = String::from_utf8_lossy(&output_u8);
833            let output = output.trim_end_matches('\\');
834            let idx = output.find("::\\\\").ok_or_else(err_fn)?;
835            Ok((output[idx + 9..]).trim().to_string())
836        }
837    }

839    /// Powershell script to get the default browser command.
840    ///
841    /// Adapted from https://stackoverflow.com/a/60972216
842    const WSL_PS_SCRIPT: &str = r#"
843$Signature = @"
844using System;
845using System.Runtime.InteropServices;
846using System.Text;
847public static class Win32Api
848{
849
850    [DllImport("Shlwapi.dll", SetLastError = true, CharSet = CharSet.Auto)]
851    static extern uint AssocQueryString(AssocF flags, AssocStr str, string pszAssoc, string pszExtra,[Out] System.Text.StringBuilder pszOut, ref uint pcchOut);
852
853    public static string GetDefaultBrowser()
854    {
855        AssocF assocF = AssocF.IsProtocol;
856        AssocStr association = AssocStr.Command;
857        string assocString = "http";
858
859        uint length = 1024; // we assume 1k is sufficient memory to hold the command
860        var sb = new System.Text.StringBuilder((int) length);
861        uint ret = ret = AssocQueryString(assocF, association, assocString, null, sb, ref length);
862
863        return (ret != 0) ? null : sb.ToString();
864    }
865
866    [Flags]
867    internal enum AssocF : uint
868    {
869        IsProtocol = 0x1000,
870    }
871
872    internal enum AssocStr
873    {
874        Command = 1,
875        Executable,
876    }
877}
878"@
879
880Add-Type -TypeDefinition $Signature
881
882Write-Output $([Win32Api]::GetDefaultBrowser())
883"#;

984    /// Sets `$BROWSER` to `browser_env_tmpl` (with `{}` in it replaced by the path of a script
985    /// that records the argv it receives), opens `url` with it, and returns the recorded argv
986    fn record_argv(name: &str, browser_env_tmpl: &str, url: &str) -> Vec<String> {
987        let flag_path = get_temp_path(name, "flag");
988        let _ = std::fs::remove_file(&flag_path);
989
990        // create a browser script which records each arg it got on its own line
991        let browser_path = get_temp_path(name, "browser");
992        {
993            let mut browser_file =
994                File::create(&browser_path).expect("failed to create browser file");
995            browser_file
996                .write_fmt(format_args!(
997                    r#"#!/bin/sh
998for arg in "$@"; do echo "$arg"; done > "{flag_path}"
999"#
1000                ))
1001                .expect("failed to write browser file");
1002            let mut perms = browser_file
1003                .metadata()
1004                .expect("failed to get permissions")
1005                .permissions();
1006            perms.set_mode(0o755);
1007            let _ = browser_file.set_permissions(perms);
1008        }
1009
1010        // we go through $BROWSER (instead of invoking the browser directly), as that's the
1011        // path on which the url gets assembled into the command line
1012        let browser_env = browser_env_tmpl.replace("{}", &browser_path);
1013        let prev_browser_env = std::env::var_os("BROWSER");
1014        std::env::set_var("BROWSER", &browser_env);
1015        let result = try_with_browser_env(url, &BrowserOptions::default());
1016        match prev_browser_env {
1017            Some(prev) => std::env::set_var("BROWSER", prev),
1018            None => std::env::remove_var("BROWSER"),
1019        }
1020        assert!(result.is_ok(), "failed to run {browser_env:?}: {result:?}");
1021
1022        // the script runs in the background, so wait for it to write the flag file
1023        let mut recorded: Option<String> = None;
1024        for _ in 0..20 {
1025            if let Ok(contents) = std::fs::read_to_string(&flag_path) {
1026                recorded = Some(contents);
1027                break;
1028            }
1029            std::thread::sleep(std::time::Duration::from_millis(250));
1030        }
1031
1032        let _ = std::fs::remove_file(&flag_path);
1033        let _ = std::fs::remove_file(&browser_path);
1034
1035        recorded
1036            .expect("flag file not found")
1037            .lines()
1038            .map(|l| l.to_owned())
1039            .collect()
1040    }
