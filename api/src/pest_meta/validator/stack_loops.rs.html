10//! Repetitions that never end because of the stack.
11//!
12//! `is_non_progressing` and `is_non_failing` treat the stack builtins as progressing, which
13//! misses for example `POP_ALL ~ PEEK_ALL*` (`PEEK_ALL` matches nothing on an empty stack) and
14//! `(PUSH("") ~ POP)*`. This pass tracks what is known about the stack through each rule and
15//! reports a repetition whose body, from every stack that can reach it, succeeds without
16//! consuming input and within a few iterations leaves the stack as it found it.
17//!
18//! It only reports what is certain, so a grammar that can terminate is never rejected:
19//! - every rule is analysed as if entered with an unknown stack, so a bare `PEEK_ALL*` is not
20//!   reported (it terminates when the stack holds a non-empty string);
21//! - each expression yields the stack after it succeeds and the stack after it fails, each
22//!   possibly "cannot happen": a repetition is only analysed from stacks that can reach it,
23//!   and a choice's right side from the stack its left side fails with (restored when the
24//!   optimizer wraps the branch in `RestoreOnErr`; a bare failed `POP_ALL` keeps what it
25//!   removed);
26//! - bounded repetitions are followed as the optimizer unrolls them, copy by copy, never past
27//!   their bound (the stacks repeat with some period, so huge bounds are cheap);
28//! - a sequence or a repetition is not known to consume nothing when `WHITESPACE` or
29//!   `COMMENT` is defined, since implicit whitespace may run between its parts;
30//! - if `WHITESPACE` or `COMMENT` can change the stack, nothing is reported.
31//!
32//! Results are cached per expression node and stack, so each node is analysed once per stack.
33

42/// The part of the stack below the entries an [`AbsStack`] tracks.
43#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
44enum StackBase {
45    /// No entries.
46    Empty,
47    /// Only empty-string entries, possibly none.
48    Blank,
49    /// Anything.
50    Any,
51}
52
53/// What is known about the stack at a point of a rule: an unknown `base` with `len` entries
54/// pushed on it, each known to be an empty string or not.
55///
56/// Empty strings matter because they are the only entries that `PEEK`, `POP`, `PEEK_ALL` and
57/// `POP_ALL` can match without consuming input; the other entries are tracked so that `DROP`,
58/// `POP` and `PEEK` are known to find something to remove or match.
59#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
60struct AbsStack {
61    base: StackBase,
62    /// Entries pushed on `base`.
63    len: u8,
64    /// Bit `i` set: entry `i` (from the bottom of the pushed ones) is certainly an empty string.
65    blank: u8,
66    /// `base` still holds what it held when the current repetition iteration started.
67    base_intact: bool,
68}
69
70impl AbsStack {
71    /// Most entries tracked on top of the base; beyond that the state is given up.
72    const MAX_LEN: u8 = 8;
73
74    const UNKNOWN: AbsStack = AbsStack {
75        base: StackBase::Any,
76        len: 0,
77        blank: 0,
78        base_intact: false,
79    };
80
81    fn mask(len: u8) -> u8 {
82        ((1u16 << len) - 1) as u8
83    }
84
85    fn join(self, other: AbsStack) -> AbsStack {
86        if self == other {
87            return self;
88        }
89        let base = match (self.base, other.base) {
90            (a, b) if a == b => a,
91            (StackBase::Any, _) | (_, StackBase::Any) => StackBase::Any,
92            // one base empty and the other all blank (in either order): all blank
93            _ => StackBase::Blank,
94        };
95        if self.len == other.len {
96            AbsStack {
97                base,
98                len: self.len,
99                // an entry is known blank only if it is in both
100                blank: self.blank & other.blank,
101                base_intact: self.base_intact && other.base_intact,
102            }
103        } else if base != StackBase::Any && self.all_blank() && other.all_blank() {
104            // different numbers of entries, all empty strings
105            AbsStack {
106                base: StackBase::Blank,
107                len: 0,
108                blank: 0,
109                base_intact: false,
110            }
111        } else {
112            AbsStack::UNKNOWN
113        }
114    }
115
116    /// The state after pushing an entry that is certainly an empty string (`blank`) or not.
117    fn push(self, blank: bool) -> AbsStack {
118        if self.len < AbsStack::MAX_LEN {
119            AbsStack {
120                len: self.len + 1,
121                blank: self.blank | (u8::from(blank) << self.len),
122                ..self
123            }
124        } else {
125            AbsStack::UNKNOWN
126        }
127    }
128
129    /// Every entry is an empty string.
130    fn all_blank(self) -> bool {
131        self.base != StackBase::Any && self.blank == AbsStack::mask(self.len)
132    }
133
134    /// Certainly no entries.
135    fn is_empty(self) -> bool {
136        self.base == StackBase::Empty && self.len == 0
137    }
138
139    /// Certainly at least one entry.
140    fn non_empty(self) -> bool {
141        self.len > 0
142    }
143
144    /// The top entry is certainly an empty string.
145    fn top_blank(self) -> bool {
146        self.len > 0 && self.blank >> (self.len - 1) & 1 == 1
147    }
148
149    /// The state after `POP_ALL` succeeds.
150    fn popped_all(self) -> AbsStack {
151        AbsStack {
152            base: StackBase::Empty,
153            len: 0,
154            blank: 0,
155            base_intact: self.base_intact && self.base == StackBase::Empty,
156        }
157    }
158
159    /// The state after one entry is removed (`POP`, `DROP`).
160    fn popped_one(self) -> AbsStack {
161        if self.len > 0 {
162            AbsStack {
163                len: self.len - 1,
164                blank: self.blank & AbsStack::mask(self.len - 1),
165                ..self
166            }
167        } else {
168            match self.base {
169                StackBase::Empty => self,
170                StackBase::Blank => AbsStack {
171                    base: StackBase::Blank,
172                    len: 0,
173                    blank: 0,
174                    base_intact: false,
175                },
176                StackBase::Any => AbsStack::UNKNOWN,
177            }
178        }
179    }
180
181    /// The same pushed entries as `other` (the base is compared through `base_intact`).
182    fn same_top(self, other: AbsStack) -> bool {
183        self.len == other.len && self.blank == other.blank
184    }
185}
186
187/// A possibly impossible state (`None`: that outcome cannot happen).
188type Maybe = Option<AbsStack>;
189
190/// Joins two possibly impossible states.
191fn join(a: Maybe, b: Maybe) -> Maybe {
192    match (a, b) {
193        (Some(a), Some(b)) => Some(a.join(b)),
194        (a, None) => a,
195        (None, b) => b,
196    }
197}

199/// What an expression does when entered with a given stack.
200#[derive(Clone, Copy, Debug, PartialEq, Eq)]
201struct Outcome {
202    /// The stack after it succeeds, or `None` if it never succeeds.
203    ok: Maybe,
204    /// The stack after it fails, or `None` if it never fails.
205    err: Maybe,
206    /// When it succeeds it consumes no input, whatever the input (only meaningful with `ok`).
207    empty: bool,
208}
209
210impl Outcome {
211    /// Nothing is known except the stack it was entered with, which it may have changed.
212    const UNKNOWN: Outcome = Outcome {
213        ok: Some(AbsStack::UNKNOWN),
214        err: Some(AbsStack::UNKNOWN),
215        empty: false,
216    };
217
218    fn succeeds(s: AbsStack, empty: bool) -> Outcome {
219        Outcome {
220            ok: Some(s),
221            err: None,
222            empty,
223        }
224    }
225
226    fn fails(s: AbsStack) -> Outcome {
227        Outcome {
228            ok: None,
229            err: Some(s),
230            empty: false,
231        }
232    }
233
234    /// May succeed (consuming input, or not) or fail, leaving the stack as `s`.
235    fn either(s: AbsStack) -> Outcome {
236        Outcome {
237            ok: Some(s),
238            err: Some(s),
239            empty: false,
240        }
241    }
242
243    /// Certainly succeeds without consuming input, whatever the input: `Some(stack after)`.
244    fn exact(self) -> Maybe {
245        if self.err.is_none() && self.empty {
246            self.ok
247        } else {
248            None
249        }
250    }
251}

253pub(super) fn validate_stack_repetition<'a, 'i: 'a>(
254    rules: &'a [ParserRule<'i>],
255) -> Vec<Error<Rule>> {
256    let map = to_hash_map(rules);
257
258    // Implicit whitespace runs between the steps of non-atomic rules. If it can change the
259    // stack, iterations that look unchanged here might not be, so report nothing.
260    let mut seen = HashSet::new();
261    if ["WHITESPACE", "COMMENT"]
262        .iter()
263        .any(|name| rule_modifies_stack(name, &map, &mut seen))
264    {
265        return vec![];
266    }
267
268    let mut analysis = StackAnalysis {
269        rules: &map,
270        skips: map.contains_key("WHITESPACE") || map.contains_key("COMMENT"),
271        memo: HashMap::new(),
272        active: HashSet::new(),
273        rule_memo: HashMap::new(),
274        rule_active: HashSet::new(),
275        errors: vec![],
276        reported: HashSet::new(),
277        walking: false,
278        walked: HashSet::new(),
279    };
280    let entry = AbsStack {
281        base: StackBase::Any,
282        len: 0,
283        blank: 0,
284        base_intact: true,
285    };
286    for rule in rules {
287        analysis.walk(&rule.node, entry);
288    }
289    analysis.errors
290}

292fn rule_modifies_stack(
293    name: &str,
294    rules: &HashMap<String, &ParserNode<'_>>,
295    seen: &mut HashSet<String>,
296) -> bool {
297    match name {
298        "POP" | "DROP" | "POP_ALL" => true,
299        _ => match rules.get(name) {
300            Some(node) if seen.insert(name.to_owned()) => {
301                expr_modifies_stack(&node.expr, rules, seen)
302            }
303            _ => false,
304        },
305    }
306}
307
308fn expr_modifies_stack(
309    expr: &ParserExpr<'_>,
310    rules: &HashMap<String, &ParserNode<'_>>,
311    seen: &mut HashSet<String>,
312) -> bool {
313    match expr {
314        ParserExpr::Push(_) => true,
315        #[cfg(feature = "grammar-extras")]
316        ParserExpr::PushLiteral(_) => true,
317        ParserExpr::Ident(name) => rule_modifies_stack(name, rules, seen),
318        ParserExpr::Str(_)
319        | ParserExpr::Insens(_)
320        | ParserExpr::Range(_, _)
321        | ParserExpr::PeekSlice(_, _) => false,
322        // Predicates restore the stack.
323        ParserExpr::PosPred(_) | ParserExpr::NegPred(_) => false,
324        ParserExpr::Seq(lhs, rhs) | ParserExpr::Choice(lhs, rhs) => {
325            expr_modifies_stack(&lhs.expr, rules, seen)
326                || expr_modifies_stack(&rhs.expr, rules, seen)
327        }
328        ParserExpr::Opt(inner)
329        | ParserExpr::Rep(inner)
330        | ParserExpr::RepOnce(inner)
331        | ParserExpr::RepExact(inner, _)
332        | ParserExpr::RepMin(inner, _)
333        | ParserExpr::RepMax(inner, _)
334        | ParserExpr::RepMinMax(inner, _, _) => expr_modifies_stack(&inner.expr, rules, seen),
335        #[cfg(feature = "grammar-extras")]
336        ParserExpr::NodeTag(inner, _) => expr_modifies_stack(&inner.expr, rules, seen),
337    }
338}

340/// A reference compared and hashed by address: one node of the grammar tree, not every node
341/// equal to it. Equal subexpressions in different places are different nodes (each is reported
342/// at its own span), and hashing by address costs O(1) where hashing the subtree would not.
343struct ByAddr<'a, T>(&'a T);
344
345impl<T> Clone for ByAddr<'_, T> {
346    fn clone(&self) -> Self {
347        *self
348    }
349}
350
351impl<T> Copy for ByAddr<'_, T> {}
352
353impl<T> PartialEq for ByAddr<'_, T> {
354    fn eq(&self, other: &Self) -> bool {
355        std::ptr::eq(self.0, other.0)
356    }
357}
358
359impl<T> Eq for ByAddr<'_, T> {}
360
361impl<T> Hash for ByAddr<'_, T> {
362    fn hash<H: Hasher>(&self, state: &mut H) {
363        std::ptr::hash(self.0, state)
364    }
365}

367struct StackAnalysis<'a, 'i> {
368    rules: &'a HashMap<String, &'a ParserNode<'i>>,
369    /// `WHITESPACE` or `COMMENT` is defined, so implicit whitespace may be consumed between
370    /// the operands of a sequence (in non-atomic rules, which are not tracked).
371    skips: bool,
372    /// Outcome of an expression node (by address) entered with a stack.
373    memo: HashMap<(ByAddr<'a, ParserExpr<'i>>, AbsStack), Outcome>,
374    active: HashSet<(ByAddr<'a, ParserExpr<'i>>, AbsStack)>,
375    /// Outcome of a rule entered with a stack.
376    rule_memo: HashMap<(String, AbsStack), Outcome>,
377    rule_active: HashSet<(String, AbsStack)>,
378    errors: Vec<Error<Rule>>,
379    /// Repetition nodes already reported, so one is not reported twice.
380    reported: HashSet<ByAddr<'a, ParserNode<'i>>>,
381    /// Inside `walk`: repetitions reached are checked and reported.
382    walking: bool,
383    /// Nodes already walked with a stack: what they report depends only on the node and the
384    /// stack, and is already in `errors`, so they are not walked again.
385    walked: HashSet<(ByAddr<'a, ParserExpr<'i>>, AbsStack)>,
386}
387
388impl<'a, 'i> StackAnalysis<'a, 'i> {
389    /// Analyses a rule's body entered with `s`, reporting the looping repetitions reached.
390    fn walk(&mut self, node: &'a ParserNode<'i>, s: AbsStack) {
391        self.walking = true;
392        let _ = self.visit(node, s);
393        self.walking = false;
394    }
395
396    /// The outcome of `node` entered with `s`; while walking, also reports the looping
397    /// repetitions it reaches. Each node is walked once per stack.
398    fn visit(&mut self, node: &'a ParserNode<'i>, s: AbsStack) -> Outcome {
399        if self.walking && self.walked.insert((ByAddr(&node.expr), s)) {
400            self.step(node, s)
401        } else {
402            self.outcome(node, s)
403        }
404    }
405
406    /// The outcome of `node` entered with `s`, cached; never reports.
407    fn outcome(&mut self, node: &'a ParserNode<'i>, s: AbsStack) -> Outcome {
408        let key = (ByAddr(&node.expr), s);
409        if let Some(&o) = self.memo.get(&key) {
410            return o;
411        }
412        if !self.active.insert(key) {
413            return Outcome::UNKNOWN;
414        }
415        let walking = std::mem::replace(&mut self.walking, false);
416        let o = self.step(node, s);
417        self.walking = walking;
418        let _ = self.active.remove(&key);
419        let _ = self.memo.insert(key, o);
420        o
421    }
422
423    fn step(&mut self, node: &'a ParserNode<'i>, s: AbsStack) -> Outcome {
424        match &node.expr {
425            ParserExpr::Str(string) | ParserExpr::Insens(string) => {
426                if string.is_empty() {
427                    Outcome::succeeds(s, true)
428                } else {
429                    Outcome::either(s)
430                }
431            }
432            ParserExpr::Range(_, _) => Outcome::either(s),
433            ParserExpr::Ident(name) => self.ident(name, s),
434            // `PEEK[..]` is `PEEK_ALL` matched bottom to top. `PEEK[0..0]` is always in range
435            // (0 is never past the end) and matches nothing. Other slices can be out of range
436            // or consume input.
437            ParserExpr::PeekSlice(0, None) => self.ident("PEEK_ALL", s),
438            ParserExpr::PeekSlice(0, Some(0)) => Outcome::succeeds(s, true),
439            ParserExpr::PeekSlice(_, _) => Outcome::either(s),
440            ParserExpr::PosPred(inner) => {
441                let o = self.visit(inner, s);
442                Outcome {
443                    ok: o.ok.map(|_| s),
444                    err: o.err.map(|_| s),
445                    empty: true,
446                }
447            }
448            ParserExpr::NegPred(inner) => {
449                let o = self.visit(inner, s);
450                Outcome {
451                    ok: o.err.map(|_| s),
452                    err: o.ok.map(|_| s),
453                    empty: true,
454                }
455            }
456            ParserExpr::Seq(lhs, rhs) => {
457                let l = self.visit(lhs, s);
458                let r = match l.ok {
459                    Some(mid) => self.visit(rhs, mid),
460                    None => Outcome {
461                        ok: None,
462                        err: None,
463                        empty: true,
464                    },
465                };
466                Outcome {
467                    ok: r.ok,
468                    // a failed sequence restores the stack (`state.sequence`)
469                    err: if l.err.is_some() || r.err.is_some() {
470                        Some(s)
471                    } else {
472                        None
473                    },
474                    // implicit whitespace may be consumed between the operands
475                    empty: l.empty && r.empty && !self.skips,
476                }
477            }
478            ParserExpr::Choice(lhs, rhs) => {
479                let l = self.visit(lhs, s);
480                // `rhs` is only tried when `lhs` fails, from the stack `lhs` failed with.
481                let r = match self.failed(l.err, s) {
482                    Some(failed) => self.visit(rhs, failed),
483                    None => Outcome {
484                        ok: None,
485                        err: None,
486                        empty: true,
487                    },
488                };
489                Outcome {
490                    ok: join(l.ok, r.ok),
491                    err: self.failed(r.err, s),
492                    empty: (l.ok.is_none() || l.empty) && (r.ok.is_none() || r.empty),
493                }
494            }
495            ParserExpr::Opt(inner) => {
496                let o = self.visit(inner, s);
497                Outcome {
498                    ok: join(o.ok, self.failed(o.err, s)),
499                    err: None,
500                    empty: o.ok.is_none() || o.empty,
501                }
502            }
503            ParserExpr::Rep(inner) => self.repeat(node, inner, s, 0, None),
504            ParserExpr::RepOnce(inner) => self.repeat(node, inner, s, 1, None),
505            ParserExpr::RepMin(inner, min) => self.repeat(node, inner, s, *min, None),
506            ParserExpr::RepExact(inner, n) => self.repeat(node, inner, s, *n, Some(*n)),
507            ParserExpr::RepMax(inner, max) => self.repeat(node, inner, s, 0, Some(*max)),
508            // `{min, max}` with `min > max` is accepted, and the optimizer then builds `max`
509            // required copies (`unroller.rs`)
510            ParserExpr::RepMinMax(inner, min, max) => {
511                self.repeat(node, inner, s, (*min).min(*max), Some(*max))
512            }
513            ParserExpr::Push(inner) => {
514                let o = self.visit(inner, s);
515                Outcome {
516                    // pushes what it matched: an empty string if it certainly matched nothing
517                    ok: o.ok.map(|after| after.push(o.empty)),
518                    err: o.err,
519                    empty: o.empty,
520                }
521            }
522            #[cfg(feature = "grammar-extras")]
523            ParserExpr::PushLiteral(string) => Outcome::succeeds(s.push(string.is_empty()), true),
524            #[cfg(feature = "grammar-extras")]
525            ParserExpr::NodeTag(inner, _) => self.visit(inner, s),
526        }
527    }
528
529    fn ident(&mut self, name: &str, s: AbsStack) -> Outcome {
530        match name {
531            // matches every entry, top to bottom; nothing to match when all are empty strings
532            "PEEK_ALL" => {
533                if s.all_blank() {
534                    Outcome::succeeds(s, true)
535                } else {
536                    Outcome::either(s)
537                }
538            }
539            "POP_ALL" => {
540                if s.all_blank() {
541                    Outcome::succeeds(s.popped_all(), true)
542                } else {
543                    // A failed POP_ALL keeps the entries it removed before the mismatch
544                    // (it is not restored), so after a failure only "unknown" is safe.
545                    Outcome {
546                        ok: Some(s.popped_all()),
547                        err: Some(AbsStack::UNKNOWN),
548                        empty: false,
549                    }
550                }
551            }
552            // On an empty stack POP and PEEK panic and DROP fails: none of them succeeds.
553            "PEEK" | "POP" => {
554                if s.is_empty() {
555                    Outcome {
556                        ok: None,
557                        err: None,
558                        empty: true,
559                    }
560                } else {
561                    let after = if name == "POP" { s.popped_one() } else { s };
562                    if s.top_blank() {
563                        Outcome::succeeds(after, true)
564                    } else {
565                        // matches the top entry: may consume input or fail
566                        Outcome {
567                            ok: Some(after),
568                            err: Some(after),
569                            empty: false,
570                        }
571                    }
572                }
573            }
574            "DROP" => {
575                if s.is_empty() {
576                    Outcome::fails(s)
577                } else if s.non_empty() {
578                    Outcome::succeeds(s.popped_one(), true)
579                } else {
580                    Outcome {
581                        ok: Some(s.popped_one()),
582                        err: Some(s),
583                        empty: true,
584                    }
585                }
586            }
587            _ => match self.rules.get(name) {
588                Some(&node) => self.rule(name, node, s),
589                // other builtins (ANY, SOI, EOI, ...) do not touch the stack
590                None => Outcome::either(s),
591            },
592        }
593    }
594
595    fn rule(&mut self, name: &str, node: &'a ParserNode<'i>, s: AbsStack) -> Outcome {
596        let key = (name.to_owned(), s);
597        if let Some(&o) = self.rule_memo.get(&key) {
598            return o;
599        }
600        if !self.rule_active.insert(key.clone()) {
601            // recursion: give up
602            return Outcome::UNKNOWN;
603        }
604        let walking = std::mem::replace(&mut self.walking, false);
605        let o = self.outcome(node, s);
606        self.walking = walking;
607        let _ = self.rule_active.remove(&key);
608        let _ = self.rule_memo.insert(key, o);
609        o
610    }
611
612    /// The stack after `inner` failed with `err` (`s`: the stack it started with), as the
613    /// alternative or optional around it sees it. The optimizer wraps some branches in
614    /// `RestoreOnErr`, which restores `s` on failure, and leaves others as they are, which
615    /// keep what a failed `POP_ALL` removed. Which branches are wrapped depends on the
616    /// optimized shape of the grammar (choices are rotated, `PUSH_LITERAL` doesn't count), so
617    /// both stacks are taken as possible.
618    fn failed(&mut self, err: Maybe, s: AbsStack) -> Maybe {
619        err.map(|err| err.join(s))
620    }
621
622    /// A repetition of `inner` between `min` and `max` times (unbounded if `None`), entered
623    /// with `s`, as the optimizer unrolls it (`unroller.rs`): `min` copies of `inner` in
624    /// sequence, then `inner*` if unbounded, or `max - min` copies of `inner?`. While walking,
625    /// the body is visited from every stack an iteration can start with, and an unbounded
626    /// repetition that loops forever is reported.
627    fn repeat(
628        &mut self,
629        node: &'a ParserNode<'i>,
630        inner: &'a ParserNode<'i>,
631        s: AbsStack,
632        min: u32,
633        max: Option<u32>,
634    ) -> Outcome {
635        if max.is_none() && self.walking && self.loops_forever(inner, s) {
636            self.report(node, inner);
637        }
638        // implicit whitespace may be consumed between iterations
639        let mut empty = !self.skips || max.is_some_and(|m| m <= 1);
640        // the required copies: one failing fails the whole sequence, which restores the stack
641        let (mid, fails) = self.steps(inner, s, u64::from(min), true, &mut empty);
642        let err = fails.then_some(s);
643        let Some(mid) = mid else {
644            return Outcome {
645                ok: None,
646                err,
647                empty: true,
648            };
649        };
650        let ok = match max {
651            Some(m) => {
652                let optional = u64::from(m.saturating_sub(min));
653                self.steps(inner, mid, optional, false, &mut empty).0
654            }
655            None => self.greedy(inner, mid, &mut empty),
656        };
657        Outcome { ok, err, empty }
658    }
659
660    /// `count` copies of `inner` from `s`, each starting where the previous one ended:
661    /// `required` ones in sequence (each must succeed), or optional ones (`inner?`, which go
662    /// on from the stack a failure leaves). Returns the stack after the last copy (`None` if a
663    /// required copy cannot succeed) and whether a required copy can fail.
664    ///
665    /// The stack a copy starts with is a function of the stack the previous one started with,
666    /// over a finite set of stacks, so the stacks repeat with some period: they are followed
667    /// until one repeats, never past `count`, and the stack after `count` copies is read off
668    /// the period. So huge bounds cost no more than small ones.
669    fn steps(
670        &mut self,
671        inner: &'a ParserNode<'i>,
672        s: AbsStack,
673        count: u64,
674        required: bool,
675        empty: &mut bool,
676    ) -> (Maybe, bool) {
677        let mut trace: Vec<AbsStack> = vec![];
678        let mut seen: HashMap<AbsStack, usize> = HashMap::new();
679        let mut at = s;
680        let mut fails = false;
681        let mut t: u64 = 0;
682        while t < count {
683            if let Some(&j) = seen.get(&at) {
684                // the stack before copy t is the one before copy j: from here they repeat
685                // with period t - j, and every stack of the period has been visited already
686                let period = t - j as u64;
687                let end = trace[j + ((count - t) % period) as usize];
688                return (Some(end), fails);
689            }
690            let _ = seen.insert(at, trace.len());
691            trace.push(at);
692            let o = self.visit(inner, at);
693            if o.ok.is_some() {
694                *empty &= o.empty;
695            }
696            t += 1;
697            let next = if required {
698                fails |= o.err.is_some();
699                o.ok
700            } else {
701                join(o.ok, self.failed(o.err, at))
702            };
703            match next {
704                Some(n) => at = n,
705                None => return (None, fails),
706            }
707        }
708        (Some(at), fails)
709    }
710
711    /// `inner*` from `s`: copies run until one fails, which ends the repetition. Returns every
712    /// stack it can end with (`None` if no copy can fail: it never ends).
713    fn greedy(&mut self, inner: &'a ParserNode<'i>, s: AbsStack, empty: &mut bool) -> Maybe {
714        let mut seen = HashSet::new();
715        let mut ok = None;
716        let mut at = s;
717        let mut first = true;
718        while seen.insert(at) {
719            let o = self.visit(inner, at);
720            if o.ok.is_some() {
721                *empty &= o.empty;
722            }
723            if o.err.is_some() {
724                // The first copy runs in an optional, the later ones in a sequence that
725                // restores the stack (in non-atomic rules), unless wrapped in RestoreOnErr.
726                let restored = if first { None } else { Some(at) };
727                ok = join(ok, join(self.failed(o.err, at), restored));
728            }
729            first = false;
730            match o.ok {
731                Some(n) => at = n,
732                None => break,
733            }
734        }
735        ok
736    }
737
738    /// A repetition of `body` entered with stack `s` never ends: from every stack `s` allows,
739    /// `body` succeeds without consuming input, and the iterations come back to a stack an
740    /// earlier iteration started with (after any number of steps), so they cycle forever.
741    ///
742    /// Pushed entries that can be non-blank are ones present before the cycle (an iteration
743    /// that consumes nothing only pushes empty strings), so the same entries with the base
744    /// untouched since the first of those iterations is the same concrete stack.
745    fn loops_forever(&mut self, body: &'a ParserNode<'i>, s: AbsStack) -> bool {
746        // each step lands in a finite set of states; this is more than enough to see a repeat
747        const MAX_STEPS: usize = 64;
748        let fresh = |st: AbsStack| AbsStack {
749            base_intact: true,
750            ..st
751        };
752        let mut cur = fresh(s);
753        let mut seen = vec![cur];
754        for _ in 0..MAX_STEPS {
755            let Some(out) = self.outcome(body, cur).exact() else {
756                return false;
757            };
758            if out.base_intact {
759                if seen.iter().any(|earlier| out.same_top(*earlier)) {
760                    return true;
761                }
762            } else {
763                // the base changed: look for a cycle from here on
764                seen.clear();
765            }
766            cur = fresh(out);
767            seen.push(cur);
768        }
769        false
770    }
771
772    fn report(&mut self, node: &'a ParserNode<'i>, inner: &'a ParserNode<'i>) {
773        // the existing check already reports these
774        if is_non_failing(&inner.expr, self.rules, &mut vec![])
775            || is_non_progressing(&inner.expr, self.rules, &mut vec![])
776        {
777            return;
778        }
779        if self.reported.insert(ByAddr(node)) {
780            self.errors.push(Error::new_from_span(
781                ErrorVariant::CustomError {
782                    message: "expression inside repetition is non-progressing and will repeat \
783                              infinitely"
784                        .to_owned(),
785                },
786                node.span,
787            ));
788        }
789    }

894            "a = { POP_ALL ~ PEEK[0..1]* }",
895            // the body consumes input
896            "a = { POP_ALL ~ (PEEK_ALL ~ \"x\")* }",
897            "a = { (PUSH(\"x\") ~ POP)* }",
898            // a negative predicate may fail
899            "a = { POP_ALL ~ (!\"x\" ~ PEEK_ALL)* }",
900            // POP or PEEK on an empty stack panic rather than loop
901            "a = { POP_ALL ~ PEEK* }",
902            // the repetition is never reached: the first alternative always succeeds
903            "a = { POP_ALL ~ (PEEK_ALL | PEEK_ALL*) }",
904            "WHITESPACE = _{ \" \" } a = { POP_ALL ~ ((PEEK_ALL ~ PEEK_ALL) | PEEK_ALL*) }",
905            // the repetition is never reached: DROP fails on the empty stack
906            "a = { POP_ALL ~ (DROP ~ PEEK_ALL*)? }",
907            "a = { POP_ALL ~ (DROP ~ PUSH(\"\") ~ PEEK*)* }",
908            "a = { POP_ALL ~ PUSH(\"a\") ~ DROP ~ DROP ~ POP_ALL ~ PEEK_ALL* }",
909            // the only iteration allowed fails at DROP and takes the second alternative
910            "a = { POP_ALL ~ ((DROP ~ PUSH(\"\") ~ PEEK*) | PUSH(\"\")){,1} }",
911            // the ninth DROP always fails, so PEEK_ALL* is never reached
912            "a = { (POP_ALL ~ PUSH(\"\"){8} ~ DROP{9} ~ PEEK_ALL*)? }",
913            // eight iterations take the first alternative, the ninth pushes a blank: there is no
914            // tenth iteration to run PEEK_ALL* on it
915            "a = { POP_ALL ~ PUSH(\"x\"){8} ~ ((&DROP ~ PEEK_ALL* ~ DROP) | PUSH(\"\")){9} }",
916            // the same with `{10,9}`: the optimizer runs nine copies, not ten
917            "a = { POP_ALL ~ PUSH(\"x\"){8} ~ ((&DROP ~ PEEK_ALL* ~ DROP) | PUSH(\"\")){10,9} }",
918            // a failed POP_ALL keeps the entries it removed: after PUSH(\"y\") ~ PUSH(\"x\"),
919            // on \"yxy!\" it removes the blank and \"x\", then PEEK* matches \"y\" once
920            "a = { PUSH(\"\") ~ (POP_ALL | PEEK*) } b = { PUSH(\"y\") ~ PUSH(\"x\") ~ a }",
921            // a failed POP in a choice or an optional is restored: \"x\" stays, PEEK_ALL* stops
922            "a = { POP_ALL ~ PUSH(\"x\") ~ (POP | PEEK_ALL*) }",
923            "a = { POP_ALL ~ PUSH(\"x\") ~ POP? ~ PEEK_ALL* }",
924            // the optimizer rotates choices, so only the bare `POP_ALL` branch is unwrapped: its
925            // failure leaves the stack empty and the first DROP stops the last branch
926            "a = { POP_ALL ~ PUSH(\"x\") ~ PUSH(\"\") ~ (!PUSH(\"\") | POP_ALL | (DROP ~ DROP ~ PEEK_ALL*)) }",

928            "WHITESPACE = _{ \" \" } a = { PUSH(\"\"{2}) ~ PEEK* }",
929            "WHITESPACE = _{ \" \" } b = !{ PEEK_ALL{2} } a = @{ POP_ALL ~ PUSH(b) ~ PEEK_ALL* }",
930            // implicit whitespace inside the PUSH is captured, so POP fails at the end
931            "WHITESPACE = _{ \" \" } a = { (PUSH(\"\" ~ \"\") ~ POP)* }",
932            // the same through a non-atomic rule called from an atomic one
933            "WHITESPACE = _{ \" \" } b = !{ \"\" ~ \"\" } a = @{ POP_ALL ~ PUSH(b) ~ PEEK_ALL* }",
934        ] {
935            assert_eq!(stack_loop_errors(input), Vec::<String>::new(), "{input}");
