12#[derive(Clone, Copy)]
13pub struct WithHelper;
14
15impl HelperDef for WithHelper {
16    fn call<'reg: 'rc, 'rc>(
17        &self,
18        h: &Helper<'rc>,
19        r: &'reg Registry<'reg>,
20        ctx: &'rc Context,
21        rc: &mut RenderContext<'reg, 'rc>,
22        out: &mut dyn Output,
23    ) -> HelperResult {
24        let param = h
25            .param(0)
26            .ok_or(RenderErrorReason::ParamNotFoundForIndex("with", 0))?;
27
28        if param.value().is_truthy(false) {
29            let mut block = create_block(param);
30
31            // inherit local variables (@key/@index/@first/@last) from the
32            // enclosing block so they remain accessible inside `#with`,
33            // matching Handlebars.js (see issue #592).
34            if let Some(parent) = rc.block() {
35                block
36                    .local_variables_mut()
37                    .clone_from(parent.local_variables());
38            }
39
40            if let Some(block_param) = h.block_param() {
41                let mut params = BlockParams::new();
42                if param.context_path().is_some() {
43                    params.add_path(block_param, Vec::with_capacity(0))?;
44                } else {
45                    params.add_value(block_param, param.value().clone())?;
46                }
47
48                block.set_block_params(params);
49            }
50
51            rc.push_block(block);
52
53            if let Some(t) = h.template() {
54                t.render(r, ctx, rc, out)?;
55            };
56
57            rc.pop_block();
58            Ok(())
59        } else if let Some(t) = h.inverse() {
60            t.render(r, ctx, rc, out)
61        } else if r.strict_mode() {
62            Err(RenderError::strict_error(param.relative_path()))
63        } else {
64            Ok(())
65        }
66    }
67}

90    #[test]
91    fn test_with() {
92        let person = Person {
93            name: "Ning Sun".to_string(),
94            age: 27,
95            addr: Address {
96                city: "Beijing".to_string(),
97                country: "China".to_string(),
98            },
99            titles: vec!["programmer".to_string(), "cartographier".to_string()],
100        };
101
102        let mut handlebars = Registry::new();
103        handlebars.register("t0", "{{#with addr}}{{city}}{{/with}}");
104        handlebars.register("t1", "{{#with notfound}}hello{{else}}world{{/with}}");
105        handlebars.register("t2", "{{#with addr/country}}{{this}}{{/with}}");
106
107        handlebars.assert_render("t0", &person, "Beijing");
108        handlebars.assert_render("t1", &person, "world");
109        handlebars.assert_render("t2", &person, "China");
110    }

166    #[test]
167    fn test_with_in_each_inherits_key_and_index() {
168        // Regression for #592: `@key`/`@index` from an enclosing `{{#each}}`
169        // must remain accessible inside `{{#with}}`, matching Handlebars.js.
170        let mut handlebars = Registry::new();
171
172        handlebars.register(
173            "obj",
174            "{{#each this}}{{#with this}}{{@key}}={{this}}|{{/with}}{{/each}}",
175        );
176        handlebars.assert_render("obj", &json!({"ftp": 21, "http": 80}), "ftp=21|http=80|");
177
178        handlebars.register(
179            "arr",
180            "{{#each this}}{{#with this}}{{@index}}={{this}}|{{/with}}{{/each}}",
181        );
182        handlebars.assert_render("arr", &json!([10, 20]), "0=10|1=20|");
183    }
