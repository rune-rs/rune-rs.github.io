1//! The syntax tree the compiler parses a source into, flattened into the
2//! order it is walked in, for tools which want to show it.
3//!
4//! This is not a stable interface. It exposes the parser's own node kinds,
5//! which change whenever the grammar does.
6
7use crate::alloc::Vec;
8use crate::ast::{Kind, Span};
9use crate::compile::{self, Options};
10use crate::SourceId;
11
12/// A node or token in a [`SyntaxTree`].
13#[derive(Debug, Clone, Copy, PartialEq, Eq)]
14#[non_exhaustive]
15pub struct Node {
16    /// How deeply the node is nested, where the root is zero.
17    pub depth: usize,
18    /// What the node is.
19    pub kind: Kind,
20    /// The byte span of source the node covers.
21    pub span: Span,
22    /// Whether this is a token rather than a node with children.
23    pub is_token: bool,
24}
25
26/// A parsed syntax tree.
27#[derive(Debug)]
28#[non_exhaustive]
29pub struct SyntaxTree {
30    /// Every node in the tree, parents before their children.
31    pub nodes: Vec<Node>,
32    /// The error parsing stopped at, if any, in which case `nodes` is the
33    /// tree as far as it got.
34    pub error: Option<compile::Error>,
35}
36
37/// Parse `source` the way the compiler does and flatten the result.
38///
39/// A source which does not parse still produces the tree up to where it
40/// stopped, alongside the error.
41///
42/// # Examples
43///
44/// ```
45/// let tree = rune::syntax_tree::parse("fn main() {}")?;
46/// assert!(tree.error.is_none());
47/// assert_eq!(tree.nodes[0].depth, 0);
48///
49/// let tree = rune::syntax_tree::parse("fn main() { a < b < c }")?;
50/// assert!(tree.error.is_some());
51/// assert!(!tree.nodes.is_empty());
52/// # Ok::<_, rune::compile::Error>(())
53/// ```
54pub fn parse(source: &str) -> compile::Result<SyntaxTree> {
55    let (tree, error) = crate::grammar::text(SourceId::EMPTY, source)
56        .max_nesting(Options::DEFAULT.max_depth)
57        .root_partial()?;
58
59    let mut nodes = Vec::new();
60
61    for (depth, kind, span, is_token) in tree.walk_with_depths() {
62        nodes.try_push(Node {
63            depth,
64            kind,
65            span,
66            is_token,
67        })?;
68    }
69
70    Ok(SyntaxTree { nodes, error })
71}
