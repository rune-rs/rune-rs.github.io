1//! Resolving intra-doc links.
2//!
3//! A link like ``[`Option::map`]`` in the documentation of an item is
4//! resolved the way the name would be if it was written in the module which
5//! holds that item: through the enclosing type, the module and the modules
6//! which hold it, the prelude, and finally the crates which are known. So
7//! `[u64]`, `[Vec::push]`, `[crate::option::Option]`, `[super::vec]` and
8//! `[::std::iter]` all link to what they name.
9
10use crate::alloc::prelude::*;
11use crate::alloc::{self, String, Vec};
12use crate::compile::Prelude;
13use crate::doc::context::{Context, Kind, Meta};
14use crate::item::ComponentRef;
15use crate::{Item, ItemBuf};

19/// The outcome of resolving a link.
20#[derive(Debug)]
21pub(crate) enum Link {
22    /// The link text does not look like a path, such as `[-1, 1]` written in
23    /// prose, so it was never meant to be a link and is left alone.
24    NotAPath,
25    /// The link looks like a path, but nothing by that name is in scope.
26    Unresolved,
27    /// The link resolved to the given target.
28    Found(Target),
29}
30
31/// What a link resolved to.
32#[derive(Debug)]
33pub(crate) struct Target {
34    /// The item which has a page of its own that the link points to.
35    pub(crate) item: ItemBuf,
36    /// The kind of page the item has.
37    pub(crate) kind: ItemKind,
38    /// The anchor on the page, like `method.push` or `variant.None`.
39    pub(crate) fragment: Option<String>,
40    /// What the link points to, for the title of the link.
41    pub(crate) what: &'static str,
42}
43
44/// Restricts what a link may resolve to, through a suffix like `!` or `()`
45/// or a prefix like `struct@`.
46#[derive(Debug, Clone, Copy)]
47enum Flavor {
48    Any,
49    Macro,
50    Callable,
51    Struct,
52    Enum,
53    Trait,
54    Type,
55    Module,
56    Function,
57    Const,
58    Variant,
59    Value,
60}
61
62impl Flavor {
63    fn from_disambiguator(s: &str) -> Option<Self> {
64        Some(match s {
65            "struct" => Flavor::Struct,
66            "enum" => Flavor::Enum,
67            "trait" => Flavor::Trait,
68            "type" | "prim" | "primitive" => Flavor::Type,
69            "mod" | "module" => Flavor::Module,
70            "macro" => Flavor::Macro,
71            "fn" | "function" | "method" => Flavor::Function,
72            "const" | "constant" => Flavor::Const,
73            "variant" => Flavor::Variant,
74            "value" => Flavor::Value,
75            _ => return None,
76        })
77    }
78
79    fn accepts(self, kind: &Kind<'_>) -> bool {
80        match self {
81            Flavor::Any => true,
82            Flavor::Macro => matches!(kind, Kind::Macro),
83            Flavor::Callable => matches!(kind, Kind::Function(..) | Kind::Variant | Kind::Struct),
84            Flavor::Struct => matches!(kind, Kind::Struct),
85            Flavor::Enum => matches!(kind, Kind::Enum),
86            Flavor::Trait => matches!(kind, Kind::Trait),
87            Flavor::Type => matches!(kind, Kind::Type | Kind::Struct | Kind::Enum),
88            Flavor::Module => matches!(kind, Kind::Module),
89            Flavor::Function => matches!(kind, Kind::Function(..)),
90            Flavor::Const => matches!(kind, Kind::Const(..)),
91            Flavor::Variant => matches!(kind, Kind::Variant),
92            Flavor::Value => matches!(kind, Kind::Function(..) | Kind::Const(..) | Kind::Variant),
93        }
94    }
95}

97/// A link which has been taken apart.
98struct Parsed<'a> {
99    flavor: Flavor,
100    absolute: bool,
101    segments: Vec<&'a str>,
102    fragment: Option<&'a str>,
103}
104
105/// Take a link apart, returning `None` if it does not look like a path.
106fn parse(link: &str) -> alloc::Result<Option<Parsed<'_>>> {
107    let link = link.trim().trim_matches('`');
108
109    let (link, fragment) = match link.split_once('#') {
110        Some((link, fragment)) => (link, Some(fragment)),
111        None => (link, None),
112    };
113
114    if fragment.is_some_and(|f| f.is_empty() || f.contains(char::is_whitespace)) {
115        return Ok(None);
116    }
117
118    let (mut flavor, link) = match link.split_once('@') {
119        Some((prefix, link)) => match Flavor::from_disambiguator(prefix) {
120            Some(flavor) => (flavor, link),
121            None => return Ok(None),
122        },
123        None => (Flavor::Any, link),
124    };
125
126    let mut link = link;
127
128    if let Some(rest) = link.strip_suffix('!') {
129        link = rest;
130        flavor = Flavor::Macro;
131    } else if let Some(rest) = link.strip_suffix(')') {
132        // Both `[Vec::new()]` and `[Some(v)]`, which names what is called
133        // rather than the call.
134        let Some((rest, _)) = rest.split_once('(') else {
135            return Ok(None);
136        };
137
138        link = rest;
139
140        if matches!(flavor, Flavor::Any) {
141            flavor = Flavor::Callable;
142        }
143    }
144
145    // Generic parameters, as in `[Result<T, E>]`, name the type they are on.
146    if let Some(rest) = link.strip_suffix('>') {
147        let Some((rest, _)) = rest.split_once('<') else {
148            return Ok(None);
149        };
150
151        link = rest;
152    }
153
154    let (absolute, link) = match link.strip_prefix("::") {
155        Some(link) => (true, link),
156        None => (false, link),
157    };
158
159    let mut segments = Vec::new();
160
161    for segment in link.split("::") {
162        if !is_ident(segment) {
163            return Ok(None);
164        }
165
166        segments.try_push(segment)?;
167    }
168
169    Ok(Some(Parsed {
170        flavor,
171        absolute,
172        segments,
173        fragment,
174    }))
175}
176
177fn is_ident(s: &str) -> bool {
178    let mut chars = s.chars();
179
180    let Some(first) = chars.next() else {
181        return false;
182    };
183
184    (first.is_alphabetic() || first == '_') && chars.all(|c| c.is_alphanumeric() || c == '_')
185}

187/// Resolve `link`, written in the documentation of `scope`.
188pub(crate) fn resolve(
189    context: &Context<'_>,
190    prelude: &Prelude,
191    scope: Meta<'_>,
192    link: &str,
193) -> alloc::Result<Link> {
194    let Some(parsed) = parse(link)? else {
195        return Ok(Link::NotAPath);
196    };
197
198    let Some((&first, rest)) = parsed.segments.split_first() else {
199        return Ok(Link::NotAPath);
200    };
201
202    let target = if parsed.absolute {
203        lookup(context, &ItemBuf::with_crate(first)?, rest, parsed.flavor)?
204    } else {
205        match first {
206            "crate" => match scope.item.first() {
207                Some(c) => lookup(context, &ItemBuf::with_item([c])?, rest, parsed.flavor)?,
208                None => None,
209            },
210            "self" => {
211                let module = module_of(context, scope)?;
212                lookup(context, module, rest, parsed.flavor)?
213            }
214            "super" => {
215                let mut module = module_of(context, scope)?.parent();
216                let mut rest = rest;
217
218                while let Some((&"super", tail)) = rest.split_first() {
219                    module = module.and_then(Item::parent);
220                    rest = tail;
221                }
222
223                match module {
224                    Some(module) => lookup(context, module, rest, parsed.flavor)?,
225                    None => None,
226                }
227            }
228            "Self" => match self_type_of(context, scope)? {
229                Some(ty) => lookup(context, ty, rest, parsed.flavor)?,
230                None => None,
231            },
232            _ => lookup_in_scope(context, prelude, scope, &parsed.segments, parsed.flavor)?,
233        }
234    };
235
236    let Some(mut target) = target else {
237        return Ok(Link::Unresolved);
238    };
239
240    if let Some(fragment) = parsed.fragment {
241        target.fragment = Some(fragment.try_to_owned()?);
242    }
243
244    Ok(Link::Found(target))
245}

247/// Look a relative path up the way a name written in `scope` would be.
248fn lookup_in_scope(
249    context: &Context<'_>,
250    prelude: &Prelude,
251    scope: Meta<'_>,
252    segments: &[&str],
253    flavor: Flavor,
254) -> alloc::Result<Option<Target>> {
255    // What an associated function or variant is documented under is its type,
256    // so the names on that type are in scope as well, which is what lets the
257    // documentation of `Peekable` say `[peek]`.
258    if let Some(ty) = self_type_of(context, scope)? {
259        if let Some(target) = lookup(context, ty, segments, flavor)? {
260            return Ok(Some(target));
261        }
262    }
263
264    let mut module = Some(module_of(context, scope)?);
265
266    while let Some(m) = module {
267        if let Some(target) = lookup(context, m, segments, flavor)? {
268            return Ok(Some(target));
269        }
270
271        module = m.parent();
272    }
273
274    if let Some((&first, rest)) = segments.split_first() {
275        if let Some(item) = prelude.get(first) {
276            if let Some(target) = lookup(context, item, rest, flavor)? {
277                return Ok(Some(target));
278            }
279        }
280
281        if let Some(target) = lookup(context, &ItemBuf::with_crate(first)?, rest, flavor)? {
282            return Ok(Some(target));
283        }
284    }
285
286    Ok(None)
287}

289/// The module whose names are in scope in the documentation of `scope`.
290fn module_of<'a>(context: &Context<'_>, scope: Meta<'a>) -> alloc::Result<&'a Item> {
291    let mut item = Some(scope.item);
292
293    while let Some(current) = item {
294        if context
295            .meta(current)?
296            .iter()
297            .any(|m| matches!(m.kind, Kind::Module))
298        {
299            return Ok(current);
300        }
301
302        item = current.parent();
303    }
304
305    // Nothing up the path is known to be a module, which happens for the
306    // items of a source which was not visited as a whole. The parent is the
307    // closest thing to one.
308    Ok(scope.item.parent().unwrap_or(scope.item))
309}

311/// The type `Self` refers to in the documentation of `scope`, if any.
312fn self_type_of<'a>(context: &Context<'_>, scope: Meta<'a>) -> alloc::Result<Option<&'a Item>> {
313    if is_type(&scope.kind) {
314        return Ok(Some(scope.item));
315    }
316
317    if let Some(parent) = scope.item.parent() {
318        if matches!(
319            scope.kind,
320            Kind::Function(..) | Kind::Variant | Kind::Const(..)
321        ) && page_kind(context, parent)?.is_some()
322        {
323            return Ok(Some(parent));
324        }
325    }
326
327    Ok(None)
328}

330fn is_type(kind: &Kind<'_>) -> bool {
331    matches!(kind, Kind::Type | Kind::Struct | Kind::Enum | Kind::Trait)
332}
333
334/// The kind of page a type has, if `item` is a type.
335fn page_kind(context: &Context<'_>, item: &Item) -> alloc::Result<Option<ItemKind>> {
336    for meta in context.meta(item)? {
337        let kind = match meta.kind {
338            Kind::Type => ItemKind::Type,
339            Kind::Struct => ItemKind::Struct,
340            Kind::Enum => ItemKind::Enum,
341            Kind::Trait => ItemKind::Trait,
342            _ => continue,
343        };
344
345        return Ok(Some(kind));
346    }
347
348    Ok(None)
349}

351/// Look up `base::segments`, and what page it is on.
352fn lookup(
353    context: &Context<'_>,
354    base: &Item,
355    segments: &[&str],
356    flavor: Flavor,
357) -> alloc::Result<Option<Target>> {
358    let item = base.join(segments)?;
359
360    let mut best = None::<(u8, Target)>;
361
362    for meta in context.meta(&item)? {
363        if !flavor.accepts(&meta.kind) {
364            continue;
365        }
366
367        let Some((rank, target)) = target_of(context, &item, &meta.kind)? else {
368            continue;
369        };
370
371        // A name can be more than one thing, like `u64` which is both a type
372        // and a module of functions on it. The type is what is meant.
373        if best.as_ref().is_none_or(|(best, _)| rank < *best) {
374            best = Some((rank, target));
375        }
376    }
377
378    Ok(best.map(|(_, target)| target))
379}

381/// The page `item` is documented on, ranked by how much it is what a name is
382/// expected to mean when it is more than one thing.
383fn target_of(
384    context: &Context<'_>,
385    item: &Item,
386    kind: &Kind<'_>,
387) -> alloc::Result<Option<(u8, Target)>> {
388    let own = |kind: ItemKind, what: &'static str| -> alloc::Result<Target> {
389        Ok(Target {
390            item: item.try_to_owned()?,
391            kind,
392            fragment: None,
393            what,
394        })
395    };
396
397    let on_parent = |prefix: &str, what: &'static str| -> alloc::Result<Option<Target>> {
398        let (Some(parent), Some(name)) = (item.parent(), item.last()) else {
399            return Ok(None);
400        };
401
402        let Some(kind) = page_kind(context, parent)? else {
403            return Ok(None);
404        };
405
406        let fragment = match name {
407            ComponentRef::Str(name) if !prefix.is_empty() => Some(try_format!("{prefix}.{name}")),
408            _ => None,
409        };
410
411        Ok(Some(Target {
412            item: parent.try_to_owned()?,
413            kind,
414            fragment,
415            what,
416        }))
417    };
418
419    Ok(Some(match kind {
420        Kind::Type => (0, own(ItemKind::Type, "type")?),
421        Kind::Struct => (0, own(ItemKind::Struct, "struct")?),
422        Kind::Enum => (0, own(ItemKind::Enum, "enum")?),
423        Kind::Trait => (0, own(ItemKind::Trait, "trait")?),
424        Kind::Variant => match on_parent("variant", "variant")? {
425            Some(target) => (1, target),
426            None => return Ok(None),
427        },
428        // An associated function is documented on the page of its type, and
429        // has no page of its own.
430        Kind::Function(..) => match on_parent("method", "method")? {
431            Some(target) => (1, target),
432            None => (1, own(ItemKind::Function, "function")?),
433        },
434        Kind::Macro => (2, own(ItemKind::Macro, "macro")?),
435        // Constants have no page of their own, nor an entry on any other, so
436        // the closest thing is the page of what holds them.
437        Kind::Const(..) => match on_parent("", "constant")? {
438            Some(target) => (2, target),
439            None => match item.parent() {
440                Some(parent) => (
441                    2,
442                    Target {
443                        item: parent.try_to_owned()?,
444                        kind: ItemKind::Module,
445                        fragment: None,
446                        what: "constant",
447                    },
448                ),
449                None => return Ok(None),
450            },
451        },
452        Kind::Module => (3, own(ItemKind::Module, "module")?),
453        Kind::Unsupported => return Ok(None),
454    }))
