12/// The signature of a descriptive type name function.
13type DebugFn = fn(&mut fmt::Formatter<'_>) -> fmt::Result;

15/// The kind of the stored value in the `AnyObj`.
16enum Kind {
17    /// Underlying access is shared.
18    Shared,
19    /// Underlying access is exclusively accessible.
20    Exclusive,
21    /// The underlying type is owned.
22    Owned,
23}
24
25pub(super) struct AnyObjVtable {
26    /// The statically known kind of reference being stored.
27    kind: Kind,
28    /// Punt the inner pointer to the type corresponding to the type hash.
29    type_id: TypeId,
30    /// Static type information.
31    type_info: AnyTypeInfo,
32    /// Type hash of the interior type.
33    type_hash: Hash,
34    /// Type information for diagnostics.
35    debug: DebugFn,
36    /// Value drop implementation. Set to `None` if the underlying value does
37    /// not need to be dropped.
38    pub(super) drop_value: Option<unsafe fn(NonNull<AnyObjData>)>,
39    /// Hand over the values the stored value is made of, so that a graph of
40    /// values is taken apart without recursing into it - see [`Dismantle`].
41    ///
42    /// Set to `None` if the value is only pointed at, since dropping the handle
43    /// then never drops it and there is nothing to hand over.
44    pub(super) dismantle: Option<unsafe fn(NonNull<AnyObjData>, &mut Handover<'_>)>,
45    /// Only drop the box implementation.
46    pub(super) drop: unsafe fn(NonNull<AnyObjData>),
47    /// Clone the literal content of the shared value.
48    pub(super) clone: unsafe fn(NonNull<AnyObjData>) -> alloc::Result<AnyObj>,
49}

51impl AnyObjVtable {
52    #[inline]
53    pub(super) const fn owned<T>() -> &'static Self
54    where
55        T: Any,
56    {
57        &Self {
58            kind: Kind::Owned,
59            type_id: const { TypeId::of::<T>() },
60            debug: debug_ref_impl::<T>,
61            type_info: T::ANY_TYPE_INFO,
62            type_hash: T::HASH,
63            drop_value: const {
64                if needs_drop::<T>() {
65                    Some(drop_value::<T>)
66                } else {
67                    None
68                }
69            },
70            dismantle: Some(dismantle::<T>),
71            drop: drop_box::<ManuallyDrop<T>>,
72            clone: clone_own::<T>,
73        }
74    }
75
76    #[inline]
77    pub(super) const fn from_ref<T>() -> &'static Self
78    where
79        T: Any,
80    {
81        &Self {
82            kind: Kind::Shared,
83            type_id: const { TypeId::of::<T>() },
84            debug: debug_ref_impl::<T>,
85            type_info: T::ANY_TYPE_INFO,
86            type_hash: T::HASH,
87            drop_value: None,
88            dismantle: None,
89            drop: drop_box::<NonNull<T>>,
90            clone: clone_ref::<T>,
91        }
92    }
93
94    #[inline]
95    pub(super) const fn from_mut<T>() -> &'static Self
96    where
97        T: Any,
98    {
99        &Self {
100            kind: Kind::Exclusive,
101            type_id: const { TypeId::of::<T>() },
102            debug: debug_mut_impl::<T>,
103            type_info: T::ANY_TYPE_INFO,
104            type_hash: T::HASH,
105            drop_value: None,
106            dismantle: None,
107            drop: drop_box::<NonNull<T>>,
108            clone: clone_mut::<T>,
109        }
110    }
111
112    #[inline]
113    pub(super) fn is<T>(&self) -> bool
114    where
115        T: 'static,
116    {
117        self.type_id == TypeId::of::<T>()
118    }
119
120    #[inline]
121    pub(super) fn type_info(&self) -> TypeInfo {
122        TypeInfo::any_type_info(self.type_info)
123    }
124
125    #[inline]
126    pub(super) fn type_hash(&self) -> Hash {
127        self.type_hash
128    }
129
130    #[inline]
131    pub(super) fn debug(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
132        (self.debug)(f)
133    }
134
135    #[inline]
136    pub(super) fn as_ptr<T>(&self, base: NonNull<AnyObjData>) -> NonNull<T> {
137        if matches!(self.kind, Kind::Owned) {
138            unsafe { base.byte_add(offset_of!(AnyObjData<T>, data)).cast() }
139        } else {
140            unsafe {
141                base.byte_add(offset_of!(AnyObjData<NonNull<T>>, data))
142                    .cast()
143                    .read()
144            }
145        }
146    }
147
148    #[inline]
149    pub(super) fn is_mutable(&self) -> bool {
150        matches!(self.kind, Kind::Exclusive | Kind::Owned)
151    }
152
153    #[inline]
154    pub(super) fn is_owned(&self) -> bool {
155        matches!(self.kind, Kind::Owned)
156    }
157}

173/// Ask the type which is stored to hand over the values it is made of.
174///
175/// # Safety
176///
177/// The caller must ensure that the data stored is of type `T` and that nothing
178/// else is accessing it.
179unsafe fn dismantle<T>(this: NonNull<AnyObjData>, out: &mut Handover<'_>)
180where
181    T: Any,
182{
183    let data = addr_of_mut!((*this.cast::<AnyObjData<T>>().as_ptr()).data);
184    Dismantle::dismantle(&mut *data, out)
185}
186
187unsafe fn drop_value<T>(this: NonNull<AnyObjData>) {
188    let data = addr_of_mut!((*this.cast::<AnyObjData<T>>().as_ptr()).data);
189    drop_in_place(data);
190}
191
192unsafe fn drop_box<T>(this: NonNull<AnyObjData>) {
193    let data = Box::from_raw_in(this.cast::<AnyObjData<T>>().as_ptr(), Global);
194    drop(data)
195}
196
197unsafe fn clone_own<T>(this: NonNull<AnyObjData>) -> alloc::Result<AnyObj>
198where
199    T: Any,
200{
201    // NB: We read the value without deallocating it from the previous location,
202    // since that would cause the returned value to be invalid.
203    let value = addr_of_mut!((*this.cast::<AnyObjData<T>>().as_ptr()).data).read();
204    AnyObj::new(value)
205}
206
207unsafe fn clone_ref<T>(this: NonNull<AnyObjData>) -> alloc::Result<AnyObj>
208where
209    T: Any,
210{
211    let value = addr_of_mut!((*this.cast::<AnyObjData<NonNull<T>>>().as_ptr()).data).read();
212    AnyObj::from_ref(value.as_ptr().cast_const())
213}
214
215unsafe fn clone_mut<T>(this: NonNull<AnyObjData>) -> alloc::Result<AnyObj>
216where
217    T: Any,
218{
219    let value = addr_of_mut!((*this.cast::<AnyObjData<NonNull<T>>>().as_ptr()).data).read();
220    AnyObj::from_mut(value.as_ptr())
221}
