1//! Constant evaluation.
2//!
3//! Constants are not interpreted. They are compiled - by the same assembler
4//! which compiles everything else - into an interior [`UnitBuilder`] which is
5//! then snapshot into a [`Unit`] and executed by a [`Vm`].
6//!
7//! The interior unit is kept for the duration of the compilation. Constant
8//! functions are assembled into it once and called as ordinary functions, so a
9//! constant function which is called in a loop is compiled once rather than
10//! re-interpreted per call, and recursion between constant functions costs
11//! virtual machine call frames rather than compiler stack frames.
12

31/// The interior unit constants are compiled into before being evaluated.
32#[derive(Default)]
33pub(crate) struct ConstUnit {
34    /// The unit being built.
35    unit: UnitBuilder,
36    /// Instruction storage associated with `unit`.
37    storage: DefaultStorage,
38    /// Counter used to hand out a distinct hash to every entry point.
39    entry: u64,
40    /// The runtime context to execute against.
41    ///
42    /// Building it copies the whole context, so it is only built if a constant
43    /// is actually evaluated, and then only once.
44    runtime: Option<Arc<RuntimeContext>>,
45}
46
47impl ConstUnit {
48    /// Mirror metadata into the interior unit.
49    ///
50    /// Runtime type information and the constants associated with an item are
51    /// registered with a unit as the item is first queried for, so the interior
52    /// unit has to see the same registrations as the unit being compiled or it
53    /// would not be able to construct or match the types a constant mentions.
54    pub(crate) fn insert_meta(
55        &mut self,
56        span: &dyn Spanned,
57        meta: &compile::meta::Meta,
58        pool: &compile::Pool,
59        query: &mut crate::query::QueryInner,
60    ) -> compile::Result<()> {
61        self.unit.insert_meta(span, meta, pool, query, false)
62    }
63
64    /// Allocate the hash of the next entry point.
65    fn next_entry(&mut self) -> Hash {
66        self.entry += 1;
67        Hash::new(self.entry)
68    }
69}

71/// What to assemble and call for a single evaluation.
72pub(crate) enum Entry<'hir> {
73    /// Evaluate an expression, which is assembled as the body of an entry
74    /// point taking no arguments.
75    Expr(&'hir hir::Exprs<'hir>, &'hir hir::Expr<'hir>),
76    /// Evaluate a block, which is assembled as the body of an entry point
77    /// taking no arguments.
78    Block(&'hir hir::Exprs<'hir>, &'hir hir::Block<'hir>),
79    /// Call the constant function with the given item, which becomes the entry
80    /// point itself.
81    Call(ItemId),
82}
83
84/// The body of an entry point.
85enum Body<'hir> {
86    Expr(&'hir hir::Expr<'hir>),
87    Block(&'hir hir::Block<'hir>),
88    Fn(&'hir hir::ItemFn<'hir>),
89}
90
91impl<'hir> Body<'hir> {
92    /// The span the body is assembled at, used as the outermost warning
93    /// context.
94    fn span(&self) -> &'hir dyn Spanned {
95        match *self {
96            Body::Expr(hir) => hir,
97            Body::Block(hir) => hir,
98            Body::Fn(hir) => hir,
99        }
100    }
101}

103/// Evaluate an expression from within a macro, as `cx.eval(..)` does.
104///
105/// The expression arrives as an [`ast::Expr`], which a macro has parsed out of
106/// its own input, so it has been through neither the grammar nor indexing. It
107/// is emitted back into a token stream and evaluated as one.
108pub(crate) fn eval_ast(
109    cx: &mut MacroContext<'_, '_, '_>,
110    ast: &ast::Expr,
111) -> compile::Result<Value> {
112    let mut stream = TokenStream::new();
113    ast.to_tokens(cx, &mut stream)?;
114    eval_stream(cx, &stream, ast.span(), "Expression")
115}

117/// Evaluate an expression from within a macro, as `cx.eval_stream(..)` does.
118///
119/// The expression arrives as the tokens it was written as, which have been
120/// through neither the grammar nor indexing. They are parsed here the same way
121/// [`crate::worker`] parses the output of a macro, so that everything
122/// downstream of it walks a tree over an explicit stack rather than recursing
123/// over a syntax tree.
124///
125/// An expression which turns out not to be constant - it names a local
126/// variable, or calls a function which is not a `const fn` - is reported as
127/// `what` having to be a constant expression, against the whole of it, with
128/// what made it not constant kept as the cause. Without it the error is only
129/// about the part of the expression responsible, as if it had been an ordinary
130/// expression, and nothing says that the macro asked for a constant.
131pub(crate) fn eval_stream(
132    cx: &mut MacroContext<'_, '_, '_>,
133    stream: &TokenStream,
134    span: Span,
135    what: &'static str,
136) -> compile::Result<Value> {
137    let location = cx.item_meta.location;
138
139    let tree = crate::grammar::token_stream(stream)
140        .max_nesting(cx.idx.q.options.max_depth)
141        .root()?;
142
143    let tree = Rc::new(tree);
144
145    // The expression may contain items - a closure, or a block with items in
146    // it - which have to be indexed before they can be lowered.
147    {
148        let items = Items::new(cx.idx.q.pool.item(cx.idx.item.id))?;
149
150        let mut idx = Indexer {
151            q: cx.idx.q.borrow(),
152            source_id: cx.idx.source_id,
153            items,
154            scopes: Scopes::new()?,
155            item: cx.idx.item,
156            nested_item: cx.idx.nested_item,
157            macro_depth: cx.idx.macro_depth,
158            root: cx.idx.root,
159            queue: cx.idx.queue.as_deref_mut(),
160            loaded: cx.idx.loaded.as_deref_mut(),
161            tree: &tree,
162        };
163
164        tree.parse_all(|p| index2::any(&mut idx, p))?;
165    }
166
167    let node = {
168        let Some([root]) = tree.nodes() else {
169            return Err(compile::Error::msg(
170                span,
171                "expected single root in evaluated expression",
172            ));
173        };
174
175        if !matches!(root.kind(), Kind::Root) {
176            return Err(compile::Error::expected(root, Kind::Root));
177        }
178
179        let Some([node]) = root.nodes() else {
180            return Err(compile::Error::msg(
181                span,
182                "expected single evaluated expression",
183            ));
184        };
185
186        if !matches!(node.kind(), Kind::Expr) {
187            return Err(compile::Error::expected(node, Kind::Expr));
188        }
189
190        node
191    };
192
193    let arena = hir::Arena::new();
194
195    let result = 'eval: {
196        let (hir, exprs) = {
197            let mut hir_cx = hir::Ctxt::with_const(&arena, cx.idx.q.borrow(), location.source_id)?;
198
199            let hir = match node.parse(|p| hir::lowering2::expr(&mut hir_cx, p)) {
200                Ok(hir) => hir,
201                Err(error) => break 'eval Err(error),
202            };
203
204            (hir, hir_cx.take_exprs())
205        };
206
207        let mut q = cx.idx.q.borrow();
208
209        eval(
210            &mut q,
211            location,
212            &hir,
213            Entry::Expr(&exprs, &hir),
214            Vec::new(),
215        )
216    };
217
218    // Only what goes wrong once the expression has been parsed can be because
219    // it is not constant.
220    match result {
221        Err(error) if is_not_const(error.kind()) => Err(compile::Error::new(
222            span,
223            ErrorKind::NotConstExpr {
224                what,
225                cause: rust_alloc::boxed::Box::new(error),
226            },
227        )),
228        result => result,
229    }
230}

232/// Test if an error raised while lowering or evaluating an expression is
233/// because the expression is not constant, rather than because evaluating it
234/// failed the way it would have at runtime too, or because the compiler ran out
235/// of something.
236fn is_not_const(kind: &ErrorKind) -> bool {
237    !matches!(
238        kind,
239        ErrorKind::AllocError { .. }
240            | ErrorKind::MaxMacroRecursion { .. }
241            | ErrorKind::MaxDepth { .. }
242            | ErrorKind::MaxNesting { .. }
243            | ErrorKind::MaxItemDepth { .. }
244            | ErrorKind::MaxConstDepth { .. }
245            | ErrorKind::MaxGenericsDepth { .. }
246            | ErrorKind::MaxAstDepth { .. }
247            | ErrorKind::ConstBudgetExceeded { .. }
248            | ErrorKind::ConstCycle
249            | ErrorKind::Vm(..)
250            | ErrorKind::NotConstExpr { .. }
251    )
252}

254/// Evaluate a constant, returning the value it produced.
255pub(crate) fn eval<'hir>(
256    q: &mut Query<'_, '_>,
257    location: Location,
258    span: &dyn Spanned,
259    entry: Entry<'hir>,
260    args: Vec<Value>,
261) -> compile::Result<Value> {
262    // The interior unit is moved out for the duration of the evaluation, so
263    // that the query engine can be reborrowed against it while it is being
264    // assembled into.
265    //
266    // This relies on an evaluation never starting another one. Metadata is
267    // resolved while lowering, not while assembling, so nothing reachable from
268    // here queries for an item - and queries are what evaluate constants.
269    let mut cu = take(&mut q.inner.const_unit);
270    let result = eval_with(q, &mut cu, location, span, entry, args);
271    q.inner.const_unit = cu;
272    result
273}

275fn eval_with<'hir>(
276    q: &mut Query<'_, '_>,
277    cu: &mut ConstUnit,
278    location: Location,
279    span: &dyn Spanned,
280    entry: Entry<'hir>,
281    args: Vec<Value>,
282) -> compile::Result<Value> {
283    let hash = match entry {
284        Entry::Call(id) => q.const_fn_hash(id).with_span(span)?,
285        Entry::Expr(exprs, hir) => {
286            let hash = cu.next_entry();
287            assemble(q, cu, location, hash, exprs, Body::Expr(hir))?;
288            hash
289        }
290        Entry::Block(exprs, hir) => {
291            let hash = cu.next_entry();
292            assemble(q, cu, location, hash, exprs, Body::Block(hir))?;
293            hash
294        }
295    };
296
297    // Assembling the entry point discovers the constant functions it calls, and
298    // assembling those discovers the ones they call in turn, so this drains
299    // rather than iterates.
300    while let Some(id) = q.inner.const_pending.pop() {
301        let result = assemble_const_fn(q, cu, span, id);
302
303        if result.is_err() {
304            // The function is not in the unit, so let a later evaluation queue
305            // it again and fail the same way, rather than calling a function
306            // which was never assembled.
307            q.inner.const_queued.remove(&id);
308        }
309
310        result?;
311    }
312
313    run(q, cu, span, hash, args)
314}

316/// Assemble a constant function into the interior unit.
317fn assemble_const_fn(
318    q: &mut Query<'_, '_>,
319    cu: &mut ConstUnit,
320    span: &dyn Spanned,
321    id: ItemId,
322) -> compile::Result<()> {
323    let const_fn = q.const_fn_for(id).with_span(span)?;
324    let hash = q.pool.item_type_hash(id);
325
326    assemble(
327        q,
328        cu,
329        const_fn.item_meta.location,
330        hash,
331        &const_fn.exprs,
332        Body::Fn(&const_fn.hir),
333    )
334}

336/// Assemble a body into the interior unit under the given hash.
337fn assemble<'hir>(
338    q: &mut Query<'_, '_>,
339    cu: &mut ConstUnit,
340    location: Location,
341    hash: Hash,
342    exprs: &'hir hir::Exprs<'hir>,
343    body: Body<'hir>,
344) -> compile::Result<()> {
345    let options = q.options;
346    let span = body.span();
347
348    let ConstUnit { unit, storage, .. } = cu;
349
350    let mut asm = unit.new_assembly(location);
351
352    let (args, size) = {
353        let mut q = q.with_unit(unit);
354        let mut scopes = v2::Scopes::new(location.source_id)?;
355
356        let mut cx = v2::Ctxt {
357            source_id: location.source_id,
358            q: q.borrow(),
359            asm: &mut asm,
360            exprs,
361            scopes: &mut scopes,
362            contexts: try_vec![span],
363            breaks: v2::Breaks::new(),
364            options,
365            select_branches: Vec::new(),
366            drop: Vec::new(),
367            const_eval: true,
368        };
369
370        let args = match body {
371            Body::Expr(hir) => {
372                v2::assemble::const_expr(&mut cx, hir)?;
373                0
374            }
375            Body::Block(hir) => {
376                v2::assemble::const_block(&mut cx, hir)?;
377                0
378            }
379            Body::Fn(hir) => {
380                v2::assemble::fn_from_item_fn(&mut cx, hir)?;
381                hir.args.len()
382            }
383        };
384
385        (args, cx.scopes.size())
386    };
387
388    unit.new_const_function(location, hash, args, asm, storage, size)
389}

391/// Snapshot the interior unit and execute the given entry point in it.
392fn run(
393    q: &mut Query<'_, '_>,
394    cu: &mut ConstUnit,
395    span: &dyn Spanned,
396    hash: Hash,
397    args: Vec<Value>,
398) -> compile::Result<Value> {
399    let runtime = match &cu.runtime {
400        Some(runtime) => runtime.clone(),
401        None => {
402            let runtime = Arc::try_new(q.context.runtime().with_span(span)?).with_span(span)?;
403            cu.runtime = Some(runtime.clone());
404            runtime
405        }
406    };
407
408    let storage = cu.storage.try_clone().with_span(span)?;
409    let unit = Arc::try_new(cu.unit.snapshot(storage).with_span(span)?).with_span(span)?;
410
411    let mut vm = Vm::new(runtime, unit);
412    let budget = q.options.const_budget;
413
414    match budget::with(budget, || vm.call(hash, args)).call() {
415        Ok(value) => Ok(value),
416        // A constant which does not terminate halts the machine rather than
417        // erroring, so it is reported as the budget being spent instead of as
418        // an unexpected halt.
419        Err(error) => match error.into_kind() {
420            VmErrorKind::Halted {
421                halt: VmHaltInfo::Limited,
422            } => Err(compile::Error::new(
423                span,
424                ErrorKind::ConstBudgetExceeded { budget },
425            )),
426            // A function which is not a constant function is not in the
427            // interior unit at all, so all the machine can say is that a hash
428            // is missing. Which function it was is what is actually useful.
429            VmErrorKind::MissingFunction { hash } => {
430                let kind = match q.pool.item_for_hash(hash) {
431                    Some(item) => ErrorKind::NotConstFn {
432                        item: item.try_to_owned().with_span(span)?,
433                    },
434                    None => ErrorKind::from(VmError::new(VmErrorKind::MissingFunction { hash })),
435                };
436
437                Err(compile::Error::new(span, kind))
438            }
439            kind => Err(compile::Error::new(
440                span,
441                ErrorKind::from(VmError::new(kind)),
442            )),
443        },
444    }
445}

447impl Query<'_, '_> {
448    /// Get the hash of the constant function with the given item, queueing it
449    /// to be assembled into the interior unit if it has not been already.
450    pub(crate) fn const_fn_hash(&mut self, id: ItemId) -> alloc::Result<Hash> {
451        if self.inner.const_queued.try_insert(id)? {
452            self.inner.const_pending.try_push(id)?;
453        }
454
455        Ok(self.pool.item_type_hash(id))
456    }
