1//! Constant evaluation.
2//!
3//! Constants are not interpreted. They are compiled - by the same assembler
4//! which compiles everything else - into an interior [`UnitBuilder`] which is
5//! then snapshot into a [`Unit`] and executed by a [`Vm`].
6//!
7//! The interior unit is kept for the duration of the compilation. Constant
8//! functions are assembled into it once and called as ordinary functions, so a
9//! constant function which is called in a loop is compiled once rather than
10//! re-interpreted per call, and recursion between constant functions costs
11//! virtual machine call frames rather than compiler stack frames.
12

31/// The interior unit constants are compiled into before being evaluated.
32#[derive(Default)]
33pub(crate) struct ConstUnit {
34    /// The unit being built.
35    unit: UnitBuilder,
36    /// Instruction storage associated with `unit`.
37    storage: DefaultStorage,
38    /// Counter used to hand out a distinct hash to every entry point.
39    entry: u64,
40    /// The runtime context to execute against.
41    ///
42    /// Building it copies the whole context, so it is only built if a constant
43    /// is actually evaluated, and then only once.
44    runtime: Option<Arc<RuntimeContext>>,
45}
46
47impl ConstUnit {
48    /// Mirror metadata into the interior unit.
49    ///
50    /// Runtime type information and the constants associated with an item are
51    /// registered with a unit as the item is first queried for, so the interior
52    /// unit has to see the same registrations as the unit being compiled or it
53    /// would not be able to construct or match the types a constant mentions.
54    pub(crate) fn insert_meta(
55        &mut self,
56        span: &dyn Spanned,
57        meta: &compile::meta::Meta,
58        pool: &compile::Pool,
59        query: &mut crate::query::QueryInner,
60    ) -> compile::Result<()> {
61        self.unit.insert_meta(span, meta, pool, query, false)
62    }
63
64    /// Allocate the hash of the next entry point.
65    fn next_entry(&mut self) -> Hash {
66        self.entry += 1;
67        Hash::new(self.entry)
68    }
69}

71/// What to assemble and call for a single evaluation.
72pub(crate) enum Entry<'hir> {
73    /// Evaluate an expression, which is assembled as the body of an entry
74    /// point taking no arguments.
75    Expr(&'hir hir::Exprs<'hir>, &'hir hir::Expr<'hir>),
76    /// Evaluate a block, which is assembled as the body of an entry point
77    /// taking no arguments.
78    Block(&'hir hir::Exprs<'hir>, &'hir hir::Block<'hir>),
79    /// Call the constant function with the given item, which becomes the entry
80    /// point itself.
81    Call(ItemId),
82}
83
84/// The body of an entry point.
85enum Body<'hir> {
86    Expr(&'hir hir::Expr<'hir>),
87    Block(&'hir hir::Block<'hir>),
88    Fn(&'hir hir::ItemFn<'hir>),
89}
90
91impl<'hir> Body<'hir> {
92    /// The span the body is assembled at, used as the outermost warning
93    /// context.
94    fn span(&self) -> &'hir dyn Spanned {
95        match *self {
96            Body::Expr(hir) => hir,
97            Body::Block(hir) => hir,
98            Body::Fn(hir) => hir,
99        }
100    }
101}

103/// Evaluate an expression from within a macro, as `cx.eval(..)` does.
104///
105/// The expression arrives as an [`ast::Expr`], which a macro has parsed out of
106/// its own input, so it has been through neither the grammar nor indexing. It
107/// is emitted back into a token stream and re-parsed here, which is what
108/// [`crate::worker`] does with the output of a macro, so that everything
109/// downstream of it walks a tree over an explicit stack rather than recursing
110/// over the syntax tree.
111pub(crate) fn eval_ast(
112    cx: &mut MacroContext<'_, '_, '_>,
113    ast: &ast::Expr,
114) -> compile::Result<Value> {
115    let location = cx.item_meta.location;
116
117    let mut stream = TokenStream::new();
118    ast.to_tokens(cx, &mut stream)?;
119
120    let tree = crate::grammar::token_stream(&stream)
121        .max_nesting(cx.idx.q.options.max_depth)
122        .root()?;
123
124    let tree = Rc::new(tree);
125
126    // The expression may contain items - a closure, or a block with items in
127    // it - which have to be indexed before they can be lowered.
128    {
129        let items = Items::new(cx.idx.q.pool.item(cx.idx.item.id))?;
130
131        let mut idx = Indexer {
132            q: cx.idx.q.borrow(),
133            source_id: cx.idx.source_id,
134            items,
135            scopes: Scopes::new()?,
136            item: cx.idx.item,
137            nested_item: cx.idx.nested_item,
138            macro_depth: cx.idx.macro_depth,
139            root: cx.idx.root,
140            queue: cx.idx.queue.as_deref_mut(),
141            loaded: cx.idx.loaded.as_deref_mut(),
142            tree: &tree,
143        };
144
145        tree.parse_all(|p| index2::any(&mut idx, p))?;
146    }
147
148    let node = {
149        let Some([root]) = tree.nodes() else {
150            return Err(compile::Error::msg(
151                ast,
152                "expected single root in evaluated expression",
153            ));
154        };
155
156        if !matches!(root.kind(), Kind::Root) {
157            return Err(compile::Error::expected(root, Kind::Root));
158        }
159
160        let Some([node]) = root.nodes() else {
161            return Err(compile::Error::msg(
162                ast,
163                "expected single evaluated expression",
164            ));
165        };
166
167        if !matches!(node.kind(), Kind::Expr) {
168            return Err(compile::Error::expected(node, Kind::Expr));
169        }
170
171        node
172    };
173
174    let arena = hir::Arena::new();
175
176    let (hir, exprs) = {
177        let mut hir_cx = hir::Ctxt::with_const(&arena, cx.idx.q.borrow(), location.source_id)?;
178        let hir = node.parse(|p| hir::lowering2::expr(&mut hir_cx, p))?;
179        (hir, hir_cx.take_exprs())
180    };
181
182    let mut q = cx.idx.q.borrow();
183    eval(
184        &mut q,
185        location,
186        &hir,
187        Entry::Expr(&exprs, &hir),
188        Vec::new(),
189    )
190}

192/// Evaluate a constant, returning the value it produced.
193pub(crate) fn eval<'hir>(
194    q: &mut Query<'_, '_>,
195    location: Location,
196    span: &dyn Spanned,
197    entry: Entry<'hir>,
198    args: Vec<Value>,
199) -> compile::Result<Value> {
200    // The interior unit is moved out for the duration of the evaluation, so
201    // that the query engine can be reborrowed against it while it is being
202    // assembled into.
203    //
204    // This relies on an evaluation never starting another one. Metadata is
205    // resolved while lowering, not while assembling, so nothing reachable from
206    // here queries for an item - and queries are what evaluate constants.
207    let mut cu = take(&mut q.inner.const_unit);
208    let result = eval_with(q, &mut cu, location, span, entry, args);
209    q.inner.const_unit = cu;
210    result
211}
212
213fn eval_with<'hir>(
214    q: &mut Query<'_, '_>,
215    cu: &mut ConstUnit,
216    location: Location,
217    span: &dyn Spanned,
218    entry: Entry<'hir>,
219    args: Vec<Value>,
220) -> compile::Result<Value> {
221    let hash = match entry {
222        Entry::Call(id) => q.const_fn_hash(id).with_span(span)?,
223        Entry::Expr(exprs, hir) => {
224            let hash = cu.next_entry();
225            assemble(q, cu, location, hash, exprs, Body::Expr(hir))?;
226            hash
227        }
228        Entry::Block(exprs, hir) => {
229            let hash = cu.next_entry();
230            assemble(q, cu, location, hash, exprs, Body::Block(hir))?;
231            hash
232        }
233    };
234
235    // Assembling the entry point discovers the constant functions it calls, and
236    // assembling those discovers the ones they call in turn, so this drains
237    // rather than iterates.
238    while let Some(id) = q.inner.const_pending.pop() {
239        let result = assemble_const_fn(q, cu, span, id);
240
241        if result.is_err() {
242            // The function is not in the unit, so let a later evaluation queue
243            // it again and fail the same way, rather than calling a function
244            // which was never assembled.
245            q.inner.const_queued.remove(&id);
246        }
247
248        result?;
249    }
250
251    run(q, cu, span, hash, args)
252}

254/// Assemble a constant function into the interior unit.
255fn assemble_const_fn(
256    q: &mut Query<'_, '_>,
257    cu: &mut ConstUnit,
258    span: &dyn Spanned,
259    id: ItemId,
260) -> compile::Result<()> {
261    let const_fn = q.const_fn_for(id).with_span(span)?;
262    let hash = q.pool.item_type_hash(id);
263
264    assemble(
265        q,
266        cu,
267        const_fn.item_meta.location,
268        hash,
269        &const_fn.exprs,
270        Body::Fn(&const_fn.hir),
271    )
272}

274/// Assemble a body into the interior unit under the given hash.
275fn assemble<'hir>(
276    q: &mut Query<'_, '_>,
277    cu: &mut ConstUnit,
278    location: Location,
279    hash: Hash,
280    exprs: &'hir hir::Exprs<'hir>,
281    body: Body<'hir>,
282) -> compile::Result<()> {
283    let options = q.options;
284    let span = body.span();
285
286    let ConstUnit { unit, storage, .. } = cu;
287
288    let mut asm = unit.new_assembly(location);
289
290    let (args, size) = {
291        let mut q = q.with_unit(unit);
292        let mut scopes = v2::Scopes::new(location.source_id)?;
293
294        let mut cx = v2::Ctxt {
295            source_id: location.source_id,
296            q: q.borrow(),
297            asm: &mut asm,
298            exprs,
299            scopes: &mut scopes,
300            contexts: try_vec![span],
301            breaks: v2::Breaks::new(),
302            options,
303            select_branches: Vec::new(),
304            drop: Vec::new(),
305            const_eval: true,
306        };
307
308        let args = match body {
309            Body::Expr(hir) => {
310                v2::assemble::const_expr(&mut cx, hir)?;
311                0
312            }
313            Body::Block(hir) => {
314                v2::assemble::const_block(&mut cx, hir)?;
315                0
316            }
317            Body::Fn(hir) => {
318                v2::assemble::fn_from_item_fn(&mut cx, hir)?;
319                hir.args.len()
320            }
321        };
322
323        (args, cx.scopes.size())
324    };
325
326    unit.new_const_function(location, hash, args, asm, storage, size)
327}

329/// Snapshot the interior unit and execute the given entry point in it.
330fn run(
331    q: &mut Query<'_, '_>,
332    cu: &mut ConstUnit,
333    span: &dyn Spanned,
334    hash: Hash,
335    args: Vec<Value>,
336) -> compile::Result<Value> {
337    let runtime = match &cu.runtime {
338        Some(runtime) => runtime.clone(),
339        None => {
340            let runtime = Arc::try_new(q.context.runtime().with_span(span)?).with_span(span)?;
341            cu.runtime = Some(runtime.clone());
342            runtime
343        }
344    };
345
346    let storage = cu.storage.try_clone().with_span(span)?;
347    let unit = Arc::try_new(cu.unit.snapshot(storage).with_span(span)?).with_span(span)?;
348
349    let mut vm = Vm::new(runtime, unit);
350    let budget = q.options.const_budget;
351
352    match budget::with(budget, || vm.call(hash, args)).call() {
353        Ok(value) => Ok(value),
354        // A constant which does not terminate halts the machine rather than
355        // erroring, so it is reported as the budget being spent instead of as
356        // an unexpected halt.
357        Err(error) => match error.into_kind() {
358            VmErrorKind::Halted {
359                halt: VmHaltInfo::Limited,
360            } => Err(compile::Error::new(
361                span,
362                ErrorKind::ConstBudgetExceeded { budget },
363            )),
364            // A function which is not a constant function is not in the
365            // interior unit at all, so all the machine can say is that a hash
366            // is missing. Which function it was is what is actually useful.
367            VmErrorKind::MissingFunction { hash } => {
368                let kind = match q.pool.item_for_hash(hash) {
369                    Some(item) => ErrorKind::NotConstFn {
370                        item: item.try_to_owned().with_span(span)?,
371                    },
372                    None => ErrorKind::from(VmError::new(VmErrorKind::MissingFunction { hash })),
373                };
374
375                Err(compile::Error::new(span, kind))
376            }
377            kind => Err(compile::Error::new(
378                span,
379                ErrorKind::from(VmError::new(kind)),
380            )),
381        },
382    }
383}

385impl Query<'_, '_> {
386    /// Get the hash of the constant function with the given item, queueing it
387    /// to be assembled into the interior unit if it has not been already.
388    pub(crate) fn const_fn_hash(&mut self, id: ItemId) -> alloc::Result<Hash> {
389        if self.inner.const_queued.try_insert(id)? {
390            self.inner.const_pending.try_push(id)?;
391        }
392
393        Ok(self.pool.item_type_hash(id))
394    }
