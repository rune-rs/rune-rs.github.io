18pub(crate) struct Indexer<'a, 'arena> {
19    /// Query engine.
20    pub(crate) q: Query<'a, 'arena>,
21    pub(crate) source_id: SourceId,
22    pub(crate) items: Items,
23    /// Helper to calculate details about an indexed scope.
24    pub(crate) scopes: Scopes,
25    /// The current item state.
26    pub(crate) item: IndexItem,
27    /// Indicates if indexer is nested privately inside of another item, and if
28    /// so, the descriptive span of its declaration.
29    ///
30    /// Private items are nested declarations inside of for example fn
31    /// declarations:
32    ///
33    /// ```text
34    /// pub fn public() {
35    ///     fn private() {
36    ///     }
37    /// }
38    /// ```
39    ///
40    /// Then, `nested_item` would point to the span of `pub fn public`.
41    pub(crate) nested_item: Option<Span>,
42    /// Depth of expression macro expansion that we're currently in.
43    pub(crate) macro_depth: usize,
44    /// The root URL that the indexed file originated from.
45    pub(crate) root: Option<SourceId>,
46    /// Imports to process.
47    pub(crate) queue: Option<&'a mut VecDeque<Task>>,
48    /// Loaded modules.
49    pub(crate) loaded: Option<&'a mut HashMap<ModId, (SourceId, Span)>>,
50    /// The current tree being processed.
51    pub(crate) tree: &'a Rc<Tree>,
52}

65impl Indexer<'_, '_> {
66    /// Bound how many components an item path is allowed to have.
67    ///
68    /// Every level of lexical nesting pushes a component, and allocating an
69    /// item hashes and stores its whole path, so nesting `n` deep costs
70    /// `O(n^2)` in both time and memory. Nothing else bounds this -
71    /// `MAX_ITEM_RECURSION` in `query` bounds how deeply items *depend* on each
72    /// other, not how deeply the source nested.
73    pub(super) fn check_item_depth(&self, span: &dyn Spanned) -> compile::Result<()> {
74        let max = self.q.options.max_item_depth;
75
76        if self.items.depth() >= max {
77            return Err(compile::Error::new(span, ErrorKind::MaxItemDepth { max }));
78        }
79
80        Ok(())
81    }
82
83    /// Push an anonymous item.
84    pub(super) fn push_id(&mut self, span: &dyn Spanned) -> compile::Result<Guard> {
85        self.check_item_depth(span)?;
86        let id = self.q.pool.next_id(self.item.id);
87        Ok(self.items.push_id(id)?)
88    }
89
90    /// Insert a new item at the current indexed location.
91    pub(crate) fn insert_new_item(
92        &mut self,
93        span: &dyn Spanned,
94        visibility: Visibility,
95        docs: &[Doc],
96    ) -> compile::Result<ItemMeta> {
97        self.q.insert_new_item(
98            &self.items,
99            self.item.module,
100            self.item.impl_item,
101            &DynLocation::new(self.source_id, span),
102            visibility,
103            docs,
104        )
105    }
106}

109pub(crate) struct IndexItem {
110    /// The current module being indexed.
111    pub(crate) module: ModId,
112    /// Whether the item has been inserted or not.
113    pub(crate) id: ItemId,
114    /// Set if we are inside of an impl self.
115    pub(crate) impl_item: Option<ItemId>,
116}
117
118impl IndexItem {
119    pub(crate) fn new(module: ModId, id: ItemId) -> Self {
120        Self {
121            module,
122            id,
123            impl_item: None,
124        }
125    }
126
127    pub(crate) fn with_impl_item(module: ModId, id: ItemId, impl_item: ItemId) -> Self {
128        Self {
129            module,
130            id,
131            impl_item: Some(impl_item),
132        }
133    }
134
135    /// Replace item we're currently in.
136    #[tracing::instrument(skip(self), fields(self.module = ?self.module, self.id = ?self.id, self.impl_item = ?self.impl_item))]
137    pub(super) fn replace(&mut self, id: ItemId) -> IndexItem {
138        tracing::debug!("replacing item");
139
140        IndexItem {
141            module: self.module,
142            id: replace(&mut self.id, id),
143            impl_item: self.impl_item,
144        }
145    }
146
147    /// Replace module id.
148    pub(super) fn replace_module(&mut self, module: ModId, id: ItemId) -> IndexItem {
149        IndexItem {
150            module: replace(&mut self.module, module),
151            id: replace(&mut self.id, id),
152            impl_item: self.impl_item,
153        }
154    }
155}

157/// Construct the calling convention based on the parameters.
158pub(super) fn validate_call(
159    is_const: bool,
160    is_async: bool,
161    layer: &Layer,
162) -> compile::Result<Option<Call>> {
163    for span in &layer.awaits {
164        if is_const {
165            return Err(compile::Error::new(span, ErrorKind::AwaitInConst));
166        }
167
168        if !is_async {
169            return Err(compile::Error::new(span, ErrorKind::AwaitOutsideAsync));
170        }
171    }
172
173    for span in &layer.yields {
174        if is_const {
175            return Err(compile::Error::new(span, ErrorKind::YieldInConst));
176        }
177    }
178
179    if is_const {
180        return Ok(None);
181    }
182
183    Ok(match (!layer.yields.is_empty(), is_async) {
184        (true, false) => Some(Call::Generator),
185        (false, false) => Some(Call::Immediate),
186        (true, true) => Some(Call::Stream),
187        (false, true) => Some(Call::Async),
188    })
