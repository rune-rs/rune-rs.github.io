29pub(crate) struct Name<'a, T> {
30    pub(crate) span: Option<Span>,
31    pub(crate) value: &'a T,
32    pub(crate) ty: syn::Type,
33    pub(crate) method: NameMethod,
34    pub(crate) format_with: Option<&'a (Span, syn::Path)>,
35}
36
37impl<T> Name<'_, T> {
38    pub(crate) fn expr(&self, ident: syn::Ident) -> syn::Expr {
39        match self.method {
40            NameMethod::Unsized(..) => syn::parse_quote!(#ident),
41            NameMethod::Sized => syn::parse_quote!(&#ident),
42        }
43    }
44
45    pub(crate) fn ty(&self) -> Cow<'_, syn::Type> {
46        match self.method {
47            NameMethod::Unsized(..) => {
48                let ty = &self.ty;
49                Cow::Owned(syn::parse_quote!(&#ty))
50            }
51            NameMethod::Sized => Cow::Borrowed(&self.ty),
52        }
53    }
54
55    pub(crate) fn name_format(&self, value: &syn::Ident) -> syn::Expr {
56        match self.format_with {
57            Some((_, path)) => syn::parse_quote!(#path(&#value)),
58            None => syn::parse_quote!(&#value),
59        }
60    }
61}

63#[derive(Default, Clone, Copy)]
64pub(crate) enum NameMethod {
65    /// Load the tag by value.
66    #[default]
67    Sized,
68    /// Load the tag by visit.
69    Unsized(UnsizedMethod),
70}
71
72impl NameMethod {
73    pub(crate) fn name_all(&self) -> Option<NameAll> {
74        match self {
75            Self::Sized => None,
76            Self::Unsized(_) => Some(NameAll::Name),
77        }
78    }
79}

81impl Parse for NameMethod {
82    fn parse(input: ParseStream<'_>) -> Result<Self, syn::Error> {
83        let string: syn::LitStr = input.parse()?;
84        let s = string.value();
85
86        match s.as_str() {
87            "sized" => Ok(Self::Sized),
88            "unsized" => Ok(Self::Unsized(UnsizedMethod::Default)),
89            "unsized_bytes" => Ok(Self::Unsized(UnsizedMethod::Bytes)),
90            _ => Err(syn::Error::new_spanned(
91                string,
92                "#[musli(name(method = ..))]: Bad value, expected one of \"value\", \"unsized\", \"unsized_bytes\"",
93            )),
94        }
95    }
96}

149impl<'a> Expander<'a> {
150    pub(crate) fn new(input: &'a syn::DeriveInput, default_crate: &str) -> Self {
151        fn fields<'a>(cx: &Ctxt, fields: &'a syn::Fields) -> Vec<FieldData<'a>> {
152            fields
153                .iter()
154                .enumerate()
155                .map(|(index, field)| FieldData {
156                    span: field.span(),
157                    index,
158                    attr: attr::field_attrs(cx, &field.attrs),
159                    ident: field.ident.as_ref(),
160                    ty: &field.ty,
161                })
162                .collect()
163        }
164
165        let cx = Ctxt::new();
166        let type_attr = attr::type_attrs(&cx, &input.attrs);
167
168        let data = match &input.data {
169            syn::Data::Struct(st) => Data::Struct(StructData {
170                name: syn::LitStr::new(&input.ident.to_string(), input.ident.span()),
171                fields: fields(&cx, &st.fields),
172                kind: match &st.fields {
173                    syn::Fields::Unit => StructKind::Empty,
174                    syn::Fields::Unnamed(f) => StructKind::Indexed(f.unnamed.len()),
175                    syn::Fields::Named(..) => StructKind::Named,
176                },
177            }),
178            syn::Data::Enum(en) => {
179                let variants = en
180                    .variants
181                    .iter()
182                    .enumerate()
183                    .map(|(index, variant)| VariantData {
184                        span: variant.span(),
185                        index,
186                        name: syn::LitStr::new(&variant.ident.to_string(), variant.ident.span()),
187                        attr: attr::variant_attrs(&cx, &variant.attrs),
188                        ident: &variant.ident,
189                        fields: fields(&cx, &variant.fields),
190                        kind: match &variant.fields {
191                            syn::Fields::Unit => StructKind::Empty,
192                            syn::Fields::Unnamed(f) => StructKind::Indexed(f.unnamed.len()),
193                            syn::Fields::Named(..) => StructKind::Named,
194                        },
195                    });
196
197                Data::Enum(EnumData {
198                    name: syn::LitStr::new(&input.ident.to_string(), input.ident.span()),
199                    variants: variants.collect(),
200                })
201            }
202            syn::Data::Union(..) => Data::Union,
203        };
204
205        let prefix = type_attr.crate_or_default(default_crate);
206
207        let default = vec![
208            ModeIdent {
209                kind: ModeKind::Binary,
210                path: syn::Path::from(syn::PathSegment::from(syn::Ident::new(
211                    "Binary",
212                    Span::call_site(),
213                ))),
214            },
215            ModeIdent {
216                kind: ModeKind::Text,
217                path: syn::Path::from(syn::PathSegment::from(syn::Ident::new(
218                    "Text",
219                    Span::call_site(),
220                ))),
221            },
222        ];
223
224        Self {
225            input,
226            cx,
227            type_attr,
228            data,
229            prefix,
230            default,
231        }
232    }

243    fn setup_builds<'b>(
244        &'b self,
245        modes: &'b [ModeIdent],
246        tokens: &'b Tokens<'b>,
247        only: Only,
248    ) -> Result<Vec<Build<'b>>> {
249        let mut builds = Vec::new();
250
251        let mut missing = BTreeMap::new();
252
253        for default in &self.default {
254            missing.insert(&default.kind, default);
255        }
256
257        let (lt, lt_exists) = 'out: {
258            if let Some(lt) = self.input.generics.lifetimes().next() {
259                break 'out (lt.lifetime.clone(), true);
260            }
261
262            (syn::Lifetime::new("'__de", Span::call_site()), false)
263        };
264
265        let (allocator_ident, allocator_exists) = 'out: {
266            for p in self.input.generics.type_params() {
267                if p.ident == "A" {
268                    break 'out (p.ident.clone(), true);
269                }
270            }
271
272            (
273                self.cx.type_with_span_permanent("__A", Span::call_site()),
274                false,
275            )
276        };
277
278        let p = Parameters::generic(lt, lt_exists, allocator_ident, allocator_exists);
279
280        for mode_ident in modes {
281            missing.remove(&mode_ident.kind);
282
283            let expansion = Expansion { mode_ident };
284
285            let mode = expansion.as_mode(tokens, only);
286            let p = self.decorate(&p, &mode, tokens);
287
288            builds.push(crate::internals::build::setup(
289                self, expansion, mode, tokens, p,
290            )?);
291        }
292
293        for (_, mode_ident) in missing {
294            let expansion = Expansion { mode_ident };
295
296            let mode = expansion.as_mode(tokens, only);
297            let p = self.decorate(&p, &mode, tokens);
298
299            builds.push(crate::internals::build::setup(
300                self, expansion, mode, tokens, p,
301            )?);
302        }
303
304        Ok(builds)
305    }

307    fn decorate(&self, p: &Parameters, mode: &Mode<'_>, tokens: &Tokens<'_>) -> Parameters {
308        let (lt, lt_exists) = 'out: {
309            let list = self.type_attr.decode_bounds_lifetimes(mode);
310
311            if let [_, rest @ ..] = list {
312                for &(span, _) in rest {
313                    self.cx
314                        .error_span(span, "More than one decoder lifetime bound is specified");
315                }
316            }
317
318            if let Some((_, ty)) = list.first() {
319                break 'out (ty, false);
320            }
321
322            (&p.lt, p.lt_exists)
323        };
324
325        let (allocator_ident, allocator_exists) = 'out: {
326            let list = self.type_attr.decode_bounds_types(mode);
327
328            if let [_, rest @ ..] = list {
329                for &(span, _) in rest {
330                    self.cx
331                        .error_span(span, "More than one decoder allocator bound is specified");
332                }
333            }
334
335            if let Some((_, ty)) = list.first() {
336                break 'out (ty, false);
337            }
338
339            (&p.allocator_ident, p.allocator_exists)
340        };
341
342        let mut p = Parameters::generic(
343            lt.clone(),
344            lt_exists,
345            allocator_ident.clone(),
346            allocator_exists,
347        );
348
349        if let Some((allocator, span)) = self.fixed_allocator(mode, tokens) {
350            if allocator_exists {
351                self.cx.error_span(
352                    span,
353                    format_args!(
354                        "#[{ATTR}] the allocator cannot be pinned for a type which already has an \
355                         `A` allocator parameter"
356                    ),
357                );
358            }
359
360            if let Some(&(span, _)) = self.type_attr.decode_bounds_types(mode).first() {
361                self.cx.error_span(
362                    span,
363                    format_args!(
364                        "#[{ATTR}] the allocator parameter cannot be named when the allocator is \
365                         pinned to a concrete type"
366                    ),
367                );
368            }
369
370            p.allocator = allocator;
371            p.allocator_fixed = true;
372        }
373
374        p
375    }

377    /// Resolve the concrete allocator to use, as specified through
378    /// `#[musli(allocator = <type>)]` or `#[musli(global)]`.
379    fn fixed_allocator(&self, mode: &Mode<'_>, tokens: &Tokens<'_>) -> Option<(syn::Type, Span)> {
380        let allocator = self.type_attr.allocator(mode);
381        let global = self.type_attr.global(mode);
382
383        match (allocator, global) {
384            (Some((span, ty)), None) => Some((ty.clone(), *span)),
385            (None, Some(&(span, ()))) => {
386                let global = tokens.global;
387                Some((syn::parse_quote_spanned!(span => #global), span))
388            }
389            (Some(_), Some(&(span, ()))) => {
390                self.cx.error_span(
391                    span,
392                    format_args!(
393                        "#[{ATTR}(global)] cannot be combined with #[{ATTR}(allocator = <type>)]"
394                    ),
395                );
396
397                None
398            }
399            (None, None) => None,
400        }
401    }

403    /// Expand Encode implementation.
404    pub(crate) fn expand_encode(&self) -> Result<TokenStream> {
405        let modes = self.cx.modes();
406        let tokens = self.tokens();
407        let builds = self.setup_builds(&modes, &tokens, Only::Encode)?;
408
409        let mut out = TokenStream::new();
410
411        for build in builds {
412            out.extend(crate::en::expand_encode_entry(&build)?);
413        }
414
415        Ok(out)
416    }

418    /// Expand Decode implementation.
419    pub(crate) fn expand_decode(&self) -> Result<TokenStream> {
420        let modes = self.cx.modes();
421        let tokens = self.tokens();
422        let builds = self.setup_builds(&modes, &tokens, Only::Decode)?;
423

425
426        for build in builds {
427            out.extend(crate::de::expand_decode_entry(&build)?);
428        }
429
430        Ok(out)
431    }

434/// A thing that determines how it's tagged.
435pub(crate) trait Taggable {
436    /// The span of the taggable item.
437    fn span(&self) -> Span;
438    /// The rename configuration the taggable item currently has.
439    fn name(&self, mode: &Mode<'_>) -> Option<&(Span, syn::Expr)>;
440    /// The index of the taggable item.
441    fn index(&self) -> usize;
442}
443
444/// Expand the given configuration to the appropriate tag expression.
445pub(crate) fn expand_name(
446    taggable: &dyn Taggable,
447    mode: &Mode<'_>,
448    name_all: NameAll,
449    ident: Option<&syn::Ident>,
450) -> (syn::Expr, Option<Span>) {
451    let lit = 'out: {
452        if let Some(&(span, ref rename)) = taggable.name(mode) {
453            return (rename.clone(), Some(span));
454        }
455
456        if let (Some(ident), name_all) = (ident, name_all) {
457            let ident = ident.to_string();
458            let ident = ident.trim_start_matches("r#");
459
460            if let Some(name) = name_all.apply(ident) {
461                break 'out syn::LitStr::new(&name, ident.span()).into();
462            }
463        }
464
465        usize_suffixed(taggable.index(), taggable.span()).into()
466    };
467
468    let expr = syn::Expr::Lit(syn::ExprLit {
469        attrs: Vec::new(),
470        lit,
471    });
472
473    (expr, None)
474}

476/// Ensure that the given integer is usize-suffixed so that it is treated as the
477/// appropriate type.
478pub(crate) fn usize_suffixed(index: usize, span: Span) -> syn::LitInt {
479    syn::LitInt::new(&format!("{index}usize"), span)
480}
